#include <io.h>   
#include <stdlib.h>   
#include <stdio.h>   
#include <string>   

unsigned long Table[256];   

char Never[]={0x0d,'N','e','v','e','r',0x0d,'2','2',0};   

char Name[255];   

char YourName[40];   

int main(int argc, char* argv[])   

{   

    unsigned long eax,edi,edx;   

    FILE *fp;   

    fp = fopen(".\\table.bin","rb");   

    if (fp==NULL)   

        return 0;   

    fread(Table,sizeof(unsigned long),256,fp);   

    fclose(fp);   

    printf("Wrtie By leo_cyl\n");   

    printf("input your name :");   

    scanf("%s",YourName);   

    Name[0]=Name[1]='2';   

    int len = strlen(YourName);   

    for (int i=0;i<len;i++)   

        Name[i+2]=YourName[i];   

    for(i=0;i<strlen(Never);i++)   

        Name[len+2+i]=Never[i];   

    Name[len+2+i]=0;   

    eax=0xffffffff;   

    for(i=0;i<strlen(Name);i++) {   

        edx=0;   

        edi=eax;   

        edi=edi&0xff;   

        edx = (unsigned long)Name[i];   

        edx=edx^edi;   

        eax=eax>>8;   

        edx=Table[edx];   

        eax=eax^edx;   

    }   

    eax=~eax;   

    printf("Key= %u\n",eax);   

    return 0;   

}
