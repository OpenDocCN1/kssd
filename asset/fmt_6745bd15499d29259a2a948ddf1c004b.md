{
  char *buff;

  char tmp;
  int i;
  for(i=0;i<unknownsize;i++)
  {
    tmp=key[buff[i]];
    tmp-=(char)key1;  //key1 为文件头中的数据
    tmp-=i;
    buff[i]=tmp;
  }
}
