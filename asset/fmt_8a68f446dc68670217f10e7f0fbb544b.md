Private Function StrToHex(ByVal s As String) As String
    Dim buff() As Byte
    Dim result As String
    Dim tmps As String
    '先将字串转换为 byte 数组，一个中文字符占两个字节，一个英文字符占一个字节
    buff = StrConv(s, vbFromUnicode)
    Dim i As Long
    For i = 0 To UBound(buff)
        '将各个字节转化为 hex 字串（每个字节两个字符，不足的前面补 0）
        tmps = Right("00" + Hex(buff(i)), 2)
        '连接各个 hex 字串
        result = result + tmps
    Next
    StrToHex = result
End Function
