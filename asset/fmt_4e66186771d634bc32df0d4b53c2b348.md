function THookClass.GetFuncRealBase(pFunc: Pointer): Pointer; 
var 
func:PImportCode; 
begin 
  Result:=pFunc; 
if pFunc=nilthen exit; 
try 
    func:=PImportCode(pFunc); 
if (func^.JumpInstruction=$25FF) then 
{指令二进制码 FF 25 汇编指令 jmp [...]} 
      result:=func^.AddressOfPointerToFunction^ 
else 
  result:=pFunc; 
except 
    Result:=nil; 
end;
