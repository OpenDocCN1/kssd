{

private void btnLoad_Click(object sender, EventArgs e)  

{

    BeginGame();  

    fs.Close();  

    resetTime();  

    tmrClock.Enabled = true;  

}

{

        tmrClock.Enabled = false;

        EndGame();  

       // resetTime();  

    }  

    else  

    {  

        if (sec != 0)

}
