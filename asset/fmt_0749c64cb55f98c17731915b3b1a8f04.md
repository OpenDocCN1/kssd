PVOID NTAPI GetJmpAddress(PVOID Fun,BOOL *Call_Code)  
{  
    PVOID Return=NULL;  
  BYTE *data=(BYTE *)Fun;  
  DWORD Old=0;  
  if(data[0]==0xe9)  
  {  
    *Call_Code=0;  
  }else if (data[0]==0xe8)  
  {  
    *Call_Code=1;  
  }else  
  {  
    *Call_Code=2;  
    return NULL;  
  }  
  memcpy((void *)&Old,&data[1],4);  
    _asm  
  {  
    mov eax,Old  
    mov edx,Fun  
    add eax,edx  
    add eax,5  
    mov Return,eax  
  }  
    return Return;  
}
