void CreateRegCode(HWND hDlg){

  if(result==0){  
    MessageBox(hDlg,TEXT("请输入用户名！"),TEXT("Error!"),MB_OK);

  memset(strReg,0,STR_LEN);  
  memset(strCode,0,STR_LEN);  
  int sum=0;  
  int k=1;  
  int len=strlen(strName);  
  for(int i=0;i<len;i++){  
    if(strName[i]>='a' && strName[i]<='z')  
      strName[i]-=32;  
    sum+=(strName[i]*k);  
    k++;  
  }  
  k=1;  
  for(int i=0;k<=12;){  
    int tmp=sum;  
    tmp+=k;  
    tmp%=0xC8;

    tmp^=tmp2;  
    char strTmp[5];

    strncat(strCode,strTmp,2);  
    tmp=k^strName[i];  
    sum+=tmp;  
    k++;  
    i++;  
    if(i==len)  
      i=0;  
  }

}
