
# 注册码计算示例

第 5 位 `46 xor ((30C*2+3) shr 8) = 40(@)`

第 6 位 `42 xor ((65B*2+3) shr 8) = 4E(N)`

第 7 位 `46 xor ((D07*2+3) shr 8) = 5C(\)`

第 8 位 `46 xor ((1A6D*2+3) shr 8) = 72(r)`

总结：我的注册码就是：`0391@N\r`

# 内存注册机信息

**中断地址**：`4DACE4`

**中断次数**：`1`

**第一字节**：`58`

**字节长度**：`1`

**寄存器**：`EDX`（内存型）

# 代码实现

'唉，VB 的注册机也有不好写的时候![](img/6d0f69ca3b0526abc5f42199a1f1f0e8.jpg)，不像 C/C++或 Pascal 语言，连左、右逻辑移位的指令都没有，到最后还得自己添上![](img/d8960f6c7afa20ba0ca27920887f4540.jpg)。![](img/a7e67d4f8f4afe4c3cd0617906cb3d9d.jpg) ![](img/a7e67d4f8f4afe4c3cd0617906cb3d9d.jpg) ![](img/a7e67d4f8f4afe4c3cd0617906cb3d9d.jpg)

```vb
Public CF As Boolean '定义 CF 标志位

Public Function SHR(iShr As Long, n As Integer) As Long 'SHR 逻辑移位函数

  Dim i As Integer

  For i = 1 To n - 1

    iShr = iShr \ 2

  Next i

  CF = iShr And 1

  SHR = iShr \ 2

End Function

Private Sub Command1_Click()

  Dim Code, Sn As String

  Dim i, k As Integer

  Dim tmp, j As Long

  Code = Text1.Text

  If Code = "" Then

    MsgBox ("错误，请输入机器码！")

    Exit Sub

  End If

  j = 0

  tmp = SHR(j, 8)

  Sn = Chr(Asc(Mid(Code, 1, 1)) Xor tmp)

  j = j + Asc(Mid(Sn, 1, 1)) + 1

  For i = 2 To Len(Code)

    tmp = SHR((j * 2 + 3), 8)

    Sn = Sn + Chr(Asc(Mid(Code, i, 1)) Xor tmp)

    j = Asc(Mid(Sn, i, 1)) + (j * 2 + 3)

  Next i

  Text2.Text = Sn

End Sub
```
