ULONG_PTR GetExportRva(IN TCHAR *FileName, IN CHAR *FunctionName) 
{ 
   HMODULE hModule = LoadLibrary(FileName); 
   if (hModule == NULL) 
      return 0; 
   ULONG_PTR VA = (ULONG_PTR) GetProcAddress(hModule, 
      FunctionName); 
   FreeLibrary(hModule); 
   return (ULONG_PTR) (VA - (ULONG_PTR) hModule);
