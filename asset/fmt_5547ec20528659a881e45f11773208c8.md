var
  eax, ebx, edi : integer;
  Seax, Sal : string;

  key: array[0 .. 255] of byte =          //密文表全部数据，共 256 组
  ($20,$00,$20,$00,$20,$00,$20,$00,$20,$00,$20,$00,$20,$00,$20,$00,

  //////// 1 ///////循环
  edi := 1;//初始化 i 变量

  Seax := '000000';
  Sal  := int2hex(key[ord(cz[edi]) * 2], 2);
  Seax := Seax + Sal;

  eax  := eax and $8;
