Private Function Int2HexStr(IntNum As Long) As String
'将 10 进制整数变为 16 进制字符串
'输入:  IntNum 十进制整数
'输出:  格式化的字符串,前面填充零(共 8 个字符)
    Dim sTmp As String
    sTmp = CStr(Hex(IntNum))
    Int2HexStr = String(8 - Len(sTmp), "0") & sTmp
End Function
