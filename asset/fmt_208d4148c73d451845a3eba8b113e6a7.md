void fun2(unsigned char src[], unsigned len, unsigned char dst[])
{
    unsigned i, j;
    unsigned ebx = 0, edi = 0;
    unsigned eax, ecx, edx, esi;
    esi = search(src[0], Tab);
    eax = len / 3;
    edx = len % 3;
    if (edx == 1) ebx = 2;
    if (edx == 2) {
        if ((esi & 0x0c) == 0) ebx = -2;
        else ebx = 4;
    }
    for (i = 0, j = 0; i < len; i++) {
        esi = 0;
        esi = search(src[i], Tab);
        edi = edi << 4;
        ebx = ebx + 4;
        esi = esi + edi;
        edi = esi;

            eax = eax >> ecx;

            eax--;
