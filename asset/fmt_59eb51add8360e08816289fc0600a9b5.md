Private Sub Command1_Click()
  
  Dim name As String, code As String, T As String, F As String
  
  Dim i As Integer
  
  T = "True code!"
  
  F = "False code!"
  
  name = Text1.Text
  
  For i = 1 To Len(name)
    
    code = code & CStr(Asc(Mid(name, i, 1)))
  
  Next i
  
  If Text2.Text = code Then
    
    MsgBox T, vbOKOnly, "P-Code(2-2)"
  
  Else
    
    MsgBox F, vbOKOnly, "P-Code(2-2)"
  
  End If

End Sub
