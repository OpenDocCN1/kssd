```vb
Dim x, z, a, b, m As String  

Private Sub Command1_Click()  

x = Text1.Text  

If x = "" Then  

MsgBox "请输入序列号^\_^", 64, "注意"  

Else  

a = "23f3a"  

z = Mid(x, 1, 6)  

m = Format("&h" + z)  

b = Format("&h" + a)  

y = Val(m) + Val(b)  

Text2.Text = y  

End If  

End Sub  
```
