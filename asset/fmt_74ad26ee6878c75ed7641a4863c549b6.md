void change(char * get,char *tem)//get 为注册码,tem 为存放二进制的数组  
{  
    char *m="CA5BDWF4H9PJK3SUM2XEL7GRZNQTY6V8";  
    int length,i,j,le=0;  
    length=strlen(get);  
    for(i=0 ;i<length; i++)  
    {  
        for(j=0;j<32;j++)  
        {  
            if(get[i]==m[j])  
            {  
                le=j;  
                break;  
            }  
            else  
                le=0xff;  
        }  
        for(j=0;j<5;j++)  
        {  
            if((1&(le>>j))==1)  
            {  
                tem[i*5+j]=1;  
            }  
        }  
    }  
}
