```
unsigned char data\[\];//缓冲区
unsigned char a,b;
```

```
int data\[\]//缓冲区地址
int n=0x31746285;
int key=0xa8e7faf1;;

{
  n=ror(n,4);
  data\[i\]^=key;
  key=rol(key,\*(char\*)&n);
}
```
