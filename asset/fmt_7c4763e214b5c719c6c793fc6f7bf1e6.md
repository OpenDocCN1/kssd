
### 单链表头节点获取

```cpp
template<typename T>
inline T CSList<T>::GetHead() const  
{
    ASSERT(0 != m_nCount);  
    return m_pNodeHead->data;  
}
```

### 按位置获取元素（非 const）

```cpp
template<typename T>
inline T& CSList<T>::GetAt(const int pos)  
{
    ASSERT(1 <= pos && pos <= m_nCount);
    int i;  
    CNode<T> *pTmpNode = m_pNodeHead;
    for (i = 1; i < pos; ++i)  
    {
        pTmpNode = pTmpNode->next;  
    }
    return pTmpNode->data;  
}
```

### 按位置设置元素

```cpp
template<typename T>
inline void CSList<T>::SetAt(const int pos, T data)  
{
    ASSERT(1 <= pos && pos <= m_nCount);
    int i;  
    CNode<T> *pTmpNode = m_pNodeHead;
    for (i = 1; i < pos; ++i)  
    {
        pTmpNode = pTmpNode->next;  
    }
    pTmpNode->data = data;  
}
```

### 查找元素位置

```cpp
template<typename T>
inline int CSList<T>::Find(const T data) const  
{
    int i;  
    int nCount;  
    CNode<T> *pTmpNode = m_pNodeHead;
    nCount = m_nCount;  
    for (i = 0; i < nCount; ++i)  
    {
        if (data == pTmpNode->data)  
            return i + 1;  
        pTmpNode = pTmpNode->next;  
    }
    return 0;  
}
```
