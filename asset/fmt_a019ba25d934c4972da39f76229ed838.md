VAR  
A,EAX,TEMP:INTEGER;  

MN,SN,STR:STRING;  

BEGIN  

MN:='2654328163';//我的机器码  

EAX:=STRTOINT(SN);  

FOR A:=1 TO LENGTH(MN) DO  
BEGIN  
  TEMP:=STRTOINT(MN[A]);  
  EAX:=EAX XOR CRC_TABLE[TEMP];  
END;  

FOR A:=0 TO $80 DO  
BEGIN  
   EAX:=EAX XOR CRC_TABLE[A];  
   EAX:=EAX XOR CRC_TABLE[$200+A];  
END;  

//因为对所有的数据进行 XOR 所以等价于 XOR 一个值;  

//等价于 EAX:=EAX XOR $DB3F761;  

STR:=INTTOSTR(EAX);  

STR:=STR[LENGTH(STR)-1]+STR[LENGTH(STR)]+STR;  

STR:=COPY(STR,1,8);  

IF STR=COPY(MN,1,8) THEN SHOWMESSBOX('成功')  
ELSE SHOWMESSAGEBOX('失败');  

END;
