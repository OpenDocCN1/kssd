```assembly
:00408651 0FBE1C38            movsx ebx, byte ptr [eax+edi]

:00408655 03DA                add ebx, edx
; 此时：EDX=26H //ebx=121+26

:00408657 03CB                add ecx, ebx
; //ecx=ebx+ecx

:00408659 40                  inc eax
; //eax=eax+1

:0040865A 3BC6                cmp eax, esi
; //if eax<esi

:0040865C 72F3                jb 00408651
; 依次循环
```
