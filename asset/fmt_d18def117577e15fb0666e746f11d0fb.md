private static byte[] doXor(byte[] key, byte[] buf) {

    byte[] ret = null;

    if (key != null && buf != null) {

        int klen = key.length;

        int blen = buf.length;

        int j = 0;

        ret = new byte[blen];

        for (int i = 0; i < blen; i++) {

            ret[i] = (byte) (buf[i] ^ key[j]);

            j++;

            if (j >= klen) {

                j = 0;

            }

        }

    }

    return ret;

}

private static void doExchange4Bits(byte[] buf) {

    if (buf != null) {

        int len = buf.length;

        int h, l, ret;

        for (int i = 0; i < len; i++) {

            l = (buf[i] & 0x0f);

            h = (buf[i] & 0x0f0);

            ret = ((l << 4) + (h >> 4));

            buf[i] = (byte) ret;

        }

    }

}
