```c
void decodeShellcode()
{
  _asm
  {
    jmp decode_end
decode_start:
    pop edx
    dec edx
    xor ecx,ecx
    mov cx,260  
decode_loop:
    xor byte PTR[edx+ecx],KEY
    loop decode_loop
    jmp decode_ok
decode_end:
    call decode_start
decode_ok:
  }
}
```
