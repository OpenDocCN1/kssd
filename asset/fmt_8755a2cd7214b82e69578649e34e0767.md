void AddHash(WORD result[], WORD subkey[], int carry)
{
  DWORD ebx, edi;
  for(int i=0; i<Len; i++)
  {
    ebx=DWORD(result[i]);
    edi=DWORD(subkey[i]);
    ebx+=edi;
    carry=(carry&0xFF);
    ebx+=carry;
    result[i]=WORD(ebx);
    if((ebx&0x10000)!=0)
      carry=1;
    else  
      carry=0;
    carry=(carry&1);
  }
}
