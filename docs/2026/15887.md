```
function hook\_sign() {
var a = Module.findExportByName("libsignature.so", "\_ZN12HNASignature4signEPhS0\_");
console.log('hook\_sign: ',a);

this.arg0 = args[0]
console.log("hook\_sign arg2==: "+ptr(args[2]).readCString()+" arg3==: "+ptr(args[3]).readCString());
},
onLeave: function (retval) {
console.log('hook\_sign onLeave: ', 'retval: ', Memory.readCString(this.arg0.readPointer()))
}
});
}
function hook\_hmacSha() {
var base\_libsinature = Module.findBaseAddress("libsignature.so");
console.log('base1: ',base\_libsinature);
var a = Module.findExportByName("libsignature.so", "\_ZN12HNASignature7hmacShaEPhS0\_");
console.log('hook\_hmacSha: ',a);
Interceptor.attach(a, {
onEnter: function(args) {
console.log("hook\_hmacSha param: "+ptr(args[2]).readCString());
console.log("hook\_hmacSha key: "+ptr(args[3]).readCString());
},
onLeave: function (retval) {
// console.log('hook\_hmacSha onLeave: ', 'retval: ', ptr(retval).readCString())
}
});
console.log("hook\_hmacSha...");
}
```
