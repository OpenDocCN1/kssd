PULONG NtfsMajorFunction = NULL;
ULONG NtfsDispatch = 0xfa8f3618; //Ntfs!NtfsFsdSetInformation

BOOLEAN GetNtfsMajorFunctionAddr()
{
        PDRIVER\_OBJECT        pDriverObject;
        UNICODE\_STRING        DeviceName;
        NTSTATUS                ntStaus;
        BOOLEAN                        bRet = FALSE;
        RtlInitUnicodeString(&DeviceName, L"\\FileSystem\\Ntfs");
        ntStaus = ObReferenceObjectByName(&DeviceName, OBJ\_CASE\_INSENSITIVE, NULL, 0, \*IoDriverObjectType, KernelMode, NULL, (PVOID\*)&pDriverObject);
        if(!NT\_SUCCESS(ntStaus))
                goto \_\_end;
        NtfsMajorFunction = (PULONG)pDriverObject->MajorFunction;
        bRet = TRUE;
\_\_end:
        return bRet;
}

NtfsDispatch = NtfsMajorFunction[IRP\_MJ\_SET\_INFORMATION];
