void \_InitApiDate()
{
        HINSTANCE hInstLibrary;
        CHAR\*        cDllName;
        CHAR\*        cApiName;
        INT                nFindId;

        for(INT i = 0;i <= CALL\_BUFFER\_NO; i++)
        {
                if( m\_szRetCall[i].wId != 0 &&
                        m\_szRetCall[i].w1stId != 0 &&
                        m\_szRetCall[i].dwApiAddr == 0 &&
                        m\_szRetCall[i].w2secId != 0
                        )
                {
                        nFindId = \_FidUseId(m\_szRetCall[i].w1stId);
                        cDllName = m\_szRetCall[nFindId].szRetBuffer;
                        hInstLibrary = LoadLibrary(cDllName);
                        nFindId = \_FidUseId(m\_szRetCall[i].w2secId);
                        cApiName = m\_szRetCall[nFindId].szRetBuffer;
                        m\_szRetCall[i].dwApiAddr = (DWORD)GetProcAddress(hInstLibrary,cApiName);
                        FreeLibrary(hInstLibrary);
                }
        }

        return;
