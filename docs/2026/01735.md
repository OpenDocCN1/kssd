  ;StrToHex是将内存中的ASCII码以十六进制形式转成一个整数（HEX）
  ;lpszStr指向要转换的ASCII码，iStrlen指向Str的长度。
  ;如果该ASCII码中包含有非数字或大于'f'的数则返回0
  ;>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>
  StrToHex proc uses esi edi ebx lpszStr,iStrLen
    xor esi,esi
    xor edx,edx
    xor ebx,ebx
    mov edi,iStrLen
  @@:
    mov eax,lpszStr
    movzx eax,BYTE ptr [eax+esi]
    test al,al
    jz @f
    .if ((al >= 'A') && (al <= 'F')) || ((al >= 'a') && (al <= 'f'))
      sub al,'W'
      adc dl,dl
      shl dl,5
      add al,dl
      jmp Next
    .elseif ((al >= '0') && (al <= '9'))
      sub al,'0'
  Next:  lea ecx,[edi-1]
      and eax,0fh
      shl ecx,2
      shl eax,cl
      add ebx,eax
      dec edi
      inc esi
      jmp @b
    .else
      xor eax,eax
      ret
    .endif
  @@:
    mov eax,ebx
    ret
  StrToHex endp
