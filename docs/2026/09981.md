注册算法为: 依次把用户名每个字母转换为ASCII码后再转换为十进制数值串起来即可。

```vb
  Private Sub Text1_Change()
  'Keygen by KuNgBiM
      Dim name As String
      Dim n, m, i As Byte
      Dim sn, tmp  As String
      name = Text1.Text
      n = Len(name)
      For i = 1 To n
          tmp = Mid(name, i, 1)
          m = Asc(tmp)
          sn = sn & m
      Next
      Text2.Text = sn
  End Sub
```
