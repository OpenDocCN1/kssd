//该结构是try except 块的向量表（一数组），是未公开的

{
        DWORD vFlag;
        PVOID TryHandler;/\*try-except{code}\*/
        PVOID NextContiueCode;/\*异常处理后的下一条指令地址\*/

}vect\_handler;

typedef struct \_\_EXCEPTION\_LIST
{
        \_\_EXCEPTION\_LIST\* next;//next一个SEH链
        PVOID             handler;//异常处理程序
        vect\_handler\*     vecthandler;
        DWORD             IndexVechandler;/\*当前处理的try-except块号,抛出异常（在try-except里） 都会修改它\*/

}EXCEPTION\_LIST;

char\* ff = "try";

DWORD WINAPI handler()
{
        /\*突破except{}里不能定义局部变量，增加局部变量\*/
        /\*调试器监测\*/
        /\*加解密TRY 块代码\*/

                \_\_asm push 0
                \_\_asm push 0
                \_\_asm push ff
                \_\_asm push 0
                \_\_asm mov eax,MessageBoxA

                \_\_asm popad

};

void   Contiue()
