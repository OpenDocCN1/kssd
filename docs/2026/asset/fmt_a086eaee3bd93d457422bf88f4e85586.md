void printbig(const char* prefix, big x)

{

	char buf[256];

	int len = big_to_bytes(0, x, buf, FALSE);

	printf("%s: ", prefix);

	for (int i=0; i<len; ++i) {

		printf("%02x",(BYTE)buf[i]);

	}

	printf("\n");

}
