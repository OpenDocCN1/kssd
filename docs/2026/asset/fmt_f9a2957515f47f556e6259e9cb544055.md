
```go
package main
import "fmt"
func main() {
var a, b int
a, b = sayHello(123)
a = a + 1
b = b + 2
fmt.Println(a, b)
}
func sayHello(a int)(i,j int){
i = a + 1
fmt.Println("execute half")
j = a + 2
return
}
```

## 多个参数，多个返回值

```go
package main
import (
"fmt"
)
func main() {
var a, b int
a, b = sayHello(1234, 5678)
a = a + 1
b = b + 2
fmt.Println(a, b)
}
func sayHello(a int, b int)(i,j int){
i = a + 1
fmt.Println("execute half")
j = b + 2
return
}
```
