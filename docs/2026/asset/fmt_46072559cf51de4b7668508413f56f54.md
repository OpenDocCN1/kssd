#include <stdio.h>
#include <Windows.h>
#include <Ntsecapi.h>

void InitLsaString(PLSA_UNICODE_STRING key_name, LPCWSTR lpString)
{
        int str_len;

        if (lpString) {
                str_len = 2 * lstrlenW(lpString);
                key_name->Length = str_len;
                key_name->Buffer = (PWSTR)lpString;
                key_name->MaximumLength = str_len + 2;
        } else {
                key_name->Buffer = 0;
                key_name->Length = 0;
                key_name->MaximumLength = 0;
        }
}

int LsCsp_RetrieveSecret(const WCHAR *key, void *a2, int a3)
{
        LSA_UNICODE_STRING    KeyName;
        LSA_OBJECT_ATTRIBUTES ObjectAttributes = {0};
        LSA_HANDLE            PolicyHandle = NULL;
        PLSA_UNICODE_STRING   Buffer = NULL;

        InitLsaString(&KeyName, key);

        if (LsaOpenPolicy(NULL, &ObjectAttributes, POLICY_GET_PRIVATE_INFORMATION, &PolicyHandle))
                return -1;

        LsaRetrievePrivateData(PolicyHandle, &KeyName, &Buffer);
        LsaClose(PolicyHandle);

        printf("RSA priv_key at:%p [%d bytes]\n", Buffer->Buffer, Buffer->Length);

        return 0;
}

int main(int argc, char** argv)
{
        LsCsp_RetrieveSecret(L"L$HYDRAENCKEY_28ada6da-d622-11d1-9cb9-00c04fb16e75", 0, 32);
        return 0;
}
