def get_sign(data, key):
    key = key.encode('utf-8')
    message = data.encode('utf-8')
    sign = base64.b64encode(hmac.new(key, message, digestmod=sha256).digest()).decode()

function stringToByte (str) {
    var ch, st, re = [];
    for (var i = 0; i < str.length; i++ ) {
        ch = str.charCodeAt(i);
        st = [];
        do {
            st.push( ch & 0xFF );
            ch = ch >> 8;
        } while ( ch );
        re = re.concat(st.reverse());
    }
    return re;
}

function getwwoken(data){
    var result = response._genericNt3(1, stringToByte(data));
    return result;
}

rpc.exports = {
    getwtoken:getwwoken
}
