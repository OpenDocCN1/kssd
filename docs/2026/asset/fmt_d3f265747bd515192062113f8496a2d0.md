```
Boolean        _VM_Analyze::Decode()
{
        /*
        LODSB
        XOR     AL, BL
        XOR     AL, 53H
        SUB     AL, 9FH
        SUB     BL, AL
        MOVZX   EAX, AL
        JMP     DWORD PTR [EDI+EAX*4]
        */
        Byte        Opcode=ReadByte();
        Byte        Key=ContextKey;
        Opcode=Opcode^Key;
        Opcode=Opcode^0x53;
        Opcode=Opcode-0x9F;
        Key=Key-Opcode;
        ContextKey=ContextKey&0xFFFFFF00|Key;
        switch (Opcode)
        {
        default:
                {
                        UInt32        K=ContextKey;
                        UInt32        P=CodePointer;
                        String        ^s=AnalyzeResult->ToString();
                        throw        s;
                }
        }
}
```

`OllyICE` 加载目标 下断`MainLoop` `JumpMethodTable`

`VS2008` `F5` 抛出异常 `Opcode=0x??` `ContextKey=0x????????`

`OllyICE` `JumpMethodTable` 条件断 `AL==0x??`

对比`ContextKey==EBX` 恭喜清洗代码应该没有什么`BUG`

按照`Opcode==0x??`拔下对应的`Method`代码

清洗之

一般都是

```
LODS?
(ADD|SUB|???) EAX, EBX
(ADD|???????) EAX, 0x???????
(ADD|???????) EAX, 0x???????
(ADD|???????) EBX, EAX
???????
???????
JMP MainLoop
```

清洗后的代码贴到`switch(Opcode)`里面

```
case        0x2E:
        /*
        LODSB
        XOR     AL, BL
        ADD     AL, 0CAH
        ADD     AL, 1BH
        ADD     BL, AL
        MOVZX   EAX, AL
        LEA     EAX, [EDI+EAX*4]
        PUSH        EAX
        */
        {
                Byte        arg1=ReadByte();
                Byte        Key=ContextKey;
                arg1=arg1^Key;
                arg1=arg1+0xCA;
                arg1=arg1+0x1B;
                Key=Key+arg1;
                ContextKey=ContextKey&0xFFFFFF00|Key;
                AnalyzeResult->WriteLine(String::Format("PUSH\t{0}",GetHexString(ContextBase+arg1*4,4)));
        }
        return        true;
```

扒掉取`arg1`和解码更新`Key`的代码
