### 对应C语言代码如下

```c
int _CalcRetComb(
        char* lpRet1,
        char* lpRet2,
        int   nLenRet1,
        int   nLenRet2,
        char** lppRet
        )
{
        int i ,j;
        char* lpBufRet = new char[nLenRet1+nLenRet2];
        for (i = 0 ;i < nLenRet1 ;i++)
        {
                j = 0;
                while (lpRet1[i] != szRand1[j])
                {
                        if (++j >= 0x10)
                                break;
                }
                lpBufRet[i] = j ^ 0x23;
        }
        for (i = 0 ;i < nLenRet2 ;i++)
        {
                j = 0;
                while (lpRet2[i] != szRand1[j])
                {
                        if (++j >= 0x10)
                                break;
                }
                lpBufRet[i+nLenRet1] = j ^ 0x23;
        }
        *lppRet = lpBufRet;
        return nLenRet1+nLenRet2;
}
```
