# Telephone

目标：获得合约所有权

## 源码

```
pragma solidity ^0.4.18;

contract Telephone {

  address public owner;

  function Telephone() public {

    owner = msg.sender;

  }//构造函数,部署的人是合约的所有者

  function changeOwner(address _owner) public {

    if (tx.origin != msg.sender) {

      owner = _owner;

    }//最初调用合约的人与调用者不一样的话,就把合约的所有者改成_owner

  }

}
```
