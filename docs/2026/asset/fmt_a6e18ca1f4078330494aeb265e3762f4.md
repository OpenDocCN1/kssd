import base64

key0 = 98765432109876543210123

part2_bi = 0

def decode_part1():
    global part1_bi
    part1_bi = key1 / key0
    partstr = bytearray(str(part1_bi))
    for i in xrange(len(partstr)):
        partstr[i] ^= ord(xordt[i])
    return partstr

def decode_part2():
    global part2_bi
    part2_bi = long(gmpy2.isqrt(key2))
    partstr = bytearray(str(part2_bi))
    for i in xrange(len(partstr)):
        partstr[i] ^= ord(xordt[i])
    return partstr
