double CEntropyMI::calculate() {
  double entropy = 0;
  DWORD dwMapSize = g_GlobalInfo.GetSize();
  for (int i = 0; i < 256; i++)
  {
    double p_x = double(g_GlobalInfo.count(((char)i))) / dwMapSize;
    double p_y = (double)1 / 256;
    double p_x_y = p_x * p_y;
    if (p_x > 0)
      entropy += -(p_x * p_x_y) * (log(((p_x * p_x_y) / p_y)));
  }
  return entropy;
}
