get_addr32crc:

    pushad

    mov ebx,[esp+8*4+4]

    mov esi,[esp+8*4+8]

    mov edx,ebx

    add edx,[edx.mz_neptr]

    add edx,ebx

    mov edi,[edx.ex_addresstablerva]

    add edi,ebx

    mov edi,[edx.ex_namepointersrva]

    add edi,ebx

    

    push edx

    mov edx,edi

    

@@next:

    add edx,ebx

    pushad

@@1: 

    xor al, [edx]

    mov bl, 8 

@@2:

    shr eax, 1 

    jnc @@3 

    xor eax, 0EDB88320h 

@@3:

    dec bl 

    jnz @@2 

    loop @@1 

    not eax 

@@4:

    mov [esp+pushad_eax],eax

    popad
