```python
import csv

from pathlib import Path

with open("dumpinfo.csv") as csvfile:
    reader = csv.DictReader(csvfile)
    for row in reader:
        icon = row['icon']
        name = row['name']
        for f in Path().rglob(icon+'.png'):
            f.rename(f.with_name('{}-{}.png'.format(icon, name)))
```
