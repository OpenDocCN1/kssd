我们先来分析被感染程序的Loader

用到的IDA里的解码脚本

- -其实渣质量的.

```
mov     eax, 0EAB8D //decode key
mov     edx, 0041901C //解码起始位置
mov     esi, 598 //固定大小

L007:
xor     dword ptr [edx+esi], eax
dec     esi
sub     esi, 3
jnz     L007
```

```
#include <idc.idc>
static decrypt(end, key)
{
    auto i, x;
    end=end+0x598;
    for (i=0x598;i!=0;i=i-4)
    {
        x=Dword(end);
        x= (x^key);
        PatchDword(end,x);
        end = end - 4;
    }
}
```

执行`decrypt(0x41901c,0x0EAB8D)`即可

既然是解码,我们就把这段叫作`DecodeProc`吧……解码段最后跳去执行Loader的`main`函数了
