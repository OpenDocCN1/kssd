```c
extern "C" __declspec(dllexport)  void DropFileAdd();    //声明输出响应拖放消息并处理的函数
char ScanningWindow[] = "Keygener_Assistant v2.1.2 - Scanning : Hash & Crypto Detector"; //相应的窗口名

je _loop_over
add eax,2
inc ecx
jmp _looper
//以下为保存文件路径
_loop_over:
push 0
push 0
push 0xfff        //多字符串的最大长度
```
