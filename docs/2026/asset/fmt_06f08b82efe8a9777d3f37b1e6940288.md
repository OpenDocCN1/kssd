macro pack\_ilm(x) {
x = zext((I4 << 4) | (I3 << 3) | (I2 << 2) | (I1 << 1) | (I0 << 0));
}
macro unpack\_ilm(x) {
I4 = (x & 0x10)!=0;
I3 = (x & 0x8)!=0;
I2 = (x & 0x4)!=0;
I1 = (x & 0x2)!=0;
I0 = (x & 0x1)!=0;
}
macro pack\_scr(x) {
x = zext((D1 << 2) | (D0 << 1) | (T << 0));
}
macro unpack\_scr(x) {
D1 = (x & 0x4)!=0;
D0 = (x & 0x2)!=0;
T = (x & 0x1)!=0;
}
macro pack\_ccr(x) {
x = zext((S << 5) | (I << 4) | (N << 3) | (Z << 2) | (V << 1) | (C << 0));
}
macro unpack\_ccr(x) {
S = (x & 0x20)!=0;
I = (x & 0x10)!=0;
N = (x & 0x8)!=0;
Z = (x & 0x4)!=0;
V = (x & 0x2)!=0;
C = (x & 0x1)!=0;
}
macro pack\_ps(x) {
x = zext(
(I4 << 20) | (I3 << 19) | (I2 << 18) | (I1 << 17) | (I0 << 16) |
(D1 << 10) | (D0 << 9) | (T << 8) |
(S << 5) | (I << 4) | (N << 3) | (Z << 2) | (V << 1) | (C << 0)
);
}
macro unpack\_ps(x) {
#unpack ILM
I4 = (x & 0x100000)!=0;
I3 = (x & 0x80000)!=0;
I2 = (x & 0x40000)!=0;
I1 = (x & 0x20000)!=0;
I0 = (x & 0x10000)!=0;
#unpack SCR
D1 = (x & 0x400)!=0;
D0 = (x & 0x200)!=0;
T = (x & 0x100)!=0;
#unpack CCR
S = (x & 0x20)!=0;
I = (x & 0x10)!=0;
N = (x & 0x8)!=0;
Z = (x & 0x4)!=0;
V = (x & 0x2)!=0;
C = (x & 0x1)!=0;
}
