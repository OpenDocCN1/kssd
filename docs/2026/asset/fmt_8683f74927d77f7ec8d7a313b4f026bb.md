        size_t len;
        if (put_user((elf_addr_t)p, sp++))
                return -EFAULT;
        len = strnlen_user((void __user *)p, MAX_ARG_STRLEN);
        if (!len || len > MAX_ARG_STRLEN)
                return -EINVAL;
        p += len;
}

        return -EFAULT;

mm->arg_end = p;
