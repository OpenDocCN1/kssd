#include <cstdlib>
#include <cstdio>
#include <cstring>

int numShift(int *num)
{
    int temp;
    temp = 0x343FD * *num + 0x269EC3;
    *num = temp;
    return ((unsigned int)temp >> 16) & 0x7FFF;
}
