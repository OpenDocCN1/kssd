首先我们在 `user32` 中寻找 `HMValidateHandle` 函数, 该函数在 `IsMenu` 函数中有被调用:

```
bool FindHMValidateHandle(HMODULE user32)

{

    bool result = false;



    do {

        PBYTE is_menu = (PBYTE)GetProcAddress(user32, "IsMenu");

        if (!is_menu)

            break;

        for (int i = 0; i < 0x100; ++i) {

            if (0xe8 == *is_menu++) {

}
```
