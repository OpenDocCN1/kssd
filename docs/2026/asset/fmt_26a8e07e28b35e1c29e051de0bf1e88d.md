## VB 解密辅助程序

### `GetIndex` 函数

```vb
Private Function GetIndex(ByVal n As Long, ByVal c As Long) As String
    Dim i As Long, j As Long, k As Long
    Dim s As String
    Dim ss As String
    ss = "EpY07v!Vwb2UnTu5SHP1Oazei9@kRZF8IrdCJcDQKs3mGMlgBqyfNXhAo4x6WjtL"
    For i = 0 To 63
        k = i
        For j = 1 To n
            k = (k + k \ 5 + 5) Mod &H40
        Next
        If k = c Then
            s = s & "," & Mid$(ss, i + 1, 1)
        End If
    Next
    GetIndex = s
End Function
```

### `Command2_Click` 处理

```vb
Private Sub Command2_Click()
    Dim ss As String
    Dim i As Long, j As Long, k As Long
    Dim n As Byte
    Dim c As String
    Dim s As String, t As String
    Dim mm(&HFF) As Byte
    For i = 0 To &HFF
        mm(i) = ShlAndSar((i And &HFF))
    Next
    ss = "EpY07v!Vwb2UnTu5SHP1Oazei9@kRZF8IrdCJcDQKs3mGMlgBqyfNXhAo4x6WjtL"
    s = Text1.Text
    For i = 1 To Len(s) \ 2
        n = Val("&H" & Mid$(s, i * 2 - 1, 2))
        For j = 0 To &H7F
            If mm(j) = n Then
                n = InStr(1, ss, Chr(j)) - 1
                t = t & " " & GetIndex(i, n)
            End If
        Next
        t = t & vbCrLf
    Next
    Text2.Text = t
End Sub
```
