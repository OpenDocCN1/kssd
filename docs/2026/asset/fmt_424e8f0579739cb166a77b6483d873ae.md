function fun2(byval s as string) as long

dim r       as long

dim slen    as integer

dim i       as integer

dim j       as integer

dim cc      as byte

dim myindex as integer

myindex = 0

for i = 1 to len(s)

    cc = asc(mid(s,i,1))

    j = 0

    do while j <= 35

        if qarray(j) = cc then

            myindex = j

            exit do
