bool VMWare::CheckHypervisorPortEnum() const {

    bool is_vm = false;

    short ioports[] = { 'VX' , 'VY' };
    short ioport;

    for (short i = 0; i < _countof(ioports); ++i) {

        ioport = ioports[i];

        for (unsigned char cmd = 0; cmd < 0x2c; ++cmd) {

            __try {

                __asm {

                    push eax
                    push ebx
                    push ecx
                    push edx
                    mov eax, 'VMXh'
                    movzx ecx, cmd
                    mov dx, ioport
                    in eax, dx      // <- key point is here
                    pop edx
                    pop ecx
                    pop ebx
                    pop eax

                }

                is_vm = true;
                break;

            }

            __except (EXCEPTION_EXECUTE_HANDLER) {}
        }

    }
