## 基本原理

在自己驱动里面调用`CMRegistercallback`注册一个回调函数，然后在回调函数里面根据

`ebx`寄存器里面存储的是当前回调函数`_EX_CALLBACK`结构地址，而这一地址又一定在系统的回调链上的这一特性， 在`XP`系统上找到`CmpCallBackVector` 首地址，在`Vista`以上系统上找到`CallbackListHead` 首地址，然后依次遍历下了，找到`CallBackFunction`和`Cookie`，然后有了`CallBackFunction` 就能找到模块名称，根据`Cookie`就能进行注册表回调的摘取了~~~

先补充下`XP`下的`CmpCallBackVector`的节点怎么找打`Cookie`和`Funtion`的方法：

`CmpCallBackVector`--> `Node.Object`

--->`EX_CALLBACK_ROUTINE_BLOCK`

-->`CM_CALLBACK_CONTEXT_BLOCKXP`

在`EX_CALLBACK_ROUTINE_BLOCK` 找到`CallBackFuntion`

在`CM_CALLBACK_CONTEXT_BLOCK`中找到`Context`和`Cookie`

`ebx`在指向的是当前函数注册的在`nt!CmpCallBackVector`里面的节点位置

由于在`XP`系统上`CmpCallBackVector`里面存放的是`EX_FAST_REF`指针，因此从当前位置一直向

低内存位置查找`DWORD`值，直到为`0`，即为`CmpCallBackVector`的第一结构

从`EX_FASTREF.object`--> `EX_CALLBACK_ROUTINE_BLOCK`使用如下函数

```
PVOID FORCEINLINE MyFastRefGetObject (EX_FAST_REF FastRef)
{
    return (PVOID) (FastRef.Value & ~MAX_FAST_REFS);
}

#if defined (_WIN64)
    #define MAX_FAST_REFS 15
#else
    #define MAX_FAST_REFS 7
#endif
```

```
_EX_REG_CALL_CONTEXT *__cdecl GetCmRegCallBackFunctions()
{
  _EX_REG_CALL_CONTEXT *pContext; // esi@1
  _EX_REG_CALL_CONTEXT *result; // eax@1
  LARGE_INTEGER Cookie; // [sp+Ch] [bp-Ch]@2
  HANDLE KeyHandle; // [sp+14h] [bp-4h]@4

  pContext = (_EX_REG_CALL_CONTEXT *)ExAllocatePoolWithTag(0, 0x4D8u, 0x20747362u);
  result = 0;
  if ( pContext )
  {
    pContext->regCallBackCount = 0;
    pContext->mutex.Count = 1;                  // ExInitializeFastMutex
    pContext->mutex.Owner = 0;
    pContext->mutex.Contention = 0;
    KeInitializeEvent(&pContext->mutex.Event, SynchronizationEvent, 0);
    if ( CmRegisterCallback((PEX_CALLBACK_FUNCTION)regCallbackFunction, pContext, &Cookie) >= 0 )// register reg call back
    {
      ZwOpenKey(&KeyHandle, 0x20019u, &ObjectAttributes);// trigger it
      CmUnRegisterCallback(Cookie);
      result = pContext;
    }
    else
    {
      ExFreePoolWithTag(pContext, 0x20747362u);
      result = 0;
    }
  }
  return result;
}
```

```
int __userpurge regCallbackFunction<eax>(int pMayCmpCallBackVector<ebx>, _EX_REG_CALL_CONTEXT *pContext, int nOpType, int nOpInformations)
```

### Vista代码

```
int __stdcall Reg_CallBackWin7(_EX_REG_CALL_CONTEXT *pContext, _REG_CREATE_KEY_INFORMATION *a2, _CM_CALLBACK_CONTEXT_BLOCK *pCurCallBackRef)
{
  BOOLEAN (__stdcall *pMMisAddressValid)(PVOID); // ebx@1
  _CM_CALLBACK_CONTEXT_BLOCK *cmListHead; // edi@5
  _EX_REG_CALL_CONTEXT *pcontext; // esi@6
  DWORD ulIndex; // ecx@9
  LARGE_INTEGER *curCookie; // eax@10
  _CM_CALLBACK_CONTEXT_BLOCK *callBackListHead; // [sp+Ch] [bp-4h]@5
  struct _FAST_MUTEX *FastMutexa; // [sp+18h] [bp+8h]@6
  DWORD nContextIndex; // [sp+20h] [bp+10h]@9

  pMMisAddressValid = MmIsAddressValid;
  if ( MmIsAddressValid(pCurCallBackRef) )
  {
    if ( MmIsAddressValid(&pCurCallBackRef->Altitude) )
    {
      if ( MmIsAddressValid(pCurCallBackRef->Function) )
      {
        if ( (int (__userpurge *)<eax>(int<ebx>, _EX_REG_CALL_CONTEXT *, int, int))pCurCallBackRef->Function == regCallbackFunction )
        {
          cmListHead = GetCallVectorListHeadWin7((int)pCurCallBackRef);
          callBackListHead = cmListHead;
          if ( cmListHead )
          {
            pcontext = pContext;
            FastMutexa = &pContext->mutex;
            ExAcquireFastMutex(FastMutexa);
            while ( 1 )
            {
              cmListHead = (_CM_CALLBACK_CONTEXT_BLOCK *)cmListHead->CallbackListEntry.Flink;
              if ( cmListHead == callBackListHead )
                break;
              if ( !pMMisAddressValid(cmListHead) || !pMMisAddressValid(&cmListHead->Altitude) )
                break;
              ulIndex = pcontext->regCallBackCount;
              nContextIndex = 0;
              if ( pcontext->regCallBackCount )
              {
                curCookie = pcontext->callBackCookie;
                do
                {
                  if ( curCookie->low == cmListHead->Cookie.low && curCookie->high == cmListHead->Cookie.high )
                    break;
                  ++nContextIndex;
                  ++curCookie;
                }
                while ( nContextIndex < ulIndex );
                pMMisAddressValid = MmIsAddressValid;
              }
              if ( nContextIndex == ulIndex && ulIndex < 0x64 )
              {
                pcontext->callBackCookie[ulIndex].low = cmListHead->Cookie.low;
                pcontext->callBackCookie[ulIndex].high = cmListHead->Cookie.high;
                pcontext->pCallBackFunnction[pcontext->regCallBackCount++] = cmListHead->Function;
              }
            }
            ExReleaseFastMutex(FastMutexa);
          }
        }
      }
    }
  }
  return 0;
}
```
