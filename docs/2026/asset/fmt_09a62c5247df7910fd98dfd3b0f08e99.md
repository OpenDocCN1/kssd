def enc_xtea(input_low, input_high):
v0, v1 = ctypes.c_uint32(input_low), ctypes.c_uint32(input_high)

for i in range(64):

v1.value += (((v0.value << 8) ^ (v0.value >> 7)) - v0.value) ^ (total2.value + key[(total2.value >> 13) & 3])
total1.value -= delta
total2.value -= delta

def dec_xtea(input_low, input_high):
