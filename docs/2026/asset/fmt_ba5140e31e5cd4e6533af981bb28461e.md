 ori_func_NtUserCreateWindowEx=kmalloc(15+14);

 irql = WPOFF();
 memcpy(ori_func_NtUserCreateWindowEx, (PVOID)funcAddr, 15);
 WPON(irql);

 tmpv = funcAddr + 15;
 memcpy(jmp_code_orifunc+6,&tmpv,8);
 memcpy((PUCHAR)ori_func_NtUserCreateWindowEx+15,jmp_code_orifunc,14);
