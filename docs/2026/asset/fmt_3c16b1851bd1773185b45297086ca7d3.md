### 有趣的小实验：

**当我们发现创建远程线程与普通线程这一点区别后，是否可以这么做？使用**`CreateRemoteThread`**去创建普通线程，这其实很简单，只需要将进程句柄传为-1即可。说干就干，上代码：**

```c
#include <Windows.h>
#include <stdio.h>

DWORD WINAPI ThreadProc(LPVOID lpParameter)
{
    printf("ThreadProc\r\n");
    return 0;
}

int main()
{
    //进程句柄传-1就表示创建普通线程。
    HANDLE hThread = CreateRemoteThread((HANDLE)-1, NULL, 0, (LPTHREAD_START_ROUTINE)ThreadProc, NULL, 0, NULL);
    if (hThread == NULL)
    {
        printf("CreateRemoteThread Error:%d", GetLastError());
        return 0;
    }

    printf("CreateRemoteThread Success!\r\n");
    CloseHandle(hThread);
    return 0;
}
```
