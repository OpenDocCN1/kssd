def TestDec(a1):
    v2 = 0
    if len(a1) >= 2:
        while v2 < len(a1) // 2:
            a1[v2], a1[v2 + 16] = a1[v2 + 16], a1[v2]
            v2 += 1
    result = ord(a1[0]) if a1 else 0
    if a1:
        a1[0], a1[1] = a1[1], chr(result)
    result = len(a1)
    if result >= 3:
        v6 = 0
        while v6 + 4 <= result:
            a1[v6 + 2], a1[v6 + 3] = a1[v6 + 3], a1[v6 + 2]
            v6 += 2
    return a1

def reverse_TestDec(encrypted_str):
