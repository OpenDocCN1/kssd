//前24位取校验值

int calc1(char*str)

{
	//只取前24个
	DWORD len = 24;
	int v1 = 0;
	if (len>0)
	{
		int i= 0;
		do
		{
			v1 ^=str[i] << 8;
			int v4 = 8;
			do
			{
				if ((v1 & 0x8000) == 0)
					v1 *= 2;
				else
					v1 = 2 * v1 ^ 0x8201;
				--v4;
			} while (v4);
			v1 = v1&0xffff;
			++i;
			--len;
		} while (len);
	}
	return v1;
}
```

`Calc3`的算法逻辑 输入进来的分段`key`,去数组`arry`中寻找到，返回索引值。然后求 ![](https://bbs.kanxue.com/upload/attach/201903/546676_TE5N94XSD6ABFNA.jpg)返回的值再乘以索引值。然后累加上一次的结果。后返回。通过此函数可以知道，所有的`key`都是此数组中的值。

```
//某几位Key取校验值

int check_num(char* num)
{
	int len=strlen(num);
	int re = 0;
	for (int i=0; i<len;i++)
	{
		for (int k=0;k<0x22;k++)
		{
			if (num[i] == arry[k])//判断Key在特征字符串的位置
			{
				//计算幂 pow(base,指数)
				double n=pow( 34.0, (double)(len - i - 1) );
				int ret = 0;
				ret = trunc(n);//取整
				ret*=k;
				re += ret;
				break;
			}
		}
	}
	return re;
}
