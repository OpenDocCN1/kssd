        LOCAL   hSnapshot

        LOCAL   dwProcessID

        mov     dwProcessID, 0

        mov     stProcess.dwSize, sizeof stProcess

        invoke  CreateToolhelp32Snapshot, TH32CS_SNAPPROCESS, 0

        mov     hSnapshot, eax

        .while eax

                .if eax==0

                        mov     dwProcessID, eax

                        .break

                .endif

                invoke  Process32Next, hSnapshot, addr stProcess

        .endw

        invoke  CloseHandle, hSnapshot

        mov     eax, dwProcessID

        ret

_GetPidFromProcName endp

end start
