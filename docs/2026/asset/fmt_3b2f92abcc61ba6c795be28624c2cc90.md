            js_code = f.read()

        js_arg = [f"'{arg}'" if isinstance(arg, str) else str(arg) for arg in args]
        js_func_call = f"{func_name}({', '.join(js_arg)})"
        js_complete_code = f"{js_code};\nconsole.log({js_func_call});"

        res = subprocess.run(
            ['node', '-e', js_complete_code],
            capture_output=True,
            text=True,
            encoding='utf-8'

    def get_headers(self):
