        case RADIUS_CODE_ACCESS_REJECT:

                client->counters.access_rejects++;
                break;

                client->counters.access_challenges++;
                break;
        }

        //发送AccessChallenge消息
        res = sendto(data->auth_sock, reply->buf, reply->buf_used, 0,
