知道上面这些 那我们很容易就能修改内存 好 我们写个**跨进程修改内存**程序

根据“断点异常”原理进行修改 主要代码（其他请下载）：

```c
//修改字符串为：Tennn：Hello Pediy ！！！
BYTE bValue[25] = {0x8D, 0xBC, 0xB7, 0xB7, 0xB7, 0xE3,0xF9,
          0x91, 0xBC, 0xB5, 0xB5, 0xB6, 0xF9,
          0x89, 0xBC, 0xBD, 0xB0, 0xA0, 0xF8,
          0xF8, 0xF8, 0xF8, 0xF9, 0xF9, 0xF9,
};
DWORD dwWriteBytes = 0, dwOldProtect ;
DWORD dwTagAddr=0x0040C000;    //要修改处地址
// 修改目标地址的保护方式为可读可写
VirtualProtectEx ( g_hProcess, (LPVOID)dwTagAddr, 25, PAGE_READWRITE, &dwOldProtect ) ;
// 修改数据
WriteProcessMemory ( g_hProcess, (LPVOID)dwTagAddr, &bValue, 25 , &dwWriteBytes ) ;
// 还原目标地址的保护方式
VirtualProtectEx ( g_hProcess, (LPVOID)dwTagAddr, 25, dwOldProtect, &dwOldProtect ) ;
```
