1. `git clone https://chromium.googlesource.com/chromium/tools/depot_tools.git`
2. `echo "export PATH=/home/user1/Downloads/v8/depot_tools:$PATH" >> ~/.bashrc`
3. `fetch v8`
4. `cd v8`
5. `./build/install-build-deps.sh`
6. `git checkout 6dc88c191f5ecc5389dc26efa3ca0907faef3598`
7. `gclient sync`
8. `git apply ../starctf2019_oob/oob.diff #题目的patch`
9. `./tools/dev/v8gen.py x64.release`
10. `ninja -C ./out.gn/x64.release #release版本`

12. `ninja -C ./out.gn/x64.debug # Debug 版本`
