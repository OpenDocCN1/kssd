	输入参数：listDetectRecord 检测记录列表
	输出参数：detectResult 检测结果信息
	返回值	：检测到：true，未检测到：false
	备注	：N/A
	*******************************************************************************/
	virtual bool Execute(list<SDetectRecord> &listDetectRecord, SDetectResult &detectResult) = 0;

	/*******************************************************************************

	输入参数：N/A

	*******************************************************************************/
	virtual void Release() = 0;

		gGlobalClient->getTimerAxis()->SetTimer(m_dwTimerID[DETECT_TIMER_ID_INDEX],
