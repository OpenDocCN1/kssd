{

  PALETTE64 *that; // r8

  that = this;

  // StartRef和iret这里为0,就是pFirstColor

  writeBuff = (BYTE *)(*(_QWORD *)(*(_QWORD *)this->BaseObject + 0x80i64) + 4 * iStartRef);
