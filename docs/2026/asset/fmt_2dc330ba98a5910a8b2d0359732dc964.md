public static byte[] testBase64(String test, byte[] arr) {

    byte[] bytes = test.getBytes();

    int encodeLen = bytes.length;

    byte[] out = new byte[( encodeLen / 3 + (encodeLen % 3 == 0 ? 0 : 1) ) * 4];

    int y = 0;

    for (int i = 0; i < bytes.length; i+=3, y+=4) {

        byte a1 = bytes[i];

        byte a2 = 0;

        if (i + 1 >= bytes.length) {

        } else {

            a2 = bytes[i+1];

        }

        byte a3 = 0;

        if (i + 2 >= bytes.length) {

        } else {

            a3 = bytes[i + 2];

        }

        int f1 = a1 >> 2 & 0x3f;

        out[y] = arr[f1];

        if (out[y] == 0) {

            throw new RuntimeException("index "+f1+" null");

        }

        if (i + 1 >= bytes.length) {

            int f2 = (a1 << 4) & 0x30;

            out[y+1] = arr[f2];

            if (out[y+1] == 0) {

                throw new RuntimeException();

            }

            out[y+2] = '6';

            out[y+3] = '6';

            return out;

        }

        int f2 = (a1 << 4) & 0x30 | (a2 >> 4) & 0x0f;

        out[y+1] = arr[f2];

        if (out[y+1] == 0) {

            throw new RuntimeException();

        }

        if (i + 2 >= bytes.length) {

            int f3 = (a2 << 2) & 0x3c;

            out[y+2] = arr[f3];

            if (out[y+2] == 0) {

                throw new RuntimeException();

            }

            out[y+3] = '6';

            return out;

        }

        int f3 = (a2 << 2) & 0x3c | (a3 >> 6) & 0x03;

        out[y+2] = arr[f3];

        if (out[y+2] == 0) {

            throw new RuntimeException();

        }

        int f4 = (a3 & 0x3f);

        out[y+3] = arr[f4];

        if (out[y+3] == 0) {

            throw new RuntimeException();

        }

    }

    return out;

}
