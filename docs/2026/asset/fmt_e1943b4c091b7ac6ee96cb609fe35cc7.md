```c
        //初始值
        int iNameKey=0x13572468;
        for (i = 0;i < strlen(szName);i++)
        {
                j=((iNameKey + szName[i]) * 0x3721273) + 0x24681357;
                //有符号整数右移，SAR
                iNameKey=(j >> 7) | (j << 0x19);
        }
```
