以上代码用C语言表示如下：

```c
WORD list[0x80*2];

void sub_0047C9C0()
{
    int i,j=0;
    DWORD ieax,iecx,iesi=0;
    WORD temp;
    for(i=0;i<0x80*2;i++)
        list[i] = 0;
    do{
        ieax = 0x0C0C1; //ieax = 49345
        iecx = 1;
        do{
            if ((iesi&iecx)!=0)
            {
                temp = (WORD)(ieax&0x0FFFF);
                list[j] = list[j]^temp;
            }
            ieax*=2;
            iecx*=2;
            ieax = ieax^0x04003;
        }while(iecx<0x100);
        j++;
        iesi++;
    }while(j<0x80*2);
}
```
