在加壳过程中，要频繁用到内存申请、内存释放的操作，为了防止内存泄漏，我封装了一个类(`AllocMemory`)用来申请内存，这个类的作用就是只管申请内存，不用管释放，这个类会自动释放内存：

``` c++
#pragma once
#include <vector>
#include <basetsd.h>
using namespace std;
class AllocMemory
{
vector<char\*>p;
public:
virtual ~AllocMemory()
{
for (int i = 0; i < p.size(); i++)
{
if (p[i]==0)
{
continue;
}
free(p[i]);
p[i] = 0;
}
p.clear();
}
public:
template<typename T>
T auto\_malloc( ULONG\_PTR MAXSIZE)
{
T tmp = (T)malloc(MAXSIZE);
memset((char\*)tmp, 0, MAXSIZE);
p.push\_back((char\*)tmp);
return tmp;
}
};
```
