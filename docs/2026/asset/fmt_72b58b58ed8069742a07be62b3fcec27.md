```
source build/envsetup.sh
#打开开关，编译时生成CMakeLists.txt
export SOONG_GEN_CMAKEFILES=1
export SOONG_GEN_CMAKEFILES_DEBUG=1
breakfast oriole
```

2. 用Clion打开`CMakeLists.txt`

3. tools--->cmake-->Change Project Root
