function SiftStr(cStr: string): string;  //过滤字符串

  var i,j:integer;

begin

    Result:='';

    j:=Length(cStr);

    for i:=0 to j do

    begin

    if cStr[i] in ['0'..'9','a'..'z','A'..'Z','~','!',

    '@','#','$','%','^','&','\*','(',')','[',']','{','}','-','+'] then

    Result:=Result+cStr[i];

    end;

end;
