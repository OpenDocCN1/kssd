End If

For A = 1 To Len(Text1.Text)

B = B + Asc(Mid(Text1.Text, A, 1))

Next A

C = "wsy54321`"

K = Len(C)

For D = 1 To Len(C)

E = E + Asc(Mid(C, D, 1))

Next D

For F = 1 To Len(Text1.Text) - 6

If F = 6 Then

For I = F To Len(C)
