```python
def downImg(imgDataStr):
    try:
        ms = re.findall(r"(-?\d+)", imgDataStr)
        intArr = []

        for m in ms:
            ival = int(m)

            if ival < 0:
                ival += 256

            intArr.append(ival)

        bs = bytes(intArr)
        m = hashlib.md5()
        m.update(bs)
        downname = m.hexdigest()

        with open("download/" + downname + ".jpg", "ab") as f:
            f.write(bs)

        print("download sucess->" + downname + ".jpg")
    except Exception as e:
        print("downImg Error->")


imginx = 0

while True:

        imgArrValue = script.exports.getimginx()

        while imgArrValue > imginx:
```
