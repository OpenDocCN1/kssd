VOID GetKey1(PUCHAR pKey)
{
	TCHAR				szDrive[] = _T("A:\\");
	TCHAR				szPath[MAX_PATH] = {0};
	TCHAR				szLetter[26] = {0};
	DWORD				dwVolumeSerialNumber;
	ULARGE_INTEGER		uliTotalNumberOfBytes;
	UCHAR				uchData[1248];
	DWORD				cchData;
	TCHAR				ch;
	TCHAR				idx;
	LPTSTR				pLetter;
	INT					i;
	INT					j;

	GetModuleFileName(NULL, szPath, MAX_PATH);
	ch = szPath[0];
	ch &= 0xDF;
	idx = ch;
	idx -= 0x41;

	if(idx <= 25)
	{
		szLetter[idx] = 0x01;
	}

	GetSystemDirectory(szPath, MAX_PATH);
	ch = szPath[0];
	ch &= 0xDF;
	idx = ch;
	idx -= 0x41;

	if(idx <= 25)
	{
		szLetter[idx] = 0x01;
	}

	pLetter = szLetter;
	for(i=0, j=0; i<26; i++, pLetter++)
	{
		if(*pLetter)
		{
			szDrive[0] = i + 0x41;
			dwVolumeSerialNumber = 0;

			GetVolumeInformation(szDrive,
				NULL,
				0,
				&dwVolumeSerialNumber,
				NULL,
				NULL,
				NULL,
				0);

			uliTotalNumberOfBytes.LowPart = 0;
			uliTotalNumberOfBytes.HighPart = 0;
			GetDiskFreeSpaceEx(szDrive,
				NULL,
				&uliTotalNumberOfBytes,
				NULL);

			*(LPDWORD)&uchData[j * sizeof(DWORD)] = dwVolumeSerialNumber;
			j++;

			*(LPDWORD)&uchData[j * sizeof(DWORD)] = uliTotalNumberOfBytes.LowPart;
			j++;

			*(LPDWORD)&uchData[j * sizeof(DWORD)] = uliTotalNumberOfBytes.HighPart;
			j++;
		}
	}

	cchData = j * sizeof(DWORD);
	GetMD5(uchData, cchData, pKey);
}

VOID GetKey2(PUCHAR pKey)
{
	HKEY				hKey;
	TCHAR				szStr[] = _T("FGDEBC@A-JKHI-NOLM-PQRS-TUVWXYZ[\\]^_");
	LPTSTR				lpStr;
	TCHAR				szMachineGuid[128] = {0};
	DWORD				cbMachineGuid;
	LPTSTR				lpMachineGuid;
	UCHAR				szBuff[96] = {0};
	UCHAR				ch1;
	UCHAR				ch2;
	DWORD				dwType;
	INT					i;

	if(RegOpenKeyEx(HKEY_LOCAL_MACHINE,
		_T("SOFTWARE\\Microsoft\\Cryptography"),
		NULL,
		KEY_READ,
		&hKey) == ERROR_SUCCESS)
	{
		dwType = REG_SZ;
		cbMachineGuid = sizeof(szMachineGuid);
		RegQueryValueEx(hKey,
			_T("MachineGuid"),
			NULL,
			&dwType,
			(LPBYTE)szMachineGuid,
			&cbMachineGuid);
		RegCloseKey(hKey);

		lpStr = szStr;
		lpMachineGuid = szMachineGuid;
		ch1 = 0x46;
		do
		{
			if(ch1 >= 0x40)
			{
				ch2 = *lpMachineGuid;
				ch2 |= 0x20;

				if(ch2 >= 0x61)
				{
					ch2 -= 0x57;
				}
				else
				{
					ch2 -= 0x30;
				}

				szBuff[ch1] = ch2;
			}

			lpStr++;
			lpMachineGuid++;
		}
		while(ch1 = *lpStr);

		for(i=0; i<16; i++)
		{
			pKey[i] = szBuff[0x40 + i * 2];
			pKey[i] <<= 4;
			pKey[i] |= szBuff[0x40 + i * 2 + 1];
		}
	}
}

VOID GenerateKey(PUCHAR puchKey)
{
	UCHAR			uchKey1[16] = {0};
	UCHAR			uchKey2[16] = {0};
	INT				i;

	GetKey1(uchKey1);
	GetKey2(uchKey2);

	for(i=0; i<16; i++)
	{
		puchKey[i] = uchKey1[i] ^ uchKey2[i];
	}
}
