int Invoke(char* pDllName, char* pFunName, ...)
{
    DWORD dwLoadLib = 0x7C801D77;

int Align(int nSize, int n)

    if(nSize % n)
    {
        int num = nSize / n;
        return (num + 1) * n;
    }

}
