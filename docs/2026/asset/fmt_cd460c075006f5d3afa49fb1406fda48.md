LPCWSTR GetProcessPath(DWORD processId) {

static wchar_t szProcessPath[MAX_PATH];
if (hProcess != nullptr) {
DWORD size = sizeof(szProcessPath) / sizeof(szProcessPath[0]);

CloseHandle(hProcess);

}

}
