IDC\_STC2                equ 1005

                db '         -=Email:qfejj@163.com=-',0

hDlg            HINSTANCE ?
sAppendString   dd ?

aAcumulate      dd 20 dup(?)    ;它前后需要很大的空间

.code
start:
        invoke GetModuleHandle,NULL
        mov hInstance,eax
        invoke DialogBoxParam,hInstance,IDD\_DLGBOX,NULL,addr DlgProc,NULL
        invoke ExitProcess,0
;>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>
DlgProc proc hWin:HWND,uMsg:UINT,wParam:WPARAM,lParam:LPARAM
        mov eax,uMsg
        .if     eax==WM\_CLOSE
                  invoke EndDialog,hWin,0
        .elseif eax==WM\_INITDIALOG
                  push hWin
                  pop  hDlg
        .elseif eax==WM\_COMMAND
                  mov eax,wParam
                  .if eax==IDC\_GENERATE
                        invoke RtlZeroMemory,offset aNameBuffer,sizeof aNameBuffer
                        invoke RtlZeroMemory,offset aStdKeyBuffer,sizeof aStdKeyBuffer
                        invoke RtlZeroMemory,offset aStdKeyOut,sizeof aStdKeyOut
                         invoke GetKey
                  .elseif eax==IDC\_ABOUT
                        invoke MessageBox,hWin,addr szMsgboxText,addr szMsgboxCaption,MB\_OK
                  .endif
        .else
                mov eax,FALSE
                ret
        .endif
        mov eax,TRUE
        ret

        invoke GetDlgItemText,hDlg,IDC\_NAME,offset aNameBuffer,sizeof aNameBuffer

        invoke lstrcat,offset aNameBuffer,offset szAppendString1
        invoke lstrlen,addr aNameBuffer

        invoke lstrcat,offset [aNameBuffer+38h],offset sAppendString  ;自己调试的时候留意一下，看这里为什么要用38h
