void DecodeMathineCode(char *szMachineCode)
{
    int i, nFlag, nLength;

    nLength = strlen(szMachineCode);

    for (i = 0; i < nLength; i++)
    {
        nFlag = i % 2 == 0 ? 1 : 2;
        szMachineCode[i] -= nFlag;
    }

    if (nLength < 11)
    {
        for (i = nLength; i < 11; i++)
        {
            szMachineCode[i] = '0';
        }
    }

    szMachineCode[11] = 0;
}
