unsigned long reg(char name[])
{

    int len;

    unsigned long ecx,edx,esi,c;

    len=strlen(name);

    ecx=0;

    edx=0;

    esi=0;

    for(int b=0;b<len;b++)

    {

        c=name**;

        if(c>=97 && c<=122)

          {c=c-32;             //把注册名中小写的转化为大写

           name**=c;}

    }

    for(int a=0;a<len;a++)     //对转化后的注册名进行运算

    {

        ecx=ecx << 4;

        edx=name[a];

        ecx=ecx + edx;

        edx=ecx;

        edx=edx & 0xf0000000;

        if(edx==0)

           {edx=~edx;

            ecx=ecx & edx;}

        else

            {esi=edx;

            esi=esi >> 0x18;

            ecx=ecx ^ esi;

            edx=~edx;

            ecx=ecx & edx;}

    }

   return ecx;                  //返回运算值

}
