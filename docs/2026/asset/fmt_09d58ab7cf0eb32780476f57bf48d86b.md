### `guess_seed.c`

```c
#include <unistd.h>
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

static void* seed = 0;

int main(int argc, char **argv)
{
    int low3 = atoi(argv[1]);
    int r = atoi(argv[2]);
    unsigned int seed;
    unsigned int i;

    for (i = 0; i <= 0xFFFFF; i++)
    {
        seed = i << 12;
        seed += low3;
        srand(seed);
        if (rand() == r)
        {
            printf("0x%x\n", rand());
            return 0;
        }
    }
    printf("end\n");
    return 0;
}
```
