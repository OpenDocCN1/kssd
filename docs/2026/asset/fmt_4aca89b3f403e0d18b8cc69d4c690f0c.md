impl LoadVerifier {
fn verify\_elf(&self, elf: &Elf) -> Result<()> {
// 1. 验证ELF结构完整性
validate\_elf\_header(elf)?;
// 2. 校验代码段哈希
let text\_hash = sha256(elf.get\_section\_data(".text"));
if !self.hash\_policy.verify(&text\_hash) {
bail!("Text section hash mismatch");
}
// 3. 验证数字签名
let sig = elf.get\_section\_data(".signature");
self.signature\_check.verify(elf, sig)?;
// 4. 检查依赖库白名单
for dep in parse\_dependencies(elf)? {
if !self.hash\_policy.allowed\_dependency(dep) {
bail!("Forbidden dependency: {}", dep);
}
}
Ok(())
}
}

use ring::signature;
let pub\_key = signature::UnparsedPublicKey::new(
&signature::RSA\_PSS\_SHA256,
&self.public\_key
);
pub\_key.verify(elf.raw\_data(), signature)
.map\_err(|\_| anyhow!("Invalid signature"))
}
