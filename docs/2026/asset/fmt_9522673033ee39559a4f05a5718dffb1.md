```
void x86\_configure\_nx(void)
{
if (boot\_cpu\_has(X86\_FEATURE\_NX) && !disable\_nx)
\_\_supported\_pte\_mask |= \_PAGE\_NX;

\_\_supported\_pte\_mask &= ~\_PAGE\_NX;
```
