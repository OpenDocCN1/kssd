在驱动程序里面可以使用 `READ_PORT_UCHAR`和`WRITE_PORT_UCHAR`函数来完成端口的读写，具体的说明上MSDN。

需要注意的是`1F0H` 数据端口，他是一个字型端口，非字节，所以呢，就不可以使用 `READ_PORT_UCHAR`和`WRITE_PORT_UCHAR`这两个函数了，因为他们是读取的字节端口。

怎么办呢？用内嵌汇编，`in ax,ax (dx=0x1f0)`

为了方便我使用的是LBA28寻址方式。

通过`0x1f3`到`0x1f6`传入。上面有。

这里一共可以存放32位数据，LBA28只需要28位就够了，剩余的4位用来干什么？当然不可能没有用，`0x1f7`的高低4位用来存放LBA28的24~27位，高4位存放一些属性。

位数 意义

4 0=主盘，1=从盘

5 永恒为1

6 寻址方式 0：CHS 1=LBA 我们使用LBA

7 永恒为1。

我一般直接填`0xe` 二进制：1110 对照上面的表看看，LBA模式，主盘。

因此，一个LBA28的地址可以描述为：

```
typedef struct _LBA28
{
	UCHAR A0_7;
	UCHAR A8_15;
	UCHAR A16_23;
	UCHAR A24_27; 	//低4位有效
	UCHAR Attribute; //低4位有效
}LBA28,*PLBA28;
```

硬盘是成块设备一个块（扇区）512字节，`0x1f0`一次传输字，所以要连续循环256次。

一个扇区可以定义为：

```
typedef struct _sector
{
	UCHAR sector[512];
}SECTOR,*PSECTOR;
```

根据资料，我们就可以编写硬盘读写函数了。

```
int readDisk(LBA28 lba,UCHAR num,PSECTOR pSector)
{
	int y,x=0;
	LONG js=0,MAX=90000;
	PUCHAR addr;
	UCHAR tmp_data,tmp_data1; 
	addr=(PUCHAR)0x1f2;
	WRITE_PORT_UCHAR(addr,num);
	addr=(PUCHAR)0x1f3;
	WRITE_PORT_UCHAR(addr,lba.A0_7);
	addr=(PUCHAR)0x1f4;
	WRITE_PORT_UCHAR(addr,lba.A8_15);
	addr=(PUCHAR)0x1f5;
	WRITE_PORT_UCHAR(addr,lba.A16_23);
	addr=(PUCHAR)0x1f6;
	tmp_data=lba.A24_27;
	tmp_data&=0xf;
	tmp_data1=lba.Attribute;
	tmp_data=((tmp_data1<<4)&0xf0)|tmp_data;
	WRITE_PORT_UCHAR(addr,tmp_data);
	
	addr=(PUCHAR)0x1f7;   
	WRITE_PORT_UCHAR(addr,0x20);
	
	//进入准备状态
	do
	{
	CHAR error;
	js++;
	if(js>MAX)
	{
		return 0;  //硬盘无响应
	}
	DbgPrint("等待中！");
	tmp_data=READ_PORT_UCHAR(addr);
	error=tmp_data&1; //是否有错误
	DbgPrint("STATUS ： %d",tmp_data);
	if(error==1)
	{
		addr=(PUCHAR)0x1f1;
		tmp_data=READ_PORT_UCHAR(addr);
		DbgPrint("error Code：%d",tmp_data);
		return 0;
	}
	
	
	tmp_data&=0x88; // 提取出 7 3位
	}while(tmp_data!=0x8);
	//进入读取状态
	addr=(PUCHAR)0x1f0;
	for(y=0;y<num;y++)
	{
		int i;
		PUCHAR tt;
		DbgPrint("Read ing %d",y);
		tt=&pSector[y].sector[0]; //sector是UCHAR
		for(i=0;i<512;i+=2)
		{
			short da=0;
			_asm
			{
				push dx
				push eax
				mov dx,0x1f0
				in ax,dx
				mov da,ax
				pop eax
				pop dx
			}
			RtlCopyMemory(&pSector[y].sector[i],&da,2);
		}
	}
	return 250;
}
```

```
	}while(tmp_data!=0x8);

	addr=(PUCHAR)0x1f0;
	for(y=0;y<num;y++)
	{
		int i;
		PUCHAR tt;
		DbgPrint("Write ing %d",y);
		//tt=&pSector[y].sector[0]; //sector是UCHAR
		for(i=0;i<512;i+=2) //循环256次
		{
			short da=0;
			RtlCopyMemory(&da,&pSector[y].sector[i],2);
			_asm
			{
				    push dx
					push eax
					mov dx,0x1f0
					mov ax,da
					out dx,ax
					pop eax
					pop dx
			}
		}
	}

}
```

看完代码，发现两个函数都有`num`参数，这个是读入的扇区数量，因为IDE控制器支持同时读入多个扇区，最大255。
