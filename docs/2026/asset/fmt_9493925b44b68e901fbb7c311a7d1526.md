__asm

 // int 3

 // 返回地址从tss中的ptl中拿出

push eax
pushfd
pop eax

or eax,0x4000
push eax
popfd

mov dwESP,esp
mov ax,ss
mov dwSS,eax
mov ax,cs
mov dwCS,eax

pop eax

iretd
}
}

// 过去当前eip，通过函数调用堆栈中的返回地址获取
void __declspec(naked) GetEIP()
{
__asm
{
mov eax,dword ptr ss:[esp]
ret
}
}
