void sub_3b9d4_reverse(unsigned char* byte) {
    for (int i = 3; i >= 0; i--) {
        byte[i] += 8 * i;
    }

    byte[0] += 0x1c;
    byte[1] ^= 0xd3;
    byte[2] += 0x5e;
    byte[3] ^= 0x86;

    for (int i = 3; i >= 0; i--) {
        byte[i] ^= i;
    }

    byte += 4;

    for (int i = 3; i >= 0; i--) {
        byte[i] += 8 * i;
    }

    byte[0] += 0x1c;
    byte[1] ^= 0xd3;
    byte[2] += 0x5e;
    byte[3] ^= 0x86;

    for (int i = 3; i >= 0; i--) {
        byte[i] ^= i;
    }
}
