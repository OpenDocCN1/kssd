  puts("!!!The testing string!!!");

int main() {
  puts("This is a testing string!");
  char ch;
  if ((ch = getchar()) == '6') {
    printf("6666%c\n", ch);
  } else {
    printf("WTF?!\n");
  }
  func("!!!The testing string!!!");
  return 0;
}
