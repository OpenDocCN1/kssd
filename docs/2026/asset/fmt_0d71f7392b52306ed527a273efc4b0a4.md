```
__asm
{
pushfd
pushad
mov eax,g_uGetGameBaseAddr
add eax,0xbc        //debugport地址
cmp edx,eax
je Detour
popad
popfd
pop dword ptr [edx]
jmp g_uTPDetourPushAddr
Detour:
popad
popfd
add esp,4
jmp g_uTPDetourPushAddr
}
```
