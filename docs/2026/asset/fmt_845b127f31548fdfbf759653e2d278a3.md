int __stdcall sub_1071A(int a1, PIRP Irp)

{

  switch ( *(_DWORD *)(Irp->Tail.Overlay.PacketType + 12) )

  {

    case 0x222004:                                                  ;前几个小节提到DeviceIoControl API的几个调用控制码都是0x222004

      dword_114D8 = 1;                                              ;这是一个流程控制标志，俗称暗桩

      break;

{
