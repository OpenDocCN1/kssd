    add     esi, edx

    inc     ebx

    dec     eax

   @004B2195:

    imul    eax, esi, $07F39F                { 把 Sum*0x7F39F的值放入EAX}

    sar     eax, 1                           { 对EAX的值算数右移1位，值依旧放入EAX}

    jns     @004B21A5                        { 非负跳转}

    adc     eax, 0

    jmp     @004B21A5

   @004B21A5:

    pop     ebx
