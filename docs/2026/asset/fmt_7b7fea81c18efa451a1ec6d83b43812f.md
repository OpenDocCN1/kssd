                if(nMemoryOnly == 1)        g\_nException = 1;

        }
        else if(g\_nAddrsize == 2)        //16位寻址模式

                        {

                        {

                                if(size < 4)        g\_nException = 1;

                                dispsize = 2;

                        }

                }

        }
        else if(nCommand == 0x05)        //32位寻址模式(无寄存器基址)

        {

                dispsize = 4;

        }

                nSIB = src[2];

                {

                        dispsize = 4;

                        if(size < 7)        g\_nException = 1;

                }
                else        //非立即数寻址

                {

                        if((nCommand & 0xC0) == 0x40)        //mod=01

                        {

                                dispsize = 1;

                                if(size < 4)        g\_nException = 1;

                        }
                        else if((nCommand & 0xC0) == 0x80)        //mod=10

                        {

                                dispsize = 4;

                                if(size < 7)        g\_nException = 1;

                        }

                }

        }
        else        //32位寻址模式(无SIB字段)

        {

                if((nCommand & 0xC0) == 0x40)        //mod=01

                {

                        dispsize = 1;

                        if(size < 3)        g\_nException = 1;

                }
                else if((nCommand & 0xC0) == 0x80)        //mod=10

                {

                        dispsize = 4;

                        if(size < 6)        g\_nException = 1;

                }

        }
        break;
