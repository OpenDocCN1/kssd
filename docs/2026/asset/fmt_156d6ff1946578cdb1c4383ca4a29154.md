.data

szTemp    db 20 dup(0)

szBuffer  db 50 dup(0)

.code

  mov     ebx, 1h

L002:

  lea     eax, hInput1

  invoke  lstrlen, eax

  push    eax

  mov     eax, ebx

  dec     eax

  pop     edx

  mov     ecx, edx

  cdq

  idiv    ecx

  lea     eax, hInput1

  mov     al, byte ptr [eax+edx]

  lea     edx, szChar

  mov     dl, byte ptr [edx+ebx-1]

  xor     al, dl

  and     eax, 0FFh

  invoke wsprintf,addr szTemp,addr szFMT,eax

  lea     eax, szTemp

  invoke  lstrlen, eax

  lea     edx, szTemp

  mov     al, byte ptr [edx+eax-1]

  mov     byte ptr [szBuffer+ebx-1], al

  inc     ebx

  cmp     ebx, 0Bh

  jnz L002

  lea     eax, szBuffer
