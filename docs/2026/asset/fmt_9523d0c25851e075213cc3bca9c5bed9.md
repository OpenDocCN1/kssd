```c
struct strarray{

    int len;

    char * sp;

};

strarray sa;

char Table[] = {"CA5BDWF4H9PJK3SUM2XEL7GRZNQTY6V8"};

void fun1(CString s,strarray * p)

{

    int          i,j,k,index;

    unsigned int x;

    if(p -> len != 0)

        delete [](p -> sp);

    p -> sp = 0;

    p -> len = s.GetLength() * 5;

    p -> sp = new char[p -> len];

    memset(p -> sp,0,p -> len);

    if(s.GetLength() <= 0)return;

    j = 0;

    k = 0;

    do{

        index = -1;

        for(i = 0; i < 32; i++)

        {

            if(s[j] = Table[i])

            {

                index = i;

                break;

            }

        }

        for(i = 0; i < 5; i++)

        {

            x = 1;

            x = x << i;

            if(x & index)

                (p -> sa)[k] = 1;

            k ++;

        }

        j ++;

    }while(j < s.GetLength());

}
```
