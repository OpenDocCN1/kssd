- 通常不显示`STDERR`
- 无法正确使用文本编辑器 `vim`
- 没有标签完成
- 没有向上的箭头历史
- 没有工作控制
- 等等...

function check_ip($prefix,$filename) {
  //echo "prefix: $prefix - fname: $filename<br>\n";
  $ret = true;
  if (!(filter_var($prefix, FILTER_VALIDATE_IP))) {
    $ret = false;
    $msg = "4tt4ck on file ".$filename.": prefix is not a valid ip ";
  } else {
    $msg = $filename;
  }
  return array($ret,$msg);
}
