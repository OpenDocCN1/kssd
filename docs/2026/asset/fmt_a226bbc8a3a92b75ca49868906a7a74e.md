{

        {
                if(monstAdd)//monstAdd=0;//为1普通怪房加攻击。为0不加
                {
                        if (g.IsMonsterDead())//怪物死亡
                        {
                                Sleep(1000);
                                i++;
                                if (i>=roomst)
                                {
                                        i=0;
                                        g.EditAttack((\*(int\*)GAME\_ADDR\_ROLE\_BASE)+4,0x298,attack);
                                        OutputDebugString("普通房加攻击");
                                }

                        else
                        {
                                OutputDebugString("普通房加攻击结束");
                                monstAdd=0;
                        }
                }
                Sleep(200);
        }
        return 1;
}
