void CFeitianDlg::OnButton1()
{
    // TODO: Add your control notification handler code here
    UpdateData();
    unsigned int tmp;
    unsigned int sum=0;
    if(m_name.IsEmpty())
    {
        return;
    }
    m_code.Empty();
    //计算K1=SUM
    for(int i=0;i<m_name.GetLength();i++)
    {
        tmp=0xff&m_name[i];
        sum+=tmp;
        sum^=0xbfa05;
    }
    _asm
    {
        mov eax,0x12BCC244
        mov edx,eax
        shl eax,4
        add eax,edx
        add sum,eax
        add sum,0x13A3D4
        xor sum,0x59ED8D
    }
    sum^= 0xC47324;//进行异或
    //求逆就是K1转换成10进制然后异或0xC47324就可以了
    //格式化为10进制//输出到编辑框

    UpdateData(FALSE);
}
