void FindLicense(BYTE* sha1data, int len, char* outLicense)
{
    unsigned long offset = 0;
    unsigned long temp = 0x0;
    char tempLong[3] = {0};
    char szTable[256] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

        return;

    {

        //取3位sha1值为一个整数
        for (int j=0; j<3; j++)
        {
            if (0 == j)
            {
                temp = sha1data[i+j];
                temp <<= 8;
                offset = temp;
            }
            if (1 == j)
            {
                temp = sha1data[i+j];
                offset += temp;
            }
            if (2 == j)
            {
                temp = sha1data[i+j];
                offset <<=8;
                offset += temp;
            }
        }

        temp = offset;
        temp &= 0xFC0000;
        temp >>= 0x12;
        outLicense[index] = szTable[temp];
        index++;
        temp = offset;
        temp &= 0x3F000;
