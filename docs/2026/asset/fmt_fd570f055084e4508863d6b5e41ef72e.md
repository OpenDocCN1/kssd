int _tmain(int argc, _TCHAR* argv[])
{

  int j=0;

  string s1,result;

  while(s1.length() != 6)

    cout<<"长度必须为 6"<<endl;

    cin>>s1;

  };

    char c;

    for(int i=0;i<6;i++)

    {

      c = s1[i];

      switch(c)

      {

      case '1':

        result += "4";

        break;

      case '4':

        result += "7";

        break;

      case '7':

        result += "1";
