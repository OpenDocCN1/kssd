int CWinNcKeyGenDlg::GetCalcHex(DWORD dwKey)
{
    int nTmp = dwKey;
    int nData = 0;

    while (nTmp != 0)
    {
        nData += nTmp % 0xA;
        nTmp = nTmp / 0xA;
    }

    if (nData > 9)
    {
        nData = GetCalcHex(nData);
    }
    return nData;
}

void CWinNcKeyGenDlg::OnBnClickedBtnKeygen()
{
    WCHAR szRegName[0x100] = { 0 };
    int nRegNameLen = 0;
    int nKey = 0x203;

    //获取注册名
    ::GetDlgItemText(m_hWnd, IDC_EDIT_REGNAME, szRegName, 0xc);
    if (szRegName[0] == 0)
    {
        return;
    }

    //算法
    nRegNameLen = wcslen(szRegName);
    for (int i = 0; i < nRegNameLen; i++)
    {
        nKey += GetCalcHex(szRegName[i]);
    }

    //长度不足0xC 处理用户名
    int nCount = nRegNameLen;
    while (nCount < 0xc)
    {
        WCHAR szTmpBuf[0x100] = { 0 };
        wcscpy_s(szTmpBuf, szRegName);
        wcscat_s(szRegName, szTmpBuf);
        nCount = wcslen(szRegName);
    }

    char szShowRegCode[15] = { 0 };
    char szRegCode[12] = { 0 };
    for (int i = 0; i < 12; i++)
    {
        szRegCode[i] = GetCalcHex(szRegName[i] + g_szKeyTable[i] + nKey);
    }

    wsprintfA(szShowRegCode, "%d%d%d%d-%d%d%d%d-%d%d%d%d",
        szRegCode[0], szRegCode[1], szRegCode[2], szRegCode[3],
        szRegCode[4], szRegCode[5], szRegCode[6], szRegCode[7],
        szRegCode[8], szRegCode[9], szRegCode[0xa], szRegCode[0xb]);

    ::SetDlgItemTextA(m_hWnd, IDC_EDIT_REGCODE, szShowRegCode);
}
