void __stdcall ExecuteCmd(char* sHexCmd)
{
	static int cn_send_oft = 2;
	if (NULL == sHexCmd){
		return;
	}
	try{
		DWORD dwKey = GetSendKey();
		if (0 == dwKey){
			return;
		}
		DWORD dwLen = *(WORD*)(sHexCmd);
		memcpy(&g_ExecuteCmdBuff[cn_send_oft],&sHexCmd[2],dwLen);
		DWORD dwBuff = DWORD(&g_ExecuteCmdBuff[cn_send_oft]);
		//执行加密组包
		__asm{
			pushad
			lea edx,dwKey
			push edx
			push dwBuff
			push dwLen
			mov edx,g_Adr_MMSend
			call edx
			ADD ESP,0x0C
			popad
		}
		dwLen = dwLen + cn_send_oft;

			send(g_InitIO.Api.GetSendSock(),g_ExecuteCmdBuff,dwLen,0);
		}
		else{
			send(g_ExecuteCmd_Sock,g_ExecuteCmdBuff,dwLen,0);

	}
