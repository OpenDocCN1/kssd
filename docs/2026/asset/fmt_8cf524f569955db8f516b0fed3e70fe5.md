    PIMAGE_FILE_HEADER pFileHeader = &pNtHeader->FileHeader;

    pFileHeader->NumberOfSections = 0;
    pFileHeader->TimeDateStamp = 0;
    PIMAGE_OPTIONAL_HEADER32 pOptionalHeader = &pNtHeader->OptionalHeader;
