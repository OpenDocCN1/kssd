  }
  if(arg1 == 0x1F){
  printf("Detect DebugFlags");
  }
  }

  if(arg1 == 0){
  printf("Query Process Basic Information \n");

  if(arg1 == 3) {
  printf("Query Object Types Information\n");

  }
