```c
void WORDSHL(WORD source[],WORD dest[],int length)

{

  DWORD result=0; //因为DWORD的存在所以比较成为可能

  int carry=0;    //进位标志

  int d=0;

  int j;

  for(j=0;j<length;j++)

  {

    result=(source[j]<<1);

    if(result>0xFFFF)

      carry=1;

    else

      carry=0;

    dest[j]=(source[j]<<1);

    if(d==1)

    {

      dest[j]=(dest[j]|1); 加上进位

    }

    d=carry;

  }

}
```
