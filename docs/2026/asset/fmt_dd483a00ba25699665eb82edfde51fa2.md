// Hide modules from GUI
if (optionHideInGUI)
{
    foreach (var vbaModule in vbaModules)
    {
        if ((vbaModule.moduleName != "ThisDocument") && (vbaModule.moduleName != "ThisWorkbook"))
        {
            Console.WriteLine("Hiding module: " + vbaModule.moduleName);
