    list_factors=[]
    for i in range(2,key+1):
        if key%i==0 and isprime(i)==1:
            list_factors.append(i)
    return list_factors

def isprime(to_judge):
    if to_judge==2:
        return 1
    else:
        for j in range(2,to_judge):
            if to_judge%j==0:
                return 0
        if j==to_judge-1:
            return 1

def maxmium(value):
    if value==1:
        return 1
    else:
        list_prime=getfactors(value)
