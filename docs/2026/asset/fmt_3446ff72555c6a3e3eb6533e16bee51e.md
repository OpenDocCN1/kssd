int CWinNcKeyGenDlg::GetCalcHex(DWORD dwKey)
{
    int nTmp = dwKey;
    int nData = 0;

    while (nTmp != 0)
    {
        nData += nTmp % 0xA;
        nTmp = nTmp / 0xA;
    }

    if (nData > 9)
    {
        nData = GetCalcHex(nData);
    }
    return nData;
}
