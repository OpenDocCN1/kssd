def removeVirtualBlock(callStack):
for item in range(len(callStack) - 1, -1, -1):
for addr in virtualBlock:
if callStack[item] == addr:
del callStack[item]

return callStack

### 0x5、如何避免死循环

避免死循环，主要就是针块的下一个跳转存在多解的情况进行处理，比如，对`csel`的特殊处理如下，其它的多解类似，上面的代码中有包含：

```python
# 返回值为赋值的寄存器id，可能填充值的寄存器id，例如：csel w8, w8, w25, eq --> [w8], [w8, w25]
def asmUniCsel(constRegs, va, bts, optypes, operands):
nzcvObj = readReg(constRegs, ARM64_REG_NZCV)
dstRegId = operands[0].reg
srcRegId1 = operands[1].reg
srcRegId2 = operands[2].reg
if "111" == optypes:
if isRegConst(nzcvObj):
nzcvVal = getRegObjVal(nzcvObj)
uniSetReg(ARM64_REG_NZCV, nzcvVal)
uniSetReg(srcRegId1, 1)
uniSetReg(srcRegId2, 2)
uniTestAsm(bts, va)
result = uniGetRegVal(dstRegId)
if result == 1:
writeReg(constRegs, dstRegId, readReg(constRegs, srcRegId1))
return dstRegId, [srcRegId1]
else:
writeReg(constRegs, dstRegId, readReg(constRegs, srcRegId2))
return dstRegId, [srcRegId2]
return dstRegId, [srcRegId1, srcRegId2]
```

比如

```
cmp x8, #0x0
mov x8, #0xXXX
csel x8, x8, x9, ne
```

假设这个结果不唯一，那么，在走到这里时会返回两个分支[`x8`, `x9`]，之后在递归时，会先走`x8`分支再走`x9`分支，所以出结果必然是`x8`在前`x9`在后
