6. 调试时几处关键的位置：

`sub_1A9EE`：执行到这里时，R9 寄存器中出现了输入的字符串。

`sub_19DC8`：继续传递输入的字符串，即将要做处理。

```asm
.text:00019DC8                 MOV             R8, R0   199310124853!

.text:00019DCA                 MOV             R0, R9   inputString

.text:00019DCC                 PUSH.W          {R4-R10,LR}

.text:00019DD0                 POP.W           {R4-R10,LR}

.text:00019DD4                 B               sub_19DFE
```

7. 开始处理输入字符串的第 1 位：

```c
int __fastcall sub_47A2(int a1, unsigned int a2, _BYTE *a3, _BYTE *a4)   //a1+8对应这一个字节数组，比较关键，我直接给dump出来，再python中引用

{

  int v4; // r10@0

  char v5; // r12@0

  int v6; // r5@3

  int v7; // r4@3

  int v8; // r10@3

  int v9; // r8@3

  int v11; // [sp+14h] [bp-8h]@0



  if ( v14 != a2 >> 3 )

  {

    v6 = (unsigned __int8)(v5 + 1);

    v7 = *(_DWORD *)(a1 + 8 + 4 * v6);

    v8 = v4 + v7;

    v9 = *(_DWORD *)(a1 + 8 + 4 * v8);

    *(_DWORD *)(a1 + 8 + 4 * v6) = v9;

    *(_DWORD *)(a1 + 8 + 4 * v8) = v7;

    *a4 = *(_DWORD *)(a1 + 8 + 4 * (unsigned __int8)(v7 + v9)) ^ *a3;

    JUMPOUT(&unk_5F9EC844);

  }

  v10 = a4;

  if ( a2 & 7 )

  {

    v11 = (unsigned __int8)(v5 + 1);    =1

    v12 = *(_DWORD *)(a1 + 8 + 4 * v11);    =0x6B

    v13 = *(_DWORD *)(a1 + 8 + 4 * (v12 + v4)); =x0x4D

    *(_DWORD *)(a1 + 8 + 4 * v11) = v13;

    *(_DWORD *)(a1 + 8 + 4 * (v12 + v4)) = v12;

    *v10 = *(_BYTE *)(a1 + 8 + 4 * (unsigned __int8)(v12 + v13)) ^ *a3;    a3=0x32 v10=7B

    JUMPOUT(&unk_5F9ECE52);

  }

  JUMPOUT(&unk_5F9ED4CA);

}
```

8. 处理后面的每位字符：

```c
v5 = *(_DWORD *)(v8 + 4 * a4); a4=2   v8是int数组，不知道怎么构造出来的

v6 = v4 + v5;   v4 =6B

v7 = *(_DWORD *)(v8 + 4 * v6);   v6=0x111

*(_DWORD *)(v8 + 4 * a4) = v7;   v7=0x26

*(_DWORD *)(v8 + 4 * v6) = v5;   v5=0xA6

*(_BYTE *)(v9 + 1) = *(_BYTE *)(v8 + 4 * (unsigned __int8)(v5 + v7)) ^ *(_BYTE *)(a3 + 1);   (a3是明文十六进制）
```

9. 最终关键的比较判断：

```c
int __fastcall sub_38B0(int a1, int a2)    //比较将用户输入的字符串先每位进行上面的处理，然后再对结果做base64，跟"JPyjup3eCyJjlkV6DmSmGHQ="做对比

{

  int result; // r0@2

  int v3; // [sp+Ch] [bp-10h]@0

  if ( *(_BYTE *)(a1 + 0x38D2 + v3) == *(_BYTE *)(a2 + v3) )    //进行最终的校验判断    JPyjup3eCyJjlkV6DmSmGHQ=!!   cKv07cqf(我们输入的是963258)

    result = sub_390E();    398E

  else

    result = sub_39B0();    3B3A

  return result;

}
```
