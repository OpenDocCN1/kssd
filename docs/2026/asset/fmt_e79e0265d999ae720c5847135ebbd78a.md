## 4. 穷举（跑了40分钟左右）

```
import hashlib

def md5(s):

        d=hashlib.md5() 

        d.update(s.encode('utf8')) 

        return d.hexdigest()

m='0123456789abcdefghijklmnopqrstuvwxyz'

#print(len(m))

def revert(s):

        a=[]

        for i in range(6):

                a.append(ord(s[i]))

        a[0]=a[0]-1

        for i in range(6):

                a[i]=a[i]-i

        r=''

        for i in range(6):

                r+=chr(a[i])

        return r[::-1]   

def solve():

        for k1 in range(36):

                for k2 in range(36):

                        for k3 in range(36):

                                for k4 in range(36):

                                        for k5 in range(36):

                                                for k6 in range(36):

                                                        f=chr(ord(m[k1])+1)+chr(ord(m[k2])+1)+chr(ord(m[k3])+2)+chr(ord(m[k4])+3)+chr(ord(m[k5])+4)+chr(ord(m[k6])+5)

                                                        #print(f)

                                                        s=md5(md5(f))

                                                        if (s[2:12]=='888aeda4ab'):

                                                                print(revert(f))

                                                                return        

print('start')

solve()

print('end')
```
