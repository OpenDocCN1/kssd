    STR = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

    i = 0

    j = 0

    Do Until i = &H18

        k = Asc(Mid$(STR, j + 1, 1)) + &H20

        If k = Asc(Mid$(UN, 1, 1)) Or k = Asc(Mid$(UN, 6, 1)) Then j = j + 1

        NEWSTR = NEWSTR & Mid$(STR, j + 1, 1)

        i = i + 1

        j = j + 2

        If j >= Len(STR) And i < &H18 Then j = 1

    Loop
