```python
import os

import torch

from torch.utils.data import DataLoader

from Configure import Configure

from LSTMModel import LSTMModel

from APIDataSet import APIDataSet, get_data

from sklearn.model_selection import KFold

def train(epoch):
    for batch_idx, data in enumerate(train_loader, 0):
        optimizer.zero_grad()
        inputs, labels = data
        inputs, labels = inputs.to(device), labels.to(device)
        y_pred = modeler(inputs)
        loss = torch.nn.functional.cross_entropy(y_pred, labels.long())
        # if batch_idx % 100 == 99:
        #     print(epoch, loss.item())
        loss.backward()
        optimizer.step()

def test():
    correct = 0
    total = 0
    with torch.no_grad():
        for data in test_loader:
            inputs, target = data
            inputs, target = inputs.to(device), target.to(device)
            outputs = modeler(inputs)
            _, predicted = torch.max(outputs.data, dim=1)
            total += target.size(0)
            correct += (predicted == target.long()).sum()
    acc = 1.0 * 100 * correct / total
    print('测试精度: %f%% [%d/%d]' % (acc, correct, total))
    return acc

if __name__ == '__main__':
    os.environ["CUDA_DEVICE_ORDER"] = "PCI_BUS_ID"
    os.environ["CUDA_VISIBLE_DEVICES"] = "0"
    device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
    conf = Configure()
    x_data, y_data = get_data(conf.calls_path, conf.types_path, conf.max_len)
    kf = KFold(n_splits=conf.folder, shuffle=True)
    k = 1
    total_acc = 0.0
    for (train_index, test_index) in kf.split(x_data):
        modeler = LSTMModel(input_size=conf.max_len,
                            embedding_units=conf.embedding_units,
                            hidden_size=conf.hidden_size,
                            output_size=conf.output_size,
                            dropout_prob=conf.dropout_prob,
                            n_layers=conf.n_layers,
                            bidirectional=conf.bidirectional)
        modeler.to(device)
        optimizer = torch.optim.Adam(modeler.parameters(), conf.lr)
        train_x_data = []
        train_y_data = []
        for i in train_index:
            train_x_data.append(x_data[i])
            train_y_data.append(y_data[i])
        train_dataset = APIDataSet(train_x_data, train_y_data)
        train_loader = DataLoader(train_dataset, batch_size=conf.batch_size,
                                  shuffle=True, num_workers=2)
        test_x_data = []
        test_y_data = []
        for i in test_index:
            test_x_data.append(x_data[i])
            test_y_data.append(y_data[i])
        test_dataset = APIDataSet(test_x_data, test_y_data)
        test_loader = DataLoader(test_dataset, batch_size=conf.batch_size,
                                 shuffle=False, num_workers=2)
        for i in range(conf.epochs):
            train(i)
        print("第%d折测试结果:" % k)
        total_acc += test()
        k = k + 1
    print("%d折交叉验证的平均精度:%f%%" % (conf.folder, total_acc / conf.folder))
```
