```asm
;功能:重建重定位表
;作者:sierra
;发布地点:看雪论坛( http://bbs.pediy.com/showthread.php?p=1063543 )
;发布时间:2012/04/13
;使用方法:
;1、运行到重定位处理语句
;2、打开内存窗口，右键Allocate Memory为代码分配内存（Olly Advanced插件的功能）
;3、右键Allocate Memory为重定位表缓存分配内存（如果人比较懒可以不分配这个，直接使用代码内存的0x100字节后面当做缓存）
;4、根据实际内存地址和实际重定位寄存器修改汇编代码
;5、运行MUltimate Assembler，粘贴代码，汇编
;6、做detour patch（要记得保存原先的代码），call到30000000（实际中需要根据你分配的内存地址修改）
;7、找一个合适的地方下断，使重定位表处理完成后会断下来（如果你像我一样熟练，就可以不下断，直接F4）
;8、F9（F4）运行，断下来之后在40000000（实际中需要根据你分配的内存地址修改）就会有完整的重定位表了，
;   查一下它的实际大小（其实p+已经指向了重定位表的结尾，减去基址都不用加一就直接是表格的大小了），
;   用LordPE保存区段，大小指定为重定位表的大小，再用PETools为dll添加刚刚保存的区段并修改重定位表地址和大小
;9、保存

;补丁开始地址
<0x30000000>        ; 需要修改1/4 根据申请的代码补丁存放地址修改
pushfd
pushad
push edi            ; 需要修改2/4 根据壳中的重定位处理语句中的寄存器修改
push eax            ; 选择修改2/4 一般重定位处理语句最多只涉及三个寄存器
push ecx            ; 选择修改2/4 如果只涉及到一个寄存器则这两个都不用改
db 0x0e8
dd 0          ;call下一条语句
pop ebx        ;得到EIP
add ebx,6      ;得到下面数据的基址
db 0x0eb,0x18  ;jmp short start

;[ebx+0x00]:;  p+
dd 0x40000000        ; 需要修改3/4 根据申请的重定位表存放地址修改

;[ebx+0x04]:;  pVA
dd 0x40000000        ; 需要修改3/4 与p+相同

;[ebx+0x08]:;  pSOB
dd 0

;[ebx+0x0c]:;  nImageBase
dd 0x00870000        ; 需要修改4/4 根据dll实际载入时的映像基址修改

;[ebx+0x10]:;  nRVA
dd 0

;[ebx+0x14]:;  nReloc
dd 0

;jmp 9c0f17
;call 30000020
;add dword ptr ds:[edi], eax
;jmp 9c0ee7

;===================以下代码可以根据具体需要修改====================
;      如果要修改后面的代码，禁止使用和修改ebx，其它的寄存器随便

;start:
pop ecx
pop eax
pop edi
mov [ebx+0x14],edi  ;前面pop之后可以添加一些运算，但最后重定位地址应保存到nReloc

;=========================以下代码不要修改==========================

;hRVA:     ;处理VA
mov eax,dword ptr ds:[ebx+0x14]
mov dword ptr ds:[ebx+0x10],eax    ;☆☆☆映像区块重定位指针地址，需手工设置，一般是寄存器，比如：upx壳是ebx☆☆☆
and dword ptr ds:[ebx+0x10],0FFFFF000
mov eax,dword ptr ds:[ebx+0x0c]
sub dword ptr ds:[ebx+0x10],eax  ;计算pVA块RVA地址
mov eax,dword ptr ds:[ebx+0x04]
mov eax,[eax]
cmp dword ptr ds:[ebx+0x10],eax    ;是否还是同一VA块
db 0x74,0x37;je short hPoint
;sub1:
mov eax,dword ptr ds:[ebx+0x00]
and al,0x0fc
cmp eax,dword ptr ds:[ebx+0x00]
db 0x74,0x0e;je short sub2
mov ecx,dword ptr ds:[ebx+0x00]
mov byte ptr ds:[ecx],0          ;这里就是 ASBSOULT 类型，已经被我sierra复杂化了！
mov ecx,dword ptr ds:[ebx+0x08]
inc dword ptr ds:[ecx]
inc dword ptr ds:[ebx+0x00]
db 0x0eb,0x0ea;jmp short sub1
;sub2:
mov dword ptr ds:[ebx+0x04],eax
mov eax,dword ptr ds:[ebx+0x10]
mov ecx,dword ptr ds:[ebx+0x04]
mov dword ptr ds:[ecx],eax
add dword ptr ds:[ebx+0x00],4  ;修正移动指针
mov eax,dword ptr ds:[ebx+0x00]
mov dword ptr ds:[ebx+0x08],eax
mov eax,dword ptr ds:[ebx+0x08]
mov dword ptr ds:[eax],8
add dword ptr ds:[ebx+0x00],4  ;修正移动指针
db 0x0eb,0x00;jmp short hPoint

;hPoint:;处理Point
mov eax,dword ptr ds:[ebx+0x14]
mov dword ptr ds:[ebx+0x10],eax ;☆☆☆放入Point地址
and dword ptr ds:[ebx+0x10],0FFF ;取低12位
add dword ptr ds:[ebx+0x10],3000 ;定为：HIGLOW 类型，这里没有考虑设计 ASBSOULT 类型，俺觉得没必要复杂化！
mov eax,dword ptr ds:[ebx+0x10]
mov ecx,dword ptr ds:[ebx+0x00]
mov dword ptr ds:[ecx],eax
add dword ptr ds:[ebx+0x00],2  ;修正移动指针
mov eax,dword ptr ds:[ebx+0x08]
add dword ptr ds:[eax],2 ;实时修正SIZEOfBlock

popad
popfd
ret
```

**题外话：**

> **sierra**：dll脱壳时dump的时机要掌握好，本贴建议的最佳dump时机是第一次停在重定位处理语句上的时候，这时候代码已经解密完成，重定位还没有处理，是最佳的dump时机，如果到了OEP再dump，那么一切都晚了，添加重定位表之后将会产生使用上的问题，此时补救的办法就是将在OEPdump出的dll的`ImageBase`更改为在OEPdump时的dll映像基址，这样应该就可以完成补救，当然最优雅的方式就是如前所述的最佳dump时机，但是喜欢哪种，习惯哪种还是自己说了算。
