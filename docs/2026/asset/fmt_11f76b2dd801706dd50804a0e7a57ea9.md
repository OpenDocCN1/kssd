### `printaddress`

```asm
push    ecx
; just a lame routine to print out
push    edx
; a hex address, no explanations ;-)
xor    ecx,ecx
.n
    rol    ebx,4
    mov    dl,bl
    and    dl,0x0F
    cmp    dl,0x9
    ja    .abcdef
    add    dl,'0'
    jmp    .a
.abcdef
    add    dl,'A'-0xA
.a
    mov    [ecx+address],dl
    inc    cl
    cmp    cl,8
    jne    .n
    push    dword 0
    push    dword written
    push    dword 9
    push    dword address
    push    dword [stdout]
    call    \_WriteConsoleA@20
    pop    edx
    pop    ecx
    retn
```
