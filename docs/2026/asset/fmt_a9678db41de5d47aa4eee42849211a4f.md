    FILE* fp;

    char filename[32];
    char line[1024];

    }else{
        snprintf(filename, sizeof(filename), "/proc/%d/maps", pid);
    }
    fp = fopen(filename,"r");
    if(fp != NULL)
    {
        while(fgets(line, sizeof(line), fp))
        {
            if(strstr(line, module_name))

                pch = strtok(line, "-");
                addr = strtoul(pch, NULL, 16);

            }
        }
        fclose(fp);

}
