```
#include <iostream>
#include<Windows.h>
#define PATH L"C:\\Users\\libaobao\\Desktop\\c2服务器.txt"
typedef struct _MYSERVCICE
{
    BYTE buf1;
    BYTE buf2;
    BYTE buf3;
    BYTE buf4;
    WORD port;
    BYTE size;
    BYTE order;
}MYSERVCICE,*PMYSERVCICE;

int main()
{
    DWORD dwRealSize = 0;
    //1.打开文件
    HANDLE hFile = CreateFileW(
        PATH,
        GENERIC_READ,
        NULL,
        NULL,
        OPEN_EXISTING,
        FILE_ATTRIBUTE_NORMAL,
        NULL
    );
    //2.获取文件大小
    DWORD dwSize = GetFileSize(hFile, NULL);
    BYTE* buf = new BYTE[dwSize] {0};
    //3.读取文件
    ReadFile(hFile, buf, dwSize, &dwRealSize, NULL);
    PMYSERVCICE Service = (PMYSERVCICE)(LPVOID)buf;
    DWORD i = 1;
    while (Service->port!=0)
    {
        printf("%d   %u.%u.%u.%u :%u  \n", i,
            Service->buf4, Service->buf3, Service->buf2, Service->buf1, Service->port);
        Service++;
        i++;
    }
    system("pause");
    return 0;
}
```
