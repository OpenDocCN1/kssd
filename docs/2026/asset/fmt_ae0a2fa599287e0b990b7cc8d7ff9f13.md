解密脚本

```
import base64

def xor_bytes(data: bytes, key: bytes) -> bytes:
    k = len(key)
    return bytes(b ^ key[i % k] for i, b in enumerate(data))

def decrypt_from_b64(b64_str: str, key: bytes) -> bytes:
    enc = base64.b64decode(b64_str)
    return xor_bytes(enc, key)

def encrypt_to_b64(plain: bytes, key: bytes) -> str:
    enc = xor_bytes(plain, key)
    return base64.b64encode(enc).decode()

if __name__ == "__main__":
    target_str = "dlZ4bWFsdjUDaiYCeCUqfGYUEhBvFW97dmtxcA=="
    xor_key = b"N1K_ROUT3R"

    recovered = decrypt_from_b64(target_str, xor_key)
    # 常见固件里会有结尾 0 字节，可按需去掉
    recovered_str = recovered.rstrip(b"\x00")

    print("raw bytes:", recovered)
    try:
        print("utf-8:", recovered_str.decode("utf-8"))
    except UnicodeDecodeError:
```
