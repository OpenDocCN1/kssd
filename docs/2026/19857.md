多了取反和交换过程，通过修改DES算法再现这一过程：



```
void DES::crypt(char out[8], char in[8], bool type/* =en */){

 static bool obit[64], temp[32], *li = &obit[0], *ri = &obit[32];

 bytetobit(obit, in, 64);

 //16轮迭代

 if (type == en){

  des_not((char *)obit, 64);

  //初始置换IP

  transt(obit, obit, IP_TABLE, 64);

  // swap

  memcpy(temp, ri, 32);

  memcpy(ri, li, 32);

  memcpy(li, temp, 32);

  for (int i = 0; i < 16; i++){

   memcpy(temp, li, 32);

   f_func(li, subkey[i]);

   xor(li, ri, 32);

   memcpy(ri, temp, 32);

  }

  //逆初始置换IP

  transt(obit, obit, IPR_TABLE, 64);

 }

 else{

  //初始置换IP

  transt(obit, obit, IP_TABLE, 64);

  for (int i = 15; i >= 0; i--){

   memcpy(temp, ri, 32);

   f_func(ri, subkey[i]);

   xor(ri, li, 32);

   memcpy(li, temp, 32);

  }

  // swap

  memcpy(temp, ri, 32);

  memcpy(ri, li, 32);

  memcpy(li, temp, 32);

  //逆初始置换IP

  transt(obit, obit, IPR_TABLE, 64);

  des_not((char *)obit, 64);

 }

 bittobyte(out, obit, 64);

}
```
