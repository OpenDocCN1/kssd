# [原创]蒙特卡罗算圆周率

import random
def main():
    sum = 1000000
    sum\_in = 0
    for i in xrange(0,sum):
        x = random.random()
        y = random.random()
        if (x-0.5)\*(x-0.5)+(y-0.5)\*(y-0.5)<=0.25:
            sum\_in = sum\_in+1

    print('%s/%s = %s'%(sum\_in,sum,float(sum\_in)/sum))
    print('PI = %s'%(4\*float(sum\_in)/sum))
