int numShift(int *num)
{
    int temp;
    temp = 0x343FD * *num + 0x269EC3;
    *num = temp;
    return ((unsigned int)temp >> 16) & 0x7FFF;

bool __cdecl keyGenerator(unsigned char *keyset, const unsigned char *reg_hex_num, unsigned int blockSize)

{
    int counter;
    unsigned int num_;
    int num;
    unsigned char n[4];
    memcpy(keyset, reg_hex_num, blockSize);
    for (int i = 0; i < (signed int)blockSize; ++i)
    {
        num = reg_hex_num;
        counter = 0;
        do
        {
            num_ = numShift(&num);
            memcpy(&n[0], &num_, 4);
            keyset[counter] ^= n[0] ^ n[1] ^ n[2] ^ n[3];
            ++counter;
        } while (counter < (signed int)blockSize);
    }
    return 1;
}
