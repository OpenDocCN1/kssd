procedure RegCreate(A,B:PAnsiChar;Reg\_Nag:integer);
 var
    cbDataB:integer;
    hkey\_reg:HKEY;
 begin
      cbDataB:=length(B)+1;  //cbData 参数为保存键值数据的长度，好像W2K，XP有用。

           //写入键值数据

           // 关闭
           RegCloseKey(hkey\_reg);
          end;

            RegCreateKey(HKEY\_CURRENT\_USER,'Software\zgsq\bfUser',hkey\_reg);
            RegSetValueEx(hkey\_reg,A,0,reg\_sz,B,cbDataB);
            RegCloseKey(hkey\_reg);
          end;
      end;
  end;

      begin
        l :=pointer(GetWindowLong(h\_Wnd, GWL\_USERDATA));
        CallWindowProc(l, h\_Wnd, Msg, wParam, lParam);
      end;
  end;
end;
