#### `md5` 的 Hook 方法

```javascript
Java.perform(function () {
    Java.use('com.dodonew.online.util.Utils').md5.implementation = function (data) {
        console.log(data);
        var res = this.md5(data);
        console.log(res);
        return res; // 劫持完代码看一下怎么个事就给你反回去
    };
});
```

![](https://bbs.kanxue.com/upload/attach/202211/963851_99G2JWB5EG8EXBX.jpg)

#### `encodeDesMap` 的 Hook 方法

```javascript
Java.perform(function () {
    var RequestUtil = Java.use('com.dodonew.online.http.RequestUtil');

    RequestUtil.encodeDesMap
        .overload('java.lang.String', 'java.lang.String', 'java.lang.String')
        .implementation = function (data, data2, data3) {
            console.log('\ndata=', data, '\ndata2=', data2, '\ndata3=', data3);

            var ees = this.encodeDesMap(data, data2, data3);
            console.log(ees);

            return this.encodeDesMap(data, data2, data3); // 劫持完代码看一下怎么个事就给你反回去
            return ees;
        };
});
```
