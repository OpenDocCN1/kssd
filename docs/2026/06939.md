   function ROL(r: LongWord; bit: byte):
LongWord;
   var
     ii: byte;
   begin
    for ii:=1 to bit do
       if r and $80000000
= $80000000 then
         r := (r shl 1) or $01

    else
         r := (r shl 1);

  Result := r;
   end;

   procedure II(var a: LongWord; b, c, d: LongWord; k, s,

not d)) + MSGs[k] + T[i]), s);
