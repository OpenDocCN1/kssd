import os

v_type = tempfile.mktemp()

v_assoc = tempfile.mktemp()

v_ext = []

os.system('assoc >%s' % v_assoc)

with open(v_type) as f:
    for line in f.readlines():
            if line.find('wscript.exe /B  "%1"') > 0:

                with open(v_assoc) as f:

                            v_ext.append(line.split('=')[0])
