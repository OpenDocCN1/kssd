## 算法注册机

```
#include <stdio.h>
#include <conio.h>
#include <windows.h>

/*
COM Explorer 2.0算法注册机 v0.2		PS:对错误之处进行了改良,简单优化了下
如有错误,请指正,谢谢!代码有点错乱,见笑了!
*/

int Key(unsigned char Name[],int Len,char Key[])//Key主要算法
{
	char Table[]="0123456789";
	int m = 5;
	unsigned int k=0;
	unsigned int n=0;
	for(int i=0;i<9;i++)
	{
		n = Name[i%Len];//直接取得的Name位数值,下边直接使用,不用多次计算
		if(n>'z')
			n = n | 0xFFFFFF00;//为了支持中文用户名
		k = n+(n+n*2)*4+Table[i%0xA]+(i*i*i*Len);//直接把算法合成一行了
		if(m == 9 || m == 13)
		{
			Key[m++] = '-';
			Key[m++] = Table[k%0xA];
			continue;
		}
		Key[m++] = Table[k%0xA];
	}
	return 0;
}

int main()
{
	static char key[17] = "6769-";
	unsigned char name[21] = {NULL};
	int len = 0;
	while(true)
	{
		printf("Name:");
		scanf("%20s",name);//太长用户名也没必要,最多让输入20位,防溢出
		len = strlen((char*)name);
		if(len<8)
		{
			printf("\n用户名不能小于8位,请重新输入!(请按任意键)");
			getch();
			system("cls");
		}
		else
			break;
	}
	for(int n=0;name[n]!=0;n++)//对汉化版中的Key不支持大写进行改进
	{//跟踪发现,软件会把输入的大写字母转换成小写字母
		if(name[n]>='A' && name[n]<='Z')
			name[n] += 32;
	}
	Key(name,len,key);
	printf("\nKey:%s\n",key);
	getch();
	return 0;
}
```
