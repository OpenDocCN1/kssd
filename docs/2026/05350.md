# [原创]浅析某P在联盟中Gaxxrpcs中的一个算法

最近一直在跟bambooqj,kido,brack几位老师苦苦学习

刚好进入一个疲惫期,恰巧朋友让帮忙看联盟中某个算法,就简单看了看

该算法在rpcs中,是以线程模式创建的,整个线程的流程是

获取加密数据->解密数据->获取hash->存放到全局变量->异步通信

1.首先是获取到的数据进行解密,解密结果如下(结果是某些脚本的HOOK地址):

![](https://bbs.kanxue.com/upload/attach/201903/654430_XAGNFF69MBCCQPS.png)

结果是一个数据结构,包含引索,模块名字,地址偏移,检测长度,此结构一共有0xC个大小

然后是根据这个结构逐一进行hash的结果获取

2.获取hash值:

该函数定义大概是:

```
uint32_t __cdecl get_hash(uintptr_t _address,uint32_t _size,uint32_t _checkflag)
```

函数实现,为了方便观看,我简单的处理了下tvm的分支,最后代码如下:

```
push ebp
mov ebp,esp
sup esp,0x18
mov dword ptr ss:[ebp-0x8],0
mov eax,dword ptr ss:[ebp+0x8]
mov dword ptr ss:[ebp-0x4],eax
mov dword ptr ss:[ebp-0x10],0
LOOP:
mov eax,dword ptr ss:[ebp-0x10]
cmp eax,dword ptr ss:[ebp+0xC]
jnb LAST

mov ecx,dword ptr ss:[ebp-0x10]
and ecx,1
jnz odd					;判断是否奇数

mov edx,dword ptr ss:[ebp-8]
shl edx,7
mov eax,dword ptr ss:[ebp-4]
movzx ecx,byte ptr ds:[eax]
xor edx,ecx
mov eax,dword ptr ss:[ebp-0x8]
shr eax,0x3
xor edx,eax
mov dword ptr ss:[ebp-0x14],edx
jmp NEXT

odd:					;如果是奇数则到这里
mov ecx,dword ptr ss:[ebp-0x8]
shl ecx,0xB
mov edx,dword ptr ss:[ebp-0x4]
movzx eax,byte ptr ds:[edx]
xor ecx,eax
mov edx,dword ptr ss:[ebp-0x8]
shr edx,0x5
xor ecx,edx
not ecx
mov dword ptr ss:[ebp-0x14],ecx

NEXT:
mov eax,dword ptr ss:[ebp-0x8]
xor eax,dword ptr ss:[ebp-0x14]
mov dword ptr ss:[ebp-0x8],eax
mov ecx,dword ptr ss:[ebp-0x4]
add ecx,0x1
mov dword ptr ss:[ebp-0x4],ecx
mov edx,dword ptr ss:[ebp-0x10]
add edx,0x1
mov dword ptr ss:[ebp-0x10],edx
jmp LOOP

LAST:
mov ecx,dword ptr ss:[ebp+0xC]
and ecx,0x1
jne LstOdd

mov edx,dword ptr ss:[ebp-0x8]
shl edx,0x7
movzx eax,byte ptr ss:[ebp+0x10]
xor edx,eax
mov ecx,dword ptr ss:[ebp-0x8]
shr ecx,0x3
xor edx,ecx
mov dword ptr ss:[ebp-0x18],edx
jmp LASTCMP

LstOdd:
mov edx,dword ptr ss:[ebp-0x8]
shl edx,0xB
movzx eax,byte ptr ss:[ebp+0x10]
xor edx,eax
mov ecx,dword ptr ss:[ebp-0x8]
shr ecx,0x5
xor edx,ecx
not edx
mov dword ptr ss:[ebp-0x18],edx

LASTCMP:
mov edx,dword ptr ss:[ebp-0x8]
xor edx,dword ptr ss:[ebp-0x18]
mov dword ptr ss:[ebp-0x8],edx
lea eax,dword ptr ss:[ebp-0x5]
mov dword ptr ss:[ebp-0xC],eax
mov ecx,dword ptr ss:[ebp-0xC]
movzx edx,byte ptr ds:[ecx]
cmp edx,0x7F
jle EXIT:
mov eax,dword ptr ss:[ebp-0xC]
mov byte ptr ds:[eax],0x7F

EXIT:
mov eax,dword ptr ss:[ebp-0x8]
mov esp,ebp
pop ebp
retn
```

写的有点粗糙,我只是为了观察,也不是为了bypass或是做其他的.

最后获取的结果存在于全局变量

结构是:

```
struct _hash_table_
{
	byte    type;
	short   index;
	int	hash;
}
```
