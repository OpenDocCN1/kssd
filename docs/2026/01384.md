
# [原创]某pdf转word v6.3.0.2算法分析

【文章标题】某pdf转word v6.3.0.2算法分析

【原版下载】`c4fK9s2c8@1M7q4)9K6b7g2)9J5c8W2)9J5c8Y4N6%4N6#2)9J5k6i4m8V1k6X3y4%4L8%4u0V1i4K6u0W2j5$3^5`.

【保护方式】序列号

【分析过程】

## 一. 去掉随机基址

直接OD载入程序，入口是“一`call`一`jmp`”，基本上就是VS高版本编译的

![](upload/attach/201712/770162_xskvrnnt5pltv97.png)

为了避免随机基址的影响，先去除随机基址。找到“PE”下一行偏移为6的字节处，将“02”修改为“03”，可去掉随机基址

![](upload/attach/201712/770162_peacp38pflivh7h.png)

//功能：硬盘序列号
//参数：
//  lpszHD：传出参数，存储最终计算的硬盘信息
//  len：默认参128
//--------------------------------------------------------------  
BOOL GetHDSerial(char *lpszHD, int len/*=128*/)

{

    BOOL        bRet = FALSE;

    DWORD       bytesRtn = 0;

    char        szhd[80] = { 0 };

    PIDSECTOR   phdinfo;

    HANDLE      hDrive = NULL;

    GETVERSIONOUTPARAMS vers;

    SENDCMDINPARAMS     in;

    SENDCMDOUTPARAMS    out;

    ZeroMemory(&vers, sizeof(vers));

    ZeroMemory(&in, sizeof(in));

    ZeroMemory(&out, sizeof(out));

    //搜索四个物理硬盘，取第一个有数据的物理硬盘  

    for (int j = 0; j < 4; j++)

    {

        sprintf(szhd, "\\\\.\\PhysicalDrive%d", j);

        hDrive = CreateFileA(szhd,

            GENERIC_READ | GENERIC_WRITE,

            FILE_SHARE_READ | FILE_SHARE_WRITE,

            0,

            OPEN_EXISTING,

            0,

            0);

        if (NULL == hDrive)

        {

            continue;

        }

        if (!DeviceIoControl(hDrive, DFP_GET_VERSION, 0, 0, &vers, sizeof(vers), &bytesRtn, 0))

        {

            CloseHandle(hDrive);

            hDrive = NULL;

            continue;

        }

        //If IDE identify command not supported, fails

        if (!(vers.fCapabilities & 1))

        {

            CloseHandle(hDrive);

            hDrive = NULL;

            continue;

        }

        //Identify the IDE drives

        if (j & 1)

        {

            in.irDriveRegs.bDriveHeadReg = 0xb0;

        }

        else

        {

            in.irDriveRegs.bDriveHeadReg = 0xa0;

        }

        if (vers.fCapabilities&(16 >> j))

        {

            //We don't detect a ATAPI device.

            CloseHandle(hDrive);

            hDrive = NULL;

            continue;

        }

        else

        {

            in.irDriveRegs.bCommandReg = 0xec;

        }

        in.bDriveNumber = j;

        in.irDriveRegs.bSectorCountReg = 1;

        in.irDriveRegs.bSectorNumberReg = 1;

        in.cBufferSize = 512;

        if (!DeviceIoControl(hDrive, DFP_RECEIVE_DRIVE_DATA, &in, sizeof(in), &out, sizeof(out), &bytesRtn, 0))

        {

            //"DeviceIoControl failed:DFP_RECEIVE_DRIVE_DATA"<<endl;

            CloseHandle(hDrive);

            hDrive = NULL;

            continue;

        }

        phdinfo = (PIDSECTOR)out.bBuffer;

        char    s[21] = { 0 };

        memcpy(s, phdinfo->sSerialNumber, 20);

        s[20] = 0;

        ChangeByteOrder(s, 20);

        memcpy(lpszHD, s, 20);

        if (strlen(lpszHD) != 0)

        {

            bRet = TRUE;

        }

        break;

    }

    CloseHandle(hDrive);

    hDrive = NULL;

    return bRet;

}

{

    int nResult = 0;

    signed int j;

    unsigned int uNum = 0;

    signed int i;

    for (i = 0; i < 256; ++i)

    {

        uNum = i;

        for (j = 8; j > 0; --j)

        strlwr((char*)uTmpBuf);          //将字符串转为小写
    }

    //尾部追加#
    strcat((char*)uTmpBuf, "#");
    //计算MD5值
    MDString((char*)uTmpBuf, uMD5Buf);
    HexToStr(uTmpBuf, uMD5Buf, 16);  //将MD5数据转为16进制字符串
    //截取前半部分字符串
    uTmpBuf[16] = '\0';
    memcpy(pMachineID, uTmpBuf, 16);
    return pMachineID;
}
