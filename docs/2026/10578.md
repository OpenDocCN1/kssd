void \_stdcall addstr(char\*str1,char\*str2){
        \_asm{
                                mov edi,[ebp+8]
test\_str1:
                        mov al,byte ptr es:[edi]
                        test al,al
                        jz addstart
                        inc edi
                        jmp test\_str1

addstart:        mov esi,[ebp+12]
round:                mov al, byte ptr ds:[esi]
                        mov byte ptr es:[edi], al
                        inc esi
                        inc edi
                        test al,al
                        jnz round



        }

}
函数功能:
      将第二个字符串添加到第一个字符串的尾部
    要求第一个字符串所在处有足够大的缓冲区
