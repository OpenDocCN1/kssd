Option Explicit

If WSH.Arguments.Count <> 1 Then
	ShowUsage
	WSH.Quit
End If

If IsMD5(WSH.Arguments(0)) Then 
	WSH.Echo "Wait..."
	WSH.Echo CMD5(WSH.Arguments(0))
Else
	WSH.Echo "It's not a md5 string!"
End If



Sub ShowUsage()'MPL@bbs.pediy.com 2009.07.24
	WSH.Echo _
		"CMD5 v1.0" & vbCrLf & _
		vbCrLf & _
		"Usage: cscript //nologo """ & WSH.ScriptName & """ md5pass" & vbCrLf & _
		"e.g.:" & vbCrLf & _
		"cscript //nologo """ & WSH.ScriptName & """ 49ba59abbe56e057"
End Sub

Function IsMD5(sMD5)
	Dim oReg
	Set oReg = New RegExp
	oReg.Pattern = "^[0-9A-Za-z]{16}$|^[0-9A-Za-z]{32}$"
	IsMD5 = oReg.Test(sMD5)
	Set oReg = Nothing
End Function

	Dim bGet'是否已经查询到MD5对应的字符串

		WSH.Sleep 100
		iTime = iTime + 100
	Loop
	If iTime > TIMEOUT Then
		bGet = False
	Else
		oIE.Document.All("md5pass").Value = sMD5
		oIE.Document.All("f").Submit
		iTime = 0
		Do While (oIE.Busy Or oIE.ReadyState <> 4) And (iTime <= TIMEOUT)
			WSH.Sleep 100
			iTime = iTime + 100
		Loop
		If iTime > TIMEOUT Then
			bGet = False
		Else
			sReturn = oIE.LocationURL
			If LCase(sReturn) = "ee7K9s2c8@1M7q4)9K6b7g2)9J5c8W2)9J5c8Y4N6%4N6#2)9J5k6i4S2E0k6o6g2Q4x3X3g2G2M7X3N6Q4x3V1k6E0k6o6g2Q4x3V1k6Y4k6i4c8H3j5i4y4K6i4K6u0W2j5i4y4H3i4K6y4r3N6s2W2H3k6g2)9K6c8r3&6G2" Then
				bGet = False				
			Else
				bGet = True
				sResult = Mid(sReturn, 42, Len(sReturn) - 41)
			End If
		End If		
	End If

	If Not bGet Then
		oIE.Navigate "004K9s2c8@1M7q4)9K6b7g2)9J5c8W2)9J5c8Y4N6%4N6#2)9J5k6i4S2E0k6o6g2Q4x3X3g2G2M7X3N6Q4x3V1k6A6L8X3c8W2P5q4)9#2k6X3y4F1x3W2)9J5k6h3S2@1L8b7`.`."

			WSH.Sleep 100
			iTime = iTime + 100
		Loop

			bGet = False
		Else
			oIE.Document.All("md5pass").Value = sMD5
			oIE.Document.All("f").Submit
			iTime = 0
			Do While (oIE.Busy Or oIE.ReadyState <> 4) And (iTime <= TIMEOUT)
				WSH.Sleep 100
				iTime = iTime + 100
			Loop
			If iTime > TIMEOUT Then
				bGet = False
			Else
				sReturn = oIE.LocationURL
				If LCase(sReturn) = "5a1K9s2c8@1M7q4)9K6b7g2)9J5c8W2)9J5c8Y4N6%4N6#2)9J5k6i4S2E0k6o6g2Q4x3X3g2G2M7X3N6Q4x3V1k6E0k6o6g2Q4x3V1k6Y4k6i4c8H3j5i4y4K6x3W2)9J5k6h3q4K6M7q4)9K6c8Y4c8&6M7r3g2Q4x3@1c8F1L8H3`.`." Then

	oIE.Quit

End Function
