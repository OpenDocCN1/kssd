var
  I, iLen: Integer;

  t: DWORD;

begin
  iLen := Length(s);

  t := 0;
  if iLen <= 4 then
  begin
    for I := iLen downto 1 do

    begin
      t := t shl 8 + Ord(S[I]);

    end
  end
  else
  begin
    for I := 4 downto 1 do

    begin
      t := t shl 8 + Ord(S[I]);

    end;

    for I := iLen downto 5 do

    begin
      t := t + Ord(S[I]);

    end
  end;
