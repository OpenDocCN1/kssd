	gsk = GetPowAndMod(g, k, p) % p;

long GetPowAndMod(long g, long x, long p)
{
	long Temp = g;

	if (x > 2)
		return (GetPowAndMod(g, x - 2, p) % p) * (g * g % p);
	else
	{
		for (long j = 1 ; j < x ; j++)
			Temp *= g;
		return Temp;
	}
}
