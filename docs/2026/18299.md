  66:81EC 4404         sub     sp, 444
  50                   push    eax
  33C0                 xor     eax, eax

  50                   push    eax
  33C0                 xor     eax, eax

  50                   push    eax
  B8 777A837C          mov     eax, kernel32.Beep
  FFD0                 call    eax
  58                   pop     eax

  B9 5BF81775          mov     ecx, 7517F85B
