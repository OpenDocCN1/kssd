## 结果

这题很精彩，但是出现了多解。先上代码

```python
#!/usr/bin/env python
# -*- coding: utf8 -*-

import subprocess
import base64
import random
import string

# 需要安装 https://github.com/guanzhi/GmSSL

# 计算 SM3 哈希
def get_sm3(data):
    cmd = 'echo -n "{}" | gmssl sm3'.format(data)
    r = subprocess.check_output(cmd, shell=True)
    sm3 = r[-65:-1]
    return sm3

# 生成指定长度的随机字符串
def get_rand_str(length):
    r = ''
    candidate = string.ascii_letters + string.digits
    for i in range(length):
        r += random.choice(candidate)
    return r

if __name__ == '__main__':
    for i in range(32):
        # data长度为9的倍数，保证两次base64编码后没有 '='
        # 开头的 '/' 用于莫尔斯解码时产生一个空格
        data = '/' + get_rand_str(8)
        b64b64 = base64.b64encode(base64.b64encode(data))
        # 判断base64中是否包含 '/' 或 '+'
        if '/' in b64b64 or '+' in b64b64:
            continue
        sm3 = get_sm3(data[:3])
        print(b64b64+sm3)
```
