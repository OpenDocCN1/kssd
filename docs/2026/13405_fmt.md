            }
        }
        else
        {
            cout << "yes1" << endl;
        }

        //导入私钥
        bRet = CryptImportKey(hCryptProv, pPrivateKey, dwPrivateKeyLength, NULL, 0, &hCryptKey);

        {
            ShowError("CryptImportKey");
