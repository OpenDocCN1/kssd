```python
# -\*- coding:UTF-8 -\*-
try:
# 这个try except是用来在pycharm中导入PYI文件,用来实现自动代码补全的
# 实际ghidra运行时会执行pass
from ghidra\_builtins import \*
except:
pass
from ghidra.app.decompiler import DecompInterface as DecompInterface
decomplib = DecompInterface()
decomplib.openProgram(currentProgram)
timeout =30
function =getFunction("fopen64") # ghidra.program.model.listing.Function
dRes = decomplib.decompileFunction(function,timeout,getMonitor()) # ghidra.app.decompiler.DecompileResults
hfunction = dRes.getHighFunction() # ghidra.program.model.pcode.HighFunction
n = 0
ops = hfunction.getPcodeOps() # Iterator[ghidra.program.model.pcode.PcodeOpAST]
while ops.hasNext():
pcodeOpAST =ops.next()
print(str(n)+" "+str(pcodeOpAST))
n+=1
```
