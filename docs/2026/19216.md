Arduino中秋快乐

作为一个不懂得如何表达自己、不会说话的男人，在全家团圆的中秋节之际，我应该向那些关心我、懂我以及我关心的人们，说些什么呢？
此时此地此景，我身边的键盘、鼠标和Arduino回应了我，来一段默默的祝福：happy Mid-Autumn festival。
下我一跳，怎么突然来了一句英文呢？此时，键盘出声了，你听说过摩尔斯电码有中文。

程序源码：
int nLED =13;
void setup() {
  pinMode(nLED,OUTPUT);
}
void Flash(int n)
{
  digitalWrite(nLED,HIGH);
  delay(n);
  digitalWrite(nLED,LOW);
  delay(n);
}
void Happy()
{
  Flash(200);Flash(200);Flash(200);Flash(200);//H ．．．．
  delay(300);
  Flash(200);Flash(600);//A  ．━
  delay(300);
  Flash(200);Flash(600);Flash(600);Flash(200);//P  ．━ ━ ．
  delay(300);
  Flash(200);Flash(600);Flash(600);Flash(200);//P  ．━ ━ ．
  delay(300);
  Flash(600);Flash(200);Flash(600);Flash(600);//Y  ━ ．━ ━
  delay(900);  //没用分隔，停一下吧

  Flash(600);Flash(600);//M  ━ ━
  delay(300);
  Flash(200);Flash(200);//I  ．．
  delay(300);
  Flash(600);Flash(200);Flash(200);//D  ━ ．．
  delay(900);//没用分隔，停一下吧

  Flash(200);Flash(600);//A  ．━
  delay(300);
  Flash(200);Flash(200);Flash(600);//U ．．━
  delay(300);
  Flash(600);Flash(600);//M  ━ ━
  delay(300);
  Flash(600);Flash(200);//N  ━ ．
  delay(300);
  Flash(200);Flash(200);Flash(600);Flash(200);//F  ．．━ ．
  delay(300);
  Flash(200);//E .
  delay(300);
  Flash(200);Flash(200);Flash(200);//S ．．．
  delay(300);
  Flash(600);//T ━
  delay(300);
  Flash(200);Flash(200);//I ．．
  delay(300);
  Flash(200);Flash(200);Flash(200);Flash(600);//V ．．．━
  delay(300);
  Flash(200);Flash(600);//A ．━
  delay(300);
  Flash(200);Flash(600);Flash(200);Flash(200);//L ．━ ．．
  delay(3000);
}
void loop() {

  Happy();

}
