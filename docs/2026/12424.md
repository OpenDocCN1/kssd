定位到pe的导入函数表，然后依次通过get\_export函数查询函数的地址，然后修改地址表。

static int get\_export(char \*name, generic\_func \*func)
{
        int i, j;

        struct wrap\_export \*exports[] = {
                ntoskernel\_exports,
                ntoskernel\_io\_exports,
                ndis\_exports,
                crt\_exports,
                hal\_exports,
                rtl\_exports,
#ifdef ENABLE\_USB
                usb\_exports,
#endif
        };

        for (j = 0; j < ARRAY\_SIZE(exports); j++)
                for (i = 0; exports[j][i].name != NULL; i++)
                        if (strcmp(exports[j][i].name, name) == 0) {
                                \*func = exports[j][i].func;
                                return 0;
                        }

        for (i = 0; i < num\_pe\_exports; i++)
                if (strcmp(pe\_exports[i].name, name) == 0) {
                        \*func = pe\_exports[i].addr;
                        return 0;
                }

        return -1;
}
