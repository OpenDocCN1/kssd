int __stdcall _CalcAddrOffset (	char*	pPartOfFileImage,
								int		nBaseAddr,
								int		word_404460)

{

	int*	pn	= 0;

	int		n	= 0;

	int		i	= 0;

	pn = (int*)pPartOfFileImage;
	n = nBaseAddr;
	pn += 3;
	for ( i = 0; i < word_404460; i++ )
	{
		if ( *pn <= n )
		{
			if ( (*pn + *(pn + 1)) > n )
			{
				n = (n - *pn) + *(pn + 2);
			}
		}
		pn += 0xA;
	}
	return n;
}
