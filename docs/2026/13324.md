GetFunctionAddress:
    mov    ebx, [eax + 3Ch]                    ;pe header
    add    ebx, eax;加上基址
    add    ebx, 120
    mov    ebx, [ebx]
    add    ebx, eax                        ;Export Address

    mov    ecx, [ebx + 32]         ;functionname
    add    ecx, eax
    push    esi
    push    edx

CompareNext:
    pop    edx
    pop    esi
    inc    edx
    mov    edi, [ecx]
    add    edi, eax
    add    ecx, 4
    push    esi
    push    edx

    mov    dl, [edi]
