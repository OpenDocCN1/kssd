```
if ( WriteSection )
    {
      v16 = MmUserProbeAddress;
      if ( WriteSection < (unsigned int)MmUserProbeAddress )
        v16 = (const void *)WriteSection;
      memcpy(&ServerView, v16, 0x18u);
      if ( ServerView != 0x18 )
        goto LABEL_17;
      if ( WriteSection & 3 )
        goto LABEL_19;
      if ( WriteSection >= (unsigned int)MmUserProbeAddress )
        *(_BYTE *)MmUserProbeAddress = 0;
      *(_BYTE *)WriteSection = *(_BYTE *)WriteSection;
      *(_BYTE *)(WriteSection + 20) = *(_BYTE *)(WriteSection + 20);
      v13 = MmUserProbeAddress;
    }
```
