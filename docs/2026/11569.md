# [原创]实时显示数据包发送和接受情况

下面是一个显示当前主机实时显示数据包发送和接收情况的程序，请大家使用。


```
#include "stdafx.h"
#include <winsock2.h>
#include <iphlpapi.h>
#include <stdio.h>

#pragma comment(lib, "IPHLPAPI.lib")



int main( int argc, char* argv[] )
{  	
	MIB_IFROW IfRow;

	DWORD dwRetVal = 0;

	IfRow.dwIndex = 2;	//根据情况需要修改

	while( true )
	{
		dwRetVal = ::GetIfEntry( &IfRow );
		if( dwRetVal == NO_ERROR)
			printf("\tPack-out:%d ,Pack-in:%d\n", IfRow.dwOutUcastPkts + IfRow.dwOutNUcastPkts, IfRow.dwInUcastPkts  );
		Sleep( 1000 );
	}

	return 0;
}
```
