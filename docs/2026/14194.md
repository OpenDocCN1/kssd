### 一、ELF动态加载核心流程
ELF动态加载器的核心工作流程可分为以下阶段：
1. \*\*文件解析\*\*：读取ELF头部、程序头表和动态段
2. \*\*内存映射\*\*：按LOAD类型段要求建立内存映射
3. \*\*符号解析\*\*：处理动态符号表(.dynsym)和重定位表(.rela.plt)
4. \*\*重定位修正\*\*：修正GOT/PLT等需要动态计算的地址
5. \*\*初始化执行\*\*：执行.init\_array初始化函数
---

```rust
use goblin::elf::program\_header::{PT\_DYNAMIC, DT\_NEEDED};
// 解析动态段获取依赖库列表
fn parse\_dynamic(elf: &Elf) -> Result<Vec<String>> {
let dynamic\_phdr = elf.program\_headers
.iter()
.find(|ph| ph.p\_type == PT\_DYNAMIC)
.context("Missing dynamic segment")?;
let dynamic\_data = &elf.data[dynamic\_phdr.p\_offset as usize..];
let mut deps = Vec::new();
// 遍历动态标签
let mut i = 0;
while i < dynamic\_data.len() {
let tag = unsafe {
\*(dynamic\_data.as\_ptr().add(i) as u64
};
let val = unsafe {
\*(dynamic\_data.as\_ptr().add(i + 8) as u64
};
match tag {
DT\_NEEDED => {
let str\_offset = elf.dynstrtab.get(val as usize)
.context("Invalid string offset")?;
deps.push(str\_offset.to\_string());
}
// 处理其他动态标签...
}
i += 16; // 每个动态项16字节
}
Ok(deps)
}
// 递归加载依赖库
fn load\_dependencies(&mut self) -> Result<()> {
for dep in &self.dependencies {
let child = DynamicLoader::load(dep)?;
self.dependencies.push(Arc::new(child));
}
Ok(())
}
```
\*\*关键机制\*\*：
- 动态段包含`DT\_NEEDED`条目指示依赖库
- 使用`dynstrtab`获取库名字符串
- 通过`Arc`共享已加载库避免重复加载

```rust
fn parse\_version\_requirements(&self) -> Result<Vec<VersionedSymbol>> {
let versym = self.elf.sections
.get\_by\_name(".gnu.version\_r")
.context("No version section")?;
let data = versym.data();
let mut symbols = Vec::new();
let mut offset = 0;
while offset < data.len() {
let vna\_flags = u16::from\_le\_bytes([data[offset], data[offset+1]]);
let vna\_ndx = u16::from\_le\_bytes([data[offset+2], data[offset+3]]);
let vna\_hash = u32::from\_le\_bytes([
data[offset+4], data[offset+5],
data[offset+6], data[offset+7]
]);
let name\_ptr = offset + 8;
let name = CStr::from\_bytes\_until\_nul(&data[name\_ptr..])?
.to\_str()?;
symbols.push(VersionedSymbol {
name: name.to\_string(),
version: None, // 需要额外解析
hash: vna\_hash,
});
offset += name.len() + 9; // 结构对齐
}
Ok(symbols)
}
// 版本感知的符号查找
fn find\_symbol\_with\_version(&self, name: &str, ver: &str) -> Result<usize> {
for lib in &self.dependencies {

.find(|s| s.name == name && s.version.as\_deref() == Some(ver))

return Ok(sym.value);

bail!("Symbol {}({}) not found", name, ver)
}
```

### 十一、复杂重定位类型处理
#### 1. R\_X86\_64\_RELATIVE处理
```rust
fn apply\_relative\_relocs(&self) -> Result<()> {
let rela\_plt = self.elf.sections
.get\_by\_name(".rela.plt")
.context("No relocation section")?;
for reloc in parse\_rela\_entries(rela\_plt.data())? {
if reloc.r\_type != R\_X86\_64\_RELATIVE {
continue;
}
let target\_addr = self.base\_addr + reloc.r\_addend as usize;
let reloc\_addr = self.base\_addr + reloc.r\_offset as usize;
unsafe {
\*(reloc\_addr as \*mut usize) = target\_addr;
}
}
Ok(())
}
// 解析Rela条目
fn parse\_rela\_entries(data: &[u8]) -> Result<Vec<Reloc>> {
data.chunks\_exact(24) // 每个Rela条目24字节
.map(|chunk| {
let offset = u64::from\_le\_bytes(chunk[0..8].try\_into()?);
let info = u64::from\_le\_bytes(chunk[8..16].try\_into()?);
let addend = i64::from\_le\_bytes(chunk[16..24].try\_into()?);
Ok(Reloc {
r\_offset: offset,
r\_type: (info & 0xFFFFFFFF) as u32,
r\_addend: addend,
})
})
.collect()
}
```
