	if (!sock)
		return err;

	if (err)
		goto out_put;

	entry = mmsg;
	compat_entry = (struct compat_mmsghdr __user *)mmsg;

					     &msg_sys, flags & ~MSG_WAITFORONE,

			if (err < 0)
				break;
			err = __put_user(err, &compat_entry->msg_len);
			++compat_entry;
		} else {
			err = ___sys_recvmsg(sock,

					     &msg_sys, flags & ~MSG_WAITFORONE,
					     datagrams);
			if (err < 0)
				break;
			err = put_user(err, &entry->msg_len);
			++entry;
		}

		if (err)
			break;
		++datagrams;

		/* MSG_WAITFORONE turns on MSG_DONTWAIT after one packet */
		if (flags & MSG_WAITFORONE)
			flags |= MSG_DONTWAIT;

		if (timeout) {
			ktime_get_ts(timeout);
			*timeout = timespec_sub(end_time, *timeout);
			if (timeout->tv_sec < 0) {
				timeout->tv_sec = timeout->tv_nsec = 0;
				break;
			}

			/* Timeout, return less than vlen datagrams */
			if (timeout->tv_nsec == 0 && timeout->tv_sec == 0)
				break;
		}

		/* Out of band data, return right away */
		if (msg_sys.msg_flags & MSG_OOB)
			break;
	}

out_put:
	fput_light(sock->file, fput_needed);

	if (err == 0)
		return datagrams;

	if (datagrams != 0) {
		/*
		 * We may return less entries than requested (vlen) if the
		 * sock is non block and there aren't enough datagrams...
		 */
		if (err != -EAGAIN) {
			/*
			 * ... or  if recvmsg returns an error after we
			 * received some datagrams, where we record the
			 * error to return on the next call or if the
			 * app asks about it using getsockopt(SO_ERROR).
			 */
			sock->sk->sk_err = -err;
		}

		return datagrams;
	}

exp程序的注释解释了，选择net\_sysctl\_root.permissions，是因为它上方有16个'\0'字节：

```
could probably whack the high bits of some pointer with nanoseconds,

执行objdump，查看内核文件.data节区中，net\_sysctl\_root上方的内容，就可以认证这一点，通常由于结构体对齐填充了0，或者编译器安排了值为0的变量在它上方：

这样一来，将上图标红的0xff字节所在地址0xffffffff81cdf496+7（图中按大端字节序书写），作为timeout地址传给\_\_sys\_recvmmsg()，就可以使timeout->tv\_sec = 0xff（标红的8字节），timeout->tv\_nsec = 0（更上方的8字节）：

![](https://bbs.kanxue.com/upload/attach/202507/815036_22Q2HBGUPHRPPPM.webp)

同理，再将0xffffffff81cdf496+6、0xffffffff81cdf496+5作为timeout地址，触发\_\_sys\_recvmmsg()执行超时处理逻辑，设置timeout->tv\_sec = 0，就可以将net\_sysctl\_root.permissions，从0xffffffff816d4ff0，改为0x000000ff816d4ff0。

刚开始，我有一个疑问，既然有timeout，为什么还要子进程执行sendto()，父进程的recvmmsg()才能退出阻塞？

通过[\_\_sys\_recvmmsg()代码](https://bbs.kanxue.com/elink@5cdK9s2c8@1M7s2y4Q4x3@1q4Q4x3V1k6Q4x3V1k6W2L8r3W2^5K9i4u0Q4x3X3g2T1L8$3!0@1L8r3W2F1i4K6u0W2j5$3!0E0i4K6u0r3L8r3W2F1N6i4S2Q4x3V1k6$3x3#2)9J5k6e0p5I4i4K6u0r3M7$3!0#2M7X3y4W2i4K6u0r3L8X3g2@1i4K6u0r3M7$3!0U0K9$3g2@1i4K6u0W2j5#2)9J5x3@1H3J5x3K6l9@1)可以看出，它的内部是个循环，直到接收vlen个msg，而每次循环都是阻塞的，必须收到msg才会退出阻塞，timeout只是为了让\_\_sys\_recvmmsg()，不用必须等到收满vlen个msg才返回，用于缓解收到少量msg，但长时间没有继续收到msg时，已经收到的msg也得不到处理。

(2) 向net\_sysctl\_root.permissions指向的用户空间，填充提权指令：

```

填充逻辑比较简单，就是调用mmap()，分配0x000000ff816d4ff0所在的虚拟页面，并设置可执行权限，然后将以上代码编译出来的指令，复制进去，其中，commit\_creds和prepare\_kernel\_cred都是函数指针，后续在触发提权代码执行之前，会让它们指向内核中的commit\_creds()和prepare\_kernel\_cred()函数。
