  X = $A934C0Af ;
  Y = $2E ;
  Z = $459 ;

  s,s1,s2:ShortString;
  i:SmallInt;
  a,a1,b,c,d :DWORD;

  s:=Edit1.Text;
  s1:=Copy(Edit2.Text,1,6);
  s2:=Copy(Edit2.Text,7,6);

    a1:=a;
    a:= a + X;

      b:=b + y;
