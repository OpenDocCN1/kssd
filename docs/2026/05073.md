BOOL guid2wstr(GUID *guid,PWCHAR pwguid)
{
	int i = 0;
	//FIX ME:WHEN guid OR pwguid == NULL!
	*pwguid = L'{';
	pwguid += 1;

	for(;i < 0x14;i++,pwguid++)
	{
		if(guidtowstr[i] != '-')
		{
			*pwguid = UnicodeNumber[((DWORD)(*((BYTE *)guid + guidtowstr[i]))) / 16];
			pwguid++;
			*pwguid = UnicodeNumber[((DWORD)(*((BYTE *)guid + guidtowstr[i]))) & 0xf];
		}
		else
		{
			*pwguid = L'-';
		}
	}

	*pwguid = L'}';
	*(pwguid + 1) = 0;

	return TRUE;
}

	return pwFinalPath;
}
