## 简单反汇编代码如下：

```cpp
#include <iostream>

using namespace std;

void swap(int &,int &);

int main()

{

     int a=19,b=24;

     swap(a,b);

     cout<<a<<endl<<b<<endl;

     return 0;

}

void swap(int & i,int & j)

{

     int temp;

     temp=i;

     i=j;

     j=temp;

}
```
