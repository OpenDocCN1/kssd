var StartCode //保存asm 代码起点
var tem  //用做临时变量

mov tem,eip
add tem,24
bp tem //当查代代码结束时断下
eob le1 //当发生中断时执行这里的脚本
run

le1:
mov tem,StartCode
add tem,19
cmp eip,tem //中断时比较是否为查加密代码处
je break0

cmp eip,MyCode//执行壳里面的代码之后会在这里
je break1

mov tem,StartCode
add tem,24
cmp eip,tem //比较搜索是否结束
je break2
mov eax,3
ret

break0:
mov MyCode,edx
mov eip,edx
sti //跟入壳代码
go edx //跳出来程序代码，壳会修改这里的地址。
break1:
sti //进入，进入解密后的call，
