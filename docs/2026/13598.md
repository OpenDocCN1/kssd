{

        NULL,
        NULL,
        OPEN_EXISTING,
        FILE_ATTRIBUTE_NORMAL,
        NULL

    //2.获取文件大小

    {
        printf("%d   %u.%u.%u.%u :%u  \n", i,

        Service++;
        i++;
    }
