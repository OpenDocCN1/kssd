//TEA算法：

void tiny_encrypt(unsigned long *const v, unsigned long *const w,
const unsigned long *const k)
{
register unsigned long
y = v[0],
z = v[1],
a = k[0],
b = k[1],
c = k[2],
d = k[3],
n = TEA_ROUNDS,
sum = 0,
delta = TEA_DELTA;

while (n-- > 0) {
sum += delta;
y += (z << 4);
y +=a ^ z ;
y +=sum ^ (z >> 5);
y += b;
z += (y << 4);
z += c ^ y ;
z += sum ^ (y >> 5);
z += d;
}
w[0] = y;
w[1] = z;
}
