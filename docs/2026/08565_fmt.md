
# [原创]anti wmware

```
#include <windows.h>

#pragma comment(linker, "/entry:main")

int main()

{
        __asm
        {
                mov                dx, 1

_l1:
                verr        dx                                //其中的 dx, 可以为其他。只要 dx & 4 == TRUE, 那么就会在 vmware 中发生异常，而外部没有发现这种现象

                shl                dx, 1

                jmp                _l1
        }

        return 0;
}
```
