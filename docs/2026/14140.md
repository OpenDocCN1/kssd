parserByte@

when (index) {

result = cur.toLong()

    var result: Long = 0L
    parserByte@
    for (index in data.indices) {
        val cur = data[index].toUInt()
        when (index) {
            0 -> {
                result = cur.toLong()
                if (cur <= 127u) {
                    //就一个byte，直接赋值，跳出循环
                    break@parserByte
                }
                //最高位为1,继续保存值到result里面，继续下一次循环
