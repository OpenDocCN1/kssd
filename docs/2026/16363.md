import os

def getMD5(str):
    hash = hashlib.md5()
    hash.update(bytes(str, encoding='utf-8'))
    return hash.hexdigest()

def check(username):
    md = getMD5(username)
    md_str = ""
    for index in range(0, len(md), 2):
        md_str = md_str + md[index]
    return md_str
