procedure TfrmMain.Button1Click(Sender: PObj);
var serial,machinecode:integer;
begin
  machinecode:=str2int(editbox1.text);
  serial:=machinecode xor ($111a7b-(machinecode mod $a));
  editbox2.text:=int2str(serial);
end;
