前言:
  上次说了一种动态的检测方案，这次则说说网游中比较常见的反外挂解决方案吧。一般来说有些公司有钱有技术，投入人力物力自己研发一套反外挂系统出来自己用。而有些公司技术实力没这么强，没有相关的专业人员，就会购买现成的反外挂系统来使用，比如常见的nProtect GameGuard,HackShield,XignCode3等等。又有些公司会出于技术沉淀或某些原因，既会购买成熟的反外挂系统，也会投入精力自己进行研发反外挂系统。今天要讲的就是一个基于nProctect + 自研发反外挂 的双层反外挂系统。
正文:
  开始之前有必要谈谈nProctect,这个反外挂系统是韩国人开发的，目前应该来说应算得上流行的吧，我先简述下这东西是怎么用的。首先你自己得搭建一个用于他更新版本的HTTP服务器，然后把FTP账户密码发给他们的技术人员，他们会往上面上传网页文件，等到他们上传好后，他们的人会通知你叫你把client跟server端的编译环境信息发给他们，他们会发把用于client跟server用的lib文件还有ini文件发给你，然后就可以开始写代码测试啦。

```
7.1.2.为什么需要CS认证

 	当GameGuard 被强行终止，游戏用户试图运用被修改的客户端连接服务器时；
```

所谓的cs模块，原来就是np客户端跟np服务端的网络通讯模块，当你打开这项功能后，就算你把np客户端给破坏掉了(比如我直接把上文中的initializeGameGuard函数给nop掉 了)，np服务端会检测到没有通讯了，也会将你从游戏中踢出断线。

```
//////////////////////////////////////////////////////////////////////////////////
void CProtectGameGuard::OnTimer( unsigned long dwTimerID )
{
	if ( mNPGameLib==0 )
	{
		return ;
	}

	// 检查错误
	if ( GetTickCount()-mLastCheckTime>1000*10 )

		case NPGAMEMON_SPEEDHACK:
			MessageBox( (HWND)gGlobalClient->getHWND(),my_utf82t(_GT("nProtect GameGuard检测到加速程序,请关闭后再启动游戏")),my_utf82t(_GT("防外挂保护")),MB_OK );
			break;
		case NPGAMEMON_GAMEHACK_KILLED:
		case NPGAMEMON_GAMEHACK_DETECT:
			MessageBox( (HWND)gGlobalClient->getHWND(),my_utf82t(_GT("nProtect GameGuard检测到非法程序,请关闭可疑程序后再启动游戏")),my_utf82t(_GT("防外挂保护")),MB_OK );
			break;
		case NPGAMEMON_GAMEHACK_DOUBT:
			MessageBox( (HWND)gGlobalClient->getHWND(),my_utf82t(_GT("nProtect GameGuard检测到可疑程序,请关闭后再启动游戏")),my_utf82t(_GT("防外挂保护")),MB_OK );

			break;
```

定时器里面就干了2件的事情:
1.        检查np的状态是否正常。
2.        定时发送cs回应包给服务端。
客户端是先收到心跳包后再发送回应包的，我们接着看当客户端收到了cs心跳包的时候是怎么处理的

```
//////////////////////////////////////////////////////////////////////////////////
void CProtectGameGuard::onMessage(ulong actionId, SGameMsgHead* head, void* data, size_t len)
{
	if ( mNPGameLib==0 )
	{
		return;
	}

	switch( head->wKeyAction )
	{
	case MSG_GATEWAY_GAMEGUARD_AUTH:
		{
		if ( len!=sizeof(GG_AUTH_DATA) || data==0 )
			return;

		GG_AUTH_DATA * pAuthData = (GG_AUTH_DATA *)data;
		mNPGameLib->Auth2(pAuthData);
		}
		break;
	default:
		break;
	}
}
```


把接收到的消息交给np的Auth2函数处理,然后NP就会自动在你的回调函数中通知你要回应的心跳包是什么了，如下:

```
//////////////////////////////////////////////////////////////////////////////////////////////////////
// nProtect保护回调
//////////////////////////////////////////////////////////////////////////////////////////////////////
BOOL CALLBACK NPGameMonCallback(DWORD dwMsg, DWORD dwArg)
{
	char szHackInfo[MAX_PATH] = {0,};
	strcpy(szHackInfo, gGlobalClient->getnProtectGameGuard()->GetInfo());

	switch (dwMsg)
	{
	case NPGAMEMON_COMM_CLOSE:
		return FALSE;
	case NPGAMEMON_COMM_ERROR:
	case NPGAMEMON_INIT_ERROR:
	case NPGAMEMON_SPEEDHACK:
	case NPGAMEMON_GAMEHACK_KILLED:
	case NPGAMEMON_GAMEHACK_DETECT:
	case NPGAMEMON_GAMEHACK_DOUBT:
		CProtectGameGuard::getInstance().SetErrorCode(dwMsg);
		break;
	case NPGAMEMON_GAMEHACK_REPORT:
		{
			DWORD dwHackInfoSize = 0;
			LPBYTE pHackInfo = NULL;
			pHackInfo = gGlobalClient->getnProtectGameGuard()->GetHackInfo(&dwHackInfoSize); // C++ 老 版快.
			if (pHackInfo && dwHackInfoSize > 0)
			{
				CProtectGameGuard::getInstance().SendLogToServer( dwArg,pHackInfo,dwHackInfoSize );
			}
			return TRUE;
		}

	case NPGAMEMON_CHECK_CSAUTH2:
		{
			GG_AUTH_DATA GGAnswer;

			// 3. NPGameMonCallback 窃荐甫 烹秦辑 函版等 菩哦阑 罐澜
			// 3. Receiving modified packet from NPGameMonCallback function
			memcpy( &GGAnswer, (PVOID)dwArg, sizeof( GG_AUTH_DATA ) );

			// 向服务器发送认证数据包，这时应同步callback 函数，因为GameGuard正在使用该函数。

			obuf128 * pBuf = new obuf128;
			obuf128 & buf = *pBuf;

			SGameMsgHead header;
			header.SrcEndPoint= MSG_ENDPOINT_CLIENT;
			header.DestEndPoint = MSG_ENDPOINT_GATEWAY;
			header.wKeyModule = MSG_MODULEID_GAMEGUARD;
			header.wKeyAction = MSG_GATEWAY_GAMEGUARD_AUTH;

			buf << header;
			buf << GGAnswer;

			// 只能在主线程发送数据包
			CProtectGameGuard::getInstance().RequestSendData(pBuf);
			return TRUE;
		}
	}

	return TRUE;
}
```

```
// 检测结果
struct SDetectResult
{
	SDetectInfo detectInfo;			// 检测信息
	SDetectRecord detectRecord;		// 检测记录
};

// 检测接口
struct IDetect
{
	/*******************************************************************************
	函数名称：Execute
	函数描述：执行检测器
	输入参数：listDetectRecord 检测记录列表
	输出参数：detectResult 检测结果信息
	返回值	：检测到：true，未检测到：false
	备注	：N/A
	*******************************************************************************/
	virtual bool Execute(list<SDetectRecord> &listDetectRecord, SDetectResult &detectResult) = 0;

	/*******************************************************************************
	函数名称：Release
	函数描述：释放函数
	输入参数：N/A
	输出参数：N/A
	返回值	：void
	备注	：N/A
	*******************************************************************************/
	virtual void Release() = 0;
};
```


其中检测器的实现使用了命令模式，每个具体的检测器实现都必须继承该类，
