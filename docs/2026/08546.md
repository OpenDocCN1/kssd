```
void Timetrap()
{
	//VMBEGIN
	DWORD nTime = 0;
	_asm //GetTickCount
	{
		mov edx,0x7FFE0000
		mov eax,dword ptr ds:[edx]
		mul dword ptr ds:[edx+4]
		shrd eax,edx,0x18
		mov nTime,eax
	}

HA:
	printf("Hello World!\n");

	_asm //GetTickCount
	{
		mov edx,0x7FFE0000
		mov eax,dword ptr ds:[edx]
		mul dword ptr ds:[edx+4]
		shrd eax,edx,0x18
		sub eax,nTime
		cmp eax,1000
		jg  HA
	}

	// 重要的一些操作.

	_asm //GetTickCount
	{
		mov edx,0x7FFE0000
		mov eax,dword ptr ds:[edx]
		mul dword ptr ds:[edx+4]
		shrd eax,edx,0x18
		sub eax,nTime
		cmp eax,1000
		jg  Byebye
	}
	return;
Byebye:
	ExitProcess(-1);
	//VMEND
}

int main(int argc, char* argv[])
{
	Timetrap();
	getch();
	return 0;
}
```
