tmp1 = tmp2 = gmpy2.powmod(2, q, p)

for x in xrange(1, 0xFFFFFF):

print "x:", x

tmp1 = x * q - 1

while True:

    if tmp1 % n == 0:

        t = tmp1 / n

    n += 1

print "n:", n

m = p - t + 1

y = (m - gmpy2.isqrt(m*m - 4*p)) / 2
