    for ( int i = 0; i <= iCount; i ++ )
    {
      DWORD dwAddr = ReadDword(dwStart + i \* 4);
      if ( dwAddr != 0 )
      {
        while ( dwAddr != 0 )
        {
          DWORD dwPoint = ReadDword(dwAddr + 4);
          if ( dwPoint != 0 )
          {
            TASK\_INFO TI = {0};
