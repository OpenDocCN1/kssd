\_ProcDlg       proc     uses   ebx edi esi hWnd,wMsg,wParam,lParam




                        mov     eax,wMsg
                        .if     eax == WM\_TIMER
                                mov eax,Every
                                xchg dwTime,esi
                                inc esi
                                xchg dwTime,esi
                                cmp eax,dwTime
                                jne  over
                                je  baidu

                          baidu:

                                      invoke ShellExecute,NULL,NULL ,addr szBuff,NULL,NULL,SW\_SHOW
                                      mov  eax,30h
                                      mov  dwTime,eax

                         over:   xor eax,eax

                        .elseif     eax ==   WM\_CLOSE
                                invoke   ExitProcess,NULL

                        .elseif eax ==   WM\_INITDIALOG
                                mov      esi,30h
                                mov      dwTime,esi
                                invoke   LoadIcon,hInstance,ICO\_MAIN
                                invoke   SendMessage,hWnd,WM\_SETICON,ICON\_BIG,eax


                        .elseif eax ==   WM\_COMMAND
                                mov      eax,wParam

                                         invoke  EnableWindow,hIDC\_TIME,TRUE

                                         mov     eax,30h

                                         .if   IDC\_FLAGS==IDC\_ON
                                               xor  eax,eax
                                               jmp  last
                                            .endif

                                           mov     ebx,IDC\_ON
                                           mov     IDC\_FLAGS,ebx
                                         invoke  GetDlgItemText,hWnd,IDC\_TIME,addr szBuffer,sizeof szBuffer
