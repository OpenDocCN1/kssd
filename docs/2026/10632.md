# [原创]C++类成员指针调用

```
#include <iostream>
template<typename dst\_type, typename src\_type>
dst\_type pointer\_cast(src\_type src)
{
return \*static\_cast<dst\_type\*>(static\_cast<void\*>(&src));
}
class aa;
typedef int (aa::\*pfun)(int);
class aa
{
public:
aa()
{
}
int push(void\* f)
{
m\_fun = f;
return 0;
}
int afun(int a)
{
pfun f = pointer\_cast<pfun>(m\_fun);
return (this->\*f)(a);
}
public:
void\* m\_fun;
};
class bb : public aa
{
public:
int bfun(int a)
{
printf("%d\n", a);
return 0;
}
};
int main()
{
bb b;
b.push(pointer\_cast<void\*>(&bb::bfun));
aa\* a = &b;
a->afun(5);
return 0;
}
```
