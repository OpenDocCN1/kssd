
# [原创]C++类成员指针调用

```cpp
#include <iostream>
template<typename dst_type, typename src_type>
dst_type pointer_cast(src_type src)
{
return *static_cast<dst_type*>(static_cast<void*>(&src));
}
class aa;
typedef int (aa::*pfun)(int);
class aa
{
public:
aa()
{
}
int push(void* f)
{
m_fun = f;
return 0;
}
int afun(int a)
{
pfun f = pointer_cast<pfun>(m_fun);
return (this->*f)(a);
}
public:
void* m_fun;
};
class bb : public aa
{
public:
int bfun(int a)
{
printf("%d\n", a);
return 0;
}
};
int main()
{
bb b;
b.push(pointer_cast<void*>(&bb::bfun));
aa* a = &b;
a->afun(5);
return 0;
}
```
