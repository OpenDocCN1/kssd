# [原创]分享一个MASM文本转换为DWORD的函数

stringtodw proc string:dword,strsiz:dword
;日期:2014/12/23
;用处:字符串数字无差转换成DWORD
;作者：落笔飞花笑百生
xor eax,eax
mov edi,string
xor ebx ,ebx
xor esi,esi

mov ecx,strsiz

fuckmm:
MOVZX ESI,BYTE PTR DS:[EDI]
cmp esi,0
je close
LEA EAX,DWORD PTR DS:[EBX+EBX\*4]
LEA EBX,DWORD PTR DS:[ESI+EAX\*2-30h]
INC EDI
loop fuckmm
close:
mov eax,ebx

ret 8
stringtodw endp
