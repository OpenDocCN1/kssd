    说个`IDA F5`复制代码的小技巧: 尽量使用`BYTE`类型, 而不是`char`类型, 因为`char`类型为数组的下标时, 会被当成4个字节, 其余3个字节是未知的,

# include <stdio.h>

# include <string.h>

#include  <atlstr.h>

#include <windows.h>



typedef unsigned int uint;

uint POLYNOMIAL = 0xEDB88320;

int have_table = 0;

uint table[256];



void make_table()

{

	int i = 0, j = 0, crc = 0;

	have_table = 1;

	for (i = 0; i < 256; i++)

		for (j = 0, table[i] = i; j < 8; j++)

			table[i] = (table[i] >> 1) ^ ((table[i] & 1) ? POLYNOMIAL : 0);

}



uint crc32(uint crc, char *buff, int len)

{

	if (!have_table) make_table();

	crc = ~crc;

	for (int i = 0; i < len; i++)

		crc = (crc >> 8) ^ table[(crc ^ buff[i]) & 0xff];

	return ~crc;

}

int GenerateKeyTable(char* szKey, signed int nKeyLen, BYTE* szKeyTable)

{

	signed int v3 = 0; // eax@1

	BYTE *v4 = 0; // ebp@1

	int result = 0; // eax@3

	BYTE v6 = 0; // esi@4

	BYTE *v7 = 0; // ebx@4

	bool v8 = 0; // zf@5

	signed int v9 = 0; // [sp+4h] [bp-8h]@4

	unsigned __int8 v10 = 0; // [sp+18h] [bp+Ch]@4

	{

		*(BYTE *)(v3 + szKeyTable) = v3;

		++v3;
