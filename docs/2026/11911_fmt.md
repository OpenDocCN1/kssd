
# [原创]关闭或打开系统资源管理器的自带托盘图标

Vista开始资源管理器自带几个托盘图标，网络、音量、电源等，win7加入了操作中心，这个挺麻烦的，会提示一些二笔的东西，我最近想把操作中心的图标隐藏，所以就有了这文章。

![](https://bbs.kanxue.com/upload/attach/201306/402913_oqa7k7khe79l998.jpg)

本来想搞操作中心在控制面板里面的那些勾选，我全取消了不得了，图标也正常了，就仔细看了下，是RPC来通知的，二笔的是Client和Server都在`explorer`内，对比了一下，win7和win8的接口不同，参数少一个，不通用算了，就不搞取消勾选了。

从托盘图标入手的话，我发现系统可以关闭这些图标，就搞了下`explorer`，发现方法还是老方法。。。

附加`explorer`，然后在任务栏托盘图标区右键属性，弹出那个图标管理的窗口，选择打开或关闭系统图标，然后`OD`在模块列表中看到`taskbarcpl.dll`，这个就是这个窗口的处理COM模块，`shell32`使用`CoCreateInstance`载入的。在模块右键查找所有调用，给`Combox`的`dui70.GetSelection`下F2

回到窗口在`Combox`的下拉选择开启或关闭，`OD`会断下来，下面那个就是关键`call`。

![](https://bbs.kanxue.com/upload/attach/201306/402913_s4k4g9fwo64lptz.gif)

跟进入后是这样的：

```
6F4752A7    PUSH EBP
6F4752A8    MOV EBP,ESP
6F4752AA    MOV ECX,taskbarc.6F4716C8
6F4752AF    OP EBP
6F4752B0    JMP taskbarc.6F4754A2
```

`MOV ECX,taskbarc.6F4716C8`，这个`ecx`是`GUID`，然后就跳到生成结构函数`taskbarc.6F4754A2`，最后给`ShellTrayWnd`发`COPYDATA`完成操作：

![](https://bbs.kanxue.com/upload/attach/201306/402913_mmkgo1se65z7dq3.gif)

C语言代码如下，就不发附件了：

```c
typedef enum _SYSTEM_TRAYAREA_NOTIFICATION_ICON_TYPE{

  SYS_TRAYICON_CLOCK,

  SYS_TRAYICON_VOLUME,

  SYS_TRAYICON_NETWORK,

  SYS_TRAYICON_POWER,

  SYS_TRAYICON_ACTIONCENTER,

  SYS_TRAYICON_INPUTFLAG

}SYSTEM_TRAYAREA_NOTIFICATION_ICON_TYPE,*PSYSTEM_TRAYAREA_NOTIFICATION_ICON_TYPE;



typedef struct _SYSTEM_TRAYAREA_NOTIFICATION_ICON_STATE{

  BOOL bVisible;

  GUID guidIcon;

}SYSTEM_TRAYAREA_NOTIFICATION_ICON_STATE,*PSYSTEM_TRAYAREA_NOTIFICATION_ICON_STATE;



__forceinline void MakeFESystemTrayIconGUID(PSYSTEM_TRAYAREA_NOTIFICATION_ICON_STATE pState,SYSTEM_TRAYAREA_NOTIFICATION_ICON_TYPE eType)

{

  switch (eType){

  case SYS_TRAYICON_CLOCK:

    IIDFromString(L"{7820AE72-23E3-4229-82C1-E41CB67D5B9C}",&pState->guidIcon);

    break;

  case SYS_TRAYICON_VOLUME:

    IIDFromString(L"{7820AE73-23E3-4229-82C1-E41CB67D5B9C}",&pState->guidIcon);

    break;

  case SYS_TRAYICON_NETWORK:

    IIDFromString(L"{7820AE74-23E3-4229-82C1-E41CB67D5B9C}",&pState->guidIcon);

    break;

  case SYS_TRAYICON_POWER:

    IIDFromString(L"{7820AE75-23E3-4229-82C1-E41CB67D5B9C}",&pState->guidIcon);

    break;

  case SYS_TRAYICON_ACTIONCENTER:

    IIDFromString(L"{7820AE76-23E3-4229-82C1-E41CB67D5B9C}",&pState->guidIcon);

    break;

  case SYS_TRAYICON_INPUTFLAG:

    IIDFromString(L"{A59B00B9-F6CD-4FED-A1DC-0F4064A12831}",&pState->guidIcon);

    break;

  }

}



VOID WINAPI ChangeFileExplorerSysTrayIconVisible(BOOL bVisible,SYSTEM_TRAYAREA_NOTIFICATION_ICON_TYPE eType)

{

  COPYDATASTRUCT cds;

  SYSTEM_TRAYAREA_NOTIFICATION_ICON_STATE sts;

  cds.dwData = 4;

  cds.lpData = &sts;

  cds.cbData = sizeof(SYSTEM_TRAYAREA_NOTIFICATION_ICON_STATE);

  sts.bVisible = bVisible;

  MakeFESystemTrayIconGUID(&sts,eType);

  HWND hTrayWnd = FindWindow(_T("Shell_TrayWnd"),NULL);

  if (IsWindow(hTrayWnd)) SendMessage(hTrayWnd,WM_COPYDATA,NULL,(LPARAM)&cds);

}
```
