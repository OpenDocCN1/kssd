DWORD CRC32_Modify( char *hashBuf, int len )
{
    int i;
    register DWORD checksum = 0xffffffff;
    register DWORD temp = 0x0;
    for( i = 0; len; len--, i++ )
        UpdCRC32( hashBuf[i], checksum );
    return checksum;
}
