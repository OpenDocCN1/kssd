```
def decrypt_main_exe(startva = 0x401000):
    eax = Dword(startva)
    #alreay decrypted?
    if eax ==0:
        return
    counter = 0x6106
    while True:
        if counter < 4:
            break
        if (counter > 0x13c) and (counter<0x182):
            startva = startva + 4
            counter = counter -4
        else:
            dword_value = Dword(startva)
            dword_value = dword_value ^ eax
            PatchDword(startva,dword_value)
            startva = startva + 4
            counter = counter -4
```
