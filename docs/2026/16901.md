$id=preg\_replace('/[-]/',"",$id);        //strip out -

$id=preg\_replace('/[#]/',"",$id);        //strip out #

$id=preg\_replace('/[\s]/',"",$id);        //strip out space

$id=preg\_replace('/[\/\\\\]/',"",$id);        //strip out slashes
