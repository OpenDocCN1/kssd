
# [原创][讨论]连连看外挂寻路算法

最近心情不太好，就想发泄一下，于是qq连连看就成了我的目标（能不能不要在意这些细节。。。），然后就搞了个秒杀外挂。外挂好搞，今天主要是来讨论一下连连看的寻路算法。

正文开始。连连看这游戏的规则就是两个相同的棋子之间的连线不能超过2折，如图所示：

![](https://bbs.kanxue.com/upload/attach/201310/340482_1ibd2nmqby8rlr9.png)

这其实意味着两颗棋子间的所有可能连通路径都被限制在了以这两个棋子为交叉点的一个#字中，如图：

![](https://bbs.kanxue.com/upload/attach/201310/340482_n570rnbpvowqtq5.png)

空心圆表示没有棋子，黑色实心圆表示有棋子，而两个红色棋子的所有可能的连通路径都在这个“#”字中，因为一旦路径中有一点在“#”字之外，路径必会超过2折。

这样范围就缩小了，然后单看纵向：

![](https://bbs.kanxue.com/upload/attach/201310/340482_svwqzg4nmrbc04c.png)

在这张图中我用颜色标示出了两个红色棋子在自己那一列的纵向上的可连通区域（黄色和绿色），其中绿色是两颗棋子的可连通区域的交集，不难发现，如果绿色区域上的两个格子（同一行）没有障碍可连通则寻路成功，横向同理（你自己把图横过来看 = =）

所以寻路算法的思路也就明晰了：

1、在纵（横）向上分别找两颗棋子的可连通区域`A1`、`A2`

2、`A1`、`A2`取交集`A`

3、判断`A`中是否有可连通路径

```cpp
struct Chessman {
    int x,y;            //棋子在棋盘上的坐标（x，y）
    int pos_x,pos_y;    //棋子的屏幕坐标
    char cdata;          //棋子种类
};

struct ConnectibleArea {
    int upper_bound, lower_bound;    //公共可连通区域的上下界
};

bool IsAdjacent(const Chessman &cm1, const Chessman &cm2) {
    //如果两颗棋子相邻则返回真，否则返回假
    if (cm1.x + 1 == cm2.x && cm1.y == cm2.y) {
        return true;
    }
    if (cm1.x - 1 == cm2.x && cm1.y == cm2.y) {
        return true;
    }
    if (cm1.x == cm2.x && cm1.y + 1 == cm2.y) {
        return true;
    }
    if (cm1.x == cm2.x && cm1.y - 1 == cm2.y) {
        return true;
    }
    return false;
}

bool IsNullLine(const Chessman &cm1, const Chessman &cm2, const Chessman cm[Y_MAX][X_MAX]) {     //棋盘数据
    //两颗棋子必须同行或同列
    //两者x坐标相等，说明同列
    if (cm1.x == cm2.x) {
        //cm1和cm2相邻则返回真
        if (IsAdjacent(cm1, cm2)) {
            return true;
        }
        //cm1和cm2不相邻
        //i代表x坐标，j代表y坐标
        if (cm1.y < cm2.y) {   //cm1在cm2上方
            for (int j = cm1.y + 1; j < cm2.y; j++) {
                if (cm[j][cm1.x].data != 0) {
                    return false;
                }
            }
        }else {              //cm1在cm2下方
            for (int j = cm1.y - 1; j > cm2.y; j--) {
                if (cm[j][cm1.x].data != 0) {
                    return false;
                }
            }
        }
    }else {   //同行
        //cm1和cm2相邻则返回真
        if (IsAdjacent(cm1, cm2)) {
            return true;
        }
        //cm1和cm2不相邻
        //i代表x坐标，j代表y坐标
        if (cm1.x < cm2.x) {   //cm1在cm2左方
            for (int i = cm1.x + 1; i < cm2.x; i++) {
                if (cm[cm1.y][i].data != 0) {
                    return false;
                }
            }
        }else {              //cm1在cm2右方
            for (int i = cm1.x - 1; i > cm2.x; i--) {
                if (cm[cm1.y][i].data != 0) {
                    return false;
                }
            }
        }
    }
    return true;
}

void GetPublicConnectibleArea(const ConnectibleArea &ca1, const ConnectibleArea &ca2, ConnectibleArea &result_pca) {
    if (ca1.upper_bound > ca2.upper_bound) {
        result_pca.upper_bound = ca2.upper_bound;
    }else {
        result_pca.upper_bound = ca1.upper_bound;
    }

    if (ca1.lower_bound > ca2.lower_bound) {
        result_pca.lower_bound = ca1.lower_bound;
    }else {
        result_pca.lower_bound = ca2.lower_bound;
    }

    if (result_pca.lower_bound > result_pca.upper_bound) {
        result_pca.lower_bound = result_pca.upper_bound = -1;
    }
}

// 先找出这两颗棋子在x、y轴上的公共可连通区域，在判断这些可连通区域中是否有可行路径
bool IsConnectible2(const Chessman &cm1,const Chessman &cm2,const Chessman cm[Y_MAX][X_MAX]) {
    //若两颗棋子不同则返回假
    if (cm1.data != cm2.data) return false;
    //如果是同一颗棋子则返回假
    if (cm1.x == cm2.x && cm1.y == cm2.y) return false;
    //两颗相同棋子相邻，可消除
    if (IsAdjacent(cm1, cm2)) return true;
    // 两颗棋子可直接联通
    if ((cm1.x == cm2.x) || (cm1.y == cm2.y)) {
        if (IsNullLine(cm1, cm2, cm)) {
            return true;
        }
    }
    // 两颗棋子不可直接连通
    ConnectibleArea ca1, ca2, pca;
    // 先纵向
    if (cm1.x != cm2.x) {
        ca1.upper_bound = ca1.lower_bound = cm1.y;
        ca2.upper_bound = ca2.lower_bound = cm2.y;
        for (int i = cm1.y - 1; i >= 0; i--) {
            if (cm[i][cm1.x].data == 0) {
                ca1.lower_bound = i;
            }else break;
        }
        for (int i = cm1.y + 1; i < Y_MAX; i++) {
            if (cm[i][cm1.x].data == 0) {
                ca1.upper_bound = i;
            }else break;
        }
        for (int i = cm2.y - 1; i >= 0; i--) {
            if (cm[i][cm2.x].data == 0) {
                ca2.lower_bound = i;
            }else break;
        }
        for (int i = cm2.y + 1; i < Y_MAX; i++) {
            if (cm[i][cm2.x].data == 0) {
                ca2.upper_bound = i;
            }else break;
        }
        GetPublicConnectibleArea(ca1, ca2, pca);
        if ((pca.lower_bound != -1) && (pca.upper_bound != -1)) {
            for (int i = pca.lower_bound; i <= pca.upper_bound; i++) {
                if (IsNullLine(cm[i][cm1.x], cm[i][cm2.x], cm)) {
                    return true;
                }
            }
        }
    }
    // 横向
    if (cm1.y != cm2.y) {
        ca1.upper_bound = ca1.lower_bound = cm1.x;
        ca2.upper_bound = ca2.lower_bound = cm2.x;
        for (int i = cm1.x - 1; i >= 0; i--) {
            if (cm[cm1.y][i].data == 0) {
                ca1.lower_bound = i;
            }else break;
        }
        for (int i = cm1.x + 1; i < X_MAX; i++) {
            if (cm[cm1.y][i].data == 0) {
                ca1.upper_bound = i;
            }else break;
        }
        for (int i = cm2.x - 1; i >= 0; i--) {
            if (cm[cm2.y][i].data == 0) {
                ca2.lower_bound = i;
            }else break;
        }
        for (int i = cm2.x + 1; i < X_MAX; i++) {
            if (cm[cm2.y][i].data == 0) {
                ca2.upper_bound = i;
            }else break;
        }
        GetPublicConnectibleArea(ca1, ca2, pca);
        if ((pca.lower_bound != -1) && (pca.upper_bound != -1)) {
            for (int i = pca.lower_bound; i <= pca.upper_bound; i++) {
                if (IsNullLine(cm[cm1.y][i], cm[cm2.y][i], cm)) {
                    return true;
                }
            }
        }
    }
    return false;
}
```
