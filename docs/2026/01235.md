#include "iostream"
using std::cout;
using std::endl;
class a
{
  int m\_a;
public:
  a(int x):m\_a(x) {}
  virtual void display()
  {
    cout<<"a is running !"<<endl;
  }
};
class b : public a
{
public:
  b(int y):a(y) {}
  void display()
  {
    cout<<"b is running !"<<endl;
  }
};
class c : public b
{
public:
  c(int z):b(z) {}
  void display()
  {
    cout<<"c is running !"<<endl;
  }
};
int main()
{
  a ca(1);
  b cb(2);
  c cc(3);
  a \* p[3]={ &ca , &cb , &cc };
  p[0]->display();
  p[1]->display();
  p[2]->display();
  return 0;
}
