    LOCAL dwFileAlig : DWORD
    LOCAL dwSecAlig : DWORD

    mov esi, pMem
    add esi, dword ptr [esi+3ch]        ;PE头地址
    mov dwNTHeader, esi
    assume esi : ptr IMAGE_NT_HEADERS
    ;更改节块数目
    mov cx, word ptr [esi].FileHeader.NumberOfSections
    movzx ecx, cx
    inc word ptr [esi].FileHeader.NumberOfSections        ;节块数目加1
    push dword ptr [esi].OptionalHeader.FileAlignment        ;文件对齐值
    pop dwFileAlig
    push dword ptr [esi].OptionalHeader.SectionAlignment        ;节块对齐值
    pop dwSecAlig
    add esi, sizeof IMAGE_NT_HEADERS           ;令esi指向节块表section table
    mov eax, sizeof IMAGE_SECTION_HEADER        ;节块大小
    mov ebx, ecx        ;节块数目
    imul ebx                ;指向最后一块节块
    add esi, eax        ;esi为原始最后一节块结尾处的文件偏移地址
    push esi
    sub esi, sizeof IMAGE_SECTION_HEADER    ; esi为原始最后一节块起始处的文件偏移地址
    mov dwLastSecTbl, esi
    pop esi
    assume esi : ptr IMAGE_SECTION_HEADER
    ;设置节块名
    push esi
    lea edi, [esi].Name1
    mov esi, pSectionName
CopySectionNameLoop:
    lodsb
    test al, al
    jz EndCopySectionNameLoop
    stosb
    jmp CopySectionNameLoop
EndCopySectionNameLoop:
    pop esi

    push 0E00000E0h                ;设置节块属性为可读可写可执行
    pop dword ptr [esi].Characteristics

    push dwSectionSize                ;设置内存中节块大小
    pop dword ptr [esi].Misc.VirtualSize

    invoke _PEAlign,dwSectionSize,dwFileAlig        ;设置新增节块进行文件对齐后的大小
    mov dword ptr [esi].SizeOfRawData, eax

    ; 新节的内存偏移 = 上一节的内存偏移 + 上一节经过节对齐后的长度
    ; 新节的文件偏移 = 上一节的文件偏移 + 上一节进过文件对齐后的长度
    mov eax, dwLastSecTbl
    assume eax : ptr IMAGE_SECTION_HEADER
    mov ecx, dword ptr [eax].VirtualAddress                ; 新增节块的相对虚拟地址
    add ecx, dword ptr [eax].Misc.VirtualSize
    mov ebx, dword ptr [eax].PointerToRawData                ; 新增节块的文件偏移地址
    add ebx, dword ptr [eax].SizeOfRawData
    invoke _PEAlign, ecx, dwSecAlig                ;设置新增节块进行节块对齐后的偏移地址
    mov dword ptr [esi].VirtualAddress, eax
    invoke _PEAlign, ebx,dwFileAlig                ;设置新增节块进行文件对齐后的偏移地址
    mov dword ptr [esi].PointerToRawData, eax

    mov eax, dword ptr [esi].VirtualAddress
    add eax, dword ptr [esi].Misc.VirtualSize
    invoke _PEAlign, eax,dwSecAlig                ;设置新增节块进行块对齐后的大小+内存偏移
    mov edx, dwNTHeader
    assume edx : ptr IMAGE_NT_HEADERS
    mov dword ptr [edx].OptionalHeader.SizeOfImage, eax        ;修改PE文件的映射大小
    push dword ptr [esi].PointerToRawData                ;PE文件中最后节块的文件偏移
    pop edi
    add edi, pMem

    ;清零工作
    mov ecx, dwSectionSize
    xor eax, eax
    cld
    rep stosb

    mov eax, esi        ;返回值为新增节表的文件偏移
    assume esi : nothing
    assume eax : nothing
    assume edx : nothing
    ret
_AddSection endp

_PEAlign proc uses ecx edx, dwTarNum : DWORD, dwAlignTo : DWORD
    ; 用于计算节对齐后的大小
    ; Algorithms:
    ; $1 = dwTarNum / dwAlignTo
    ; if remain != 0
    ; $r = $1 + 1 * dwAlignTo
    ; return $r
    mov ecx, dwAlignTo
    mov eax, dwTarNum
    xor edx, edx
    div ecx
    cmp edx, 0
    jz AlreadyAligned
    inc eax
AlreadyAligned:
    mul ecx
    ret

_PEAlign endp
