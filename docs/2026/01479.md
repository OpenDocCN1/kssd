题目要求输入`XDCSC2012`作为name，同时得到的key1为`d4021e38`

那么这个key1就可以作为密钥解密`A2096DD3FE74B4EC`

在网上解得为`8018faaf`，这是password经过MD5加密后的一部分，写程序取password从一个字符开始，字符为字母数字，一次MD5加密，取5到8个字节与`8018faaf`相等成功。

```BOOL MD5Hash()
{
  MD5_CTX context;
  long i;
  int j,k,m;
  char szName[MAXINPUTLEN]={"QWERTYUIOPASDFGHJKLZXCVBNM1234567890qwertyuiopasdfghjklzxcvbnm"};
  char szHash[MAXINPUTLEN]={0};
  char szBuffer[MAXINPUTLEN]={0};
  char  duan[3]={0};
  char  buff[] = {"8018FAAF"};
  for (i=1;i<=2;i++)
  {
    for (j=0;j<62;j++)
    {
      duan[0] = szName[j];
      if (i==1)
      {
        MD5Init(&context);
        MD5Update(&context, duan, i);
```
