int Solve(char *a,int n)   
{   
	int dp[30];	
    dp[0]=1;   
    int ans=0;   
    for(int i=1;i<n;i++)   
    {   
        int max=0;   
        for(int j=0;j<i;j++)   
        {   
            if(a[i]>=a[j] && max<dp[j])   
                max=dp[j];   
        }   
        dp[i]=max+1;  
		                   
        if(ans<dp[i])     
            ans=dp[i];      
    }   
    return ans;   
}

}
