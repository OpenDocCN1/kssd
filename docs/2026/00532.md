VOID   PrintTree(MMVAD*   Root)
{
    ULONG  RetLen = 0;
    if (!Root)
        return ;

    __try
    {
        if (MmIsAddressValid(Root->Subsection) && MmIsAddressValid(Root->Subsection->ControlArea))
        {
            if (MmIsAddressValid((PVOID)Root->Subsection->ControlArea->FilePointer.Value))
            {
                PFILE_OBJECT  pFileObj = (PFILE_OBJECT)((Root->Subsection->ControlArea->FilePointer.Value >> 3) << 3);
                if(MmIsAddressValid(pFileObj))
                {
                    WCHAR* strWchar = 0;
                    if (MmIsAddressValid((PVOID)*(ULONG64*)((char*)pFileObj + 0x58)))
                        strWchar = (WCHAR*)*(ULONG64*)((char*)pFileObj + 0x58);
                    if (MmIsAddressValid((PVOID)*(ULONG64*)((char*)pFileObj + 0x60)))
                        strWchar = (WCHAR*)*(ULONG64*)((char*)pFileObj + 0x60);
                    //DbgPrintEx(101, 0, "Base:%08X\tSize:%08X\t", Root->Subsection->ControlArea->Segment->BasedAddress, (Root->Subsection->ControlArea->Segment->SizeOfSegment));
                    DbgPrintEx(101, 0, "Base:%08X\tEndingVpn:%08X\t", Root->StartingVpn<<12,Root->EndingVpn<<12+0x1000);
                    DbgPrintEx(101, 0, "Name:%ws\n", strWchar);
                }
            }
        }
    }
    __except (1)
    {
        DbgBreakPoint();
        DbgPrintEx(101,0,"Invalid?Address！\n");
    }

    __try
    {
        if (MmIsAddressValid(Root->LeftChild))
            PrintTree(Root->LeftChild);
        if (MmIsAddressValid(Root->RightChild))
            PrintTree(Root->RightChild);
    }
    __except (1)
    {
        DbgBreakPoint();
        DbgPrintEx(101, 0, "异常！！\n");
        return;
    }
}
