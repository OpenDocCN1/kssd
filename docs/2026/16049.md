struct input\_event ev0[64];

{

        int button , i, rd;

        button = 0;

        rd = read(event0\_fd, ev0, sizeof(struct input\_event) \* 64);

        printf("handle\_event0 rd = %d  \n",rd);


        if ( rd < sizeof(struct input\_event) )
        {
                 return 0;

        }

         printf("handle\_event0 read OK ...\n");

        for (i = 0; i < rd / sizeof(struct input\_event); i++)
        {
         if (ev0[i].type == 1&&ev0[i].value == 1)
                {
                        printf("Press Key: %3d\n", ev0[i].code);
              if (ev0[i].code == 158)
                           {
                                  printf("Press Key code: 158 ...\n");
                  return 0;
               }

     }

         printf("Press Key: test ...\n");

      return 1;
  }

  int main(int argc, char\* argv[])
