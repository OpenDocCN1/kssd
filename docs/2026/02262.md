### `rand proc`

```asm
rand    proc range
        LOCAL @stime:SYSTEMTIME
        pushad
        invoke GetSystemTime,addr @stime
        mov eax,range
        .if ah==al
        mov bl,al
        jmp rand_exit
        .endif
        mov bx,@stime.wMilliseconds
        mov ecx,1000
randlop:dec ecx
            jnz next
        mov bl,38h
        jmp rand_exit
 next:
            add bh,bl
        add bl,rand_init
        add bl,2
        cmp bl,al
        jb  randlop
        cmp bl,ah
        ja  randlop
rand_exit:
        add rand_init,3
        mov rand_out,bl
        popad
        ret
rand    endp
```
