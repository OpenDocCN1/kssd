//Flash转屏保1.0版注册机（delphi版）

{$IFDEF KOL_MCK}

  Controls, mckCtrls {$ENDIF};

{$ELSE}

{$I uses.inc}

  Windows, Messages, SysUtils, Classes, Graphics, Controls, Forms, Dialogs,

  mirror;

{$ENDIF}

type

  {$IFDEF KOL_MCK}

  {$I MCKfakeClasses.inc}

  PfrmMain = ^TfrmMain;

  TfrmMain = object(TObj)

    Form: PControl;

  {$ELSE not_KOL_MCK}

  TfrmMain = class(TForm)

  {$ENDIF KOL_MCK}

    KOLForm1: TKOLForm;

    KOLProject1: TKOLProject;

    Label1: TKOLLabel;

    Label2: TKOLLabel;

    EditBox2: TKOLEditBox;

    { Private declarations }

  public

    { Public declarations }

  end;

  frmMain {$IFDEF KOL_MCK} : PfrmMain {$ELSE} : TfrmMain {$ENDIF} ;

{$IFDEF KOL_MCK}

procedure NewfrmMain( var Result: PfrmMain; AParent: PControl );

{$ENDIF}

implementation

{$IFNDEF KOL_MCK} {$R *.DFM} {$ENDIF}

{$IFDEF KOL_MCK}

{$I Unit1_1.inc}

{$ENDIF}

procedure TfrmMain.Button1Click(Sender: PObj);

var serial,machinecode:integer;

begin

  machinecode:=str2int(editbox1.text);

  serial:=machinecode xor ($111a7b-(machinecode mod $a));

  editbox2.text:=int2str(serial);

end;

end.
