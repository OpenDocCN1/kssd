def basen_e(i, n):
    assert(n <= 62)
    res = ""
    while i > 0:
        res += base62c[i % n]
        i /= n
    return res

def basen_d(s, n):
    sums = 0
    mults = 1
    for i in range(len(s)):
        num = base62c.index(s[i])
        sums += num * mults
        mults *= n
    return sums

def compute(s):
    i = basen_d(s, 62)
    return basen_e(i, 18)

def inverse_compute(s):
    i = basen_d(s, 18)
    return basen_e(i, 62)
