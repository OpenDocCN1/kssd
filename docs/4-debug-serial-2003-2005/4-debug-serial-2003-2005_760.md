    00000 - 01001   ==> '0' - '9'

unsigned Sub2\_StringToInt(char \*data)  
{  
  int i,j;  
  int len = strlen(data); // 16  
  unsigned EAX, EBX;  
  
  EBX=0xFFFF;  
  for (i=0;i<len;i++)  
  {  
    EAX = (unsigned)data\[i\];  
    EBX ^= EAX;  
    for(j=0;j<8;j++)  
    {  
      if ((EBX&1)==0){  
        EBX >>=1;  
      }  
      else{  
        EBX >>=1;  
        EBX  ^= 0xA001;  
      }  
    }  
  }  
  return EBX;  
}
