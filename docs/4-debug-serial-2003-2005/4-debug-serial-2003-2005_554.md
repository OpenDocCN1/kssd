        hWnd:DWORD,wMsg:DWORD,wParam:DWORD,lParam:DWORD

;\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*cal.asm   
Cal    proc   
        PUSHAD   
          
        lea edi, Rc2   
        or ecx, 0ffffffffh   
        xor eax, eax   
        repnz scasb   
        not ecx   
        dec ecx   
        test ecx, ecx   
        jz no\_input   
        push ecx   
        ;   
        lea edi, Rc1   
        or ecx, 0ffffffffh   
        repnz scasb   
        not ecx   
        dec ecx   
        test ecx, ecx   
        jnz next\_1   
        pop ecx   
        jmp no\_input   
next\_1:   
        lea edi, Rc1   
        add edi, ecx   
        lea esi, Rc2   
        pop ecx   
        rep movsb   
        ;   
        ;main proc   
        ;   
        xor eax, eax   
        lea esi, Rc1   
        mov ebx, eax   
        inc ebx   
main\_loop:   
        lodsb   
        test al, al   
        jz next\_2   
        imul ebx, ebx, 022d6h   
        add ebx, eax   
        jmp main\_loop   
next\_2:   
        mov eax, ebx   
        cdq   
        xor eax, edx   
        sub eax, edx    ;done   
        invoke wsprintfA, addr Serial, addr format, eax   
no\_input:   
  
        POPAD   
        ret   
Cal    endp
