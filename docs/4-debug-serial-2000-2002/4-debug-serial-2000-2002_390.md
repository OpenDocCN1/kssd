void int2str(int n, char \*pt) {   
  int k = 0;   
  ldiv\_t lx;   
  lx.quot = n;   
  while (lx.quot != 0) {   
    lx = ldiv(lx.quot, 10L);   
    \*(pt+k) = lx.rem + 0x30;   
    k++;   
  }   
  \*(pt+k) = '\\0';   
  strrev(pt);
