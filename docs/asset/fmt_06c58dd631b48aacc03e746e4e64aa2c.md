main()
{

        ebx = i + 1;
        if (i >= d) ebx -= d;
        ecx = ebx;
        ecx *= ebx;
        eax = ecx;
        eax *= ebx;
        edi = ebx + 0x14;
        edi = eax % edi;
        eax = ecx;
        edx = ebx + 0xa;
        ecx = edx;
        edi += eax % ecx;
        eax = edi;
        edx = ebx;
        edx += edx;
        eax += edx;
        if (i < d) {
            eax += 8;
            eax += string[i];

            if (d > 10) {
                if (i == d) break;
            }
            if (d <= 10) {
                if (i == 10) break;
            }
            eax += 1;
            eax += c;
        }
        while (eax > 0x7a)
            eax -= 0xa;
        while (eax < 0x61)
            eax += 0xa;
        string[i] = eax;
    }
    printf("Your Register code is  : ");
    for (; i >= 0; i--)
        printf("%c", string[i - 1]);
    printf("\n  *******************************     ---     ---     ---\n *Welcome to WWW.CRACKNOW.COM*     /     /     / --\n *******************************     ---     ---     --/\n");
}
