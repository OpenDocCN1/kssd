用代码来表示如下：

```c
len=strlen(sName);
int num1[32]={0};
int num1len=0;
for(int i=0;i<len;i++){
    int tmp=i;
    tmp=tmp & 0x80000001;
    if(tmp==0){
        tmp=(i+1) % 0xFF;
        if(tmp!=0){
            num1[num1len]=tmp;
            num1len++;
        }
    }
}
```
