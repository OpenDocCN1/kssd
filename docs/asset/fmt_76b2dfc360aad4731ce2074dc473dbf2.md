#include<iostream>
using namespace std;

unsigned long Crc32Table[256];
void GetCRC32Table(){
   int i,j;
   unsigned long Crc;
   for (i = 0; i < 256; i++){
        Crc = i;
        for (j = 0; j < 8; j++){
            if (Crc & 1)
                Crc = (Crc >> 1) ^ 0xEDB88320;
            else
                Crc >>= 1;
        }
        Crc32Table[i] = Crc;
    }
}

//获取字符串 CRC32 校验值
unsigned int GetCrc32(char* InStr,int len){
    GetCRC32Table();
    unsigned int crc=0xffffffff;

    for(int i=0; i<len; i++){
      unsigned int tmp=InStr[i];
        crc = (crc >> 8) ^ Crc32Table[(crc & 0xFF) ^ tmp];
    }    
     return crc;
}
