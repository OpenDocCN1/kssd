```cpp
void Account_Key_3(string& sRealKey, string sKey)
{
  string sTempKey;
  int nLenth = sKey.length();
  int n;
  for ( n = 0; n < nLenth; n++)
  {
    switch (sKey[n])
    {
    case '0':
    case '1':
    case '2':
    case '3':
    case '4':
    case '5':
    case '6':
    case '7':
    case '8':
    case '9':
      sTempKey += sKey[n];
      break;
    default:
      break;
    }
  }

  nLenth = sTempKey.length();
  if ( nLenth > 0)
  {
    int nTemp = 1;
    char* pTempKey = (char*)sTempKey.c_str();
    do 
    {
      if ( (nTemp % 2) == 0 )
      {
        char c = *(char*)(pTempKey + nTemp - 1);
        sRealKey += c;
      }
      nTemp++;
      nLenth--;
    } while (nLenth);
  }
}
```
