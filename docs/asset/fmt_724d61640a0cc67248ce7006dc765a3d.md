```c
// 求模函数
int mod(int sourcenum, int modnum)
{
  int x = sourcenum; // 注意：原文此处变量名可能有误，根据逻辑应为初始化 x 为 sourcenum
  while(1)
  {
    if(x >= modnum)
      x = x - modnum;
    else
      break;
  }
  return x;
}
```
