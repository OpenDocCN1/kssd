        public int[,,] Grade  

        {  

            get { return grade; }  

            set { grade = value; }  

        }  

        public int[,] Map  

        {  

            get { return map; }  

            set { map = value; }  

        }  

        int turn;//当为 0 时；轮到黑旗，为 1 时轮到白旗，

        {  

            get { return turn; }  

            set { turn = value; }  

        }  

        /// <summary>  

        /// 初始化棋盘  

        /// </summary>  

        public void initMap()  

        {  

            for (int i = 0; i < 19; i++)  

            {  

                for (int j = 0; j < 19; j++)  

                    map[i, j] = 0;  

            }  

        }  

        //将权值初始化  

        public void initGrade()  

        {  

            for (int i = 0; i < 19; i++)  

            {  

                for(int j=0;j<19;j++)  

                    for (int k=0; k < 2; k++)  

                    {  

                        grade[i, j, k] = 0;  

                    }  

            }  

        }  

        public Five()  

        {  

            initMap();  

            initGrade();  

            Turn = 0;//黑棋先行  mus

            if (map[i, j] == 0)//五子棋只有在没有落子处可以下  

            {  

                map[i, j] = Turn+1;  

                //更新权值表  

                updateGrade(i, j);  

                Turn = (Turn + 1) % 2;  

            }
