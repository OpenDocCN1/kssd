## 5.注册机

检查注册码的结果保存在一个结构里面：

```
typedef struct result   {    
    unsigned short Time; //45
    //30959(1990.1),Not equal with 30E0B,
    //00030959=199001
    //0030E0B=200203,等于 200203 时要检验黑名单
    char  a ;            //46
    // ;<=0，设为 0
    char  t  ;           //47
    //54'T'
    char  e  ;           //48
    //45'E'
    char  v  ;           //49
    //19＝25，版本号为 2.5，
    char  flag ;         //4A
    //1,为 1 表示 true
    unsigned short   c;   //4B
    //01,01F4,2,63,3E8,7CF
    //；为 1 时，直接去比较时间是否为 200203
    //；2《＝x《＝63   good boy
    //；63《x《1F4 bad boy
    //；1F4《＝x《＝3E7 good boy
    //；3E8《＝x《＝7CF good boy
    //；》7CF bad boy
    //；为其他值时，时间必须不等于 200203
    int  remainder ;      //4C
    //0001C0E0 ,必须不在黑名单内
} RESULT;
```

算注册码调用 `keygen()`就行了

```
void keygen(void)
{
    RESULT r;
    unsigned long iCheckSum;
    UINT64 temp,base5B;
    int flag;

    char *p=serials;
    BOOL bOK;
    r.Time=0;
    r.a=0;
    r.t='T';
    r.e='E';
    r.v=(char)0x19;//2.5
    r.flag=1;
    r.c=1;

    do
    {
        r.remainder= (GetTickCount() * rand()) % 0x00989680;//生成随机数
        //r.remainder=0x1C0E0;
        iCheckSum=CheckSum(r);

        temp=r.remainder+((INT64)(iCheckSum))*0x00989680;
        base5B=temp*0x31;
        //base5B=0x000017bb3fbc5660;

        //判断 base5B 的合法性
        bOK=GetSerial(p,12,23,base5B);

        if(bOK)
        {
            flag=(iCheckSum & 0x3F) ^ (r.flag);

            temp=r.c+ flag * 0x7D0;

            base5B=temp*0x258;

            bOK= GetSerial(p,5,11,base5B);
        }
    }while(!bOK);

    bOK=!bOK;
}
```
