    cmp byte ptr [ecx+01], 50
    jne 0040E4FF

    jne 0040E4FF
    cmp byte ptr [ecx+03], 38

    add al, dl

    add al, dl
    mov dl, byte ptr [ecx+0F]

    add dl, 30
    cmp dl, 39

    mov byte ptr esp_04, dl

RA_0E40B:

;0040E40F:

;0040E466:
