Dim tmp As String

    Exit Sub

uname = Text1.Text

sn = &H5DFEE4A4

For I = 1 To Len(uname) - 4
    tmp = "&H" & Hex(Asc(Mid(uname, I + 3, 1))) & Hex(Asc(Mid(uname, I + 2, 1))) & Hex(Asc(Mid(uname, I + 1, 1))) & Hex(Asc(Mid(uname, I, 1)))
    sn = sn Xor Val(tmp)
Next I

Text2.Text = CStr(sn)
End Sub
