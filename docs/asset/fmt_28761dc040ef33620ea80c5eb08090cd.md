### 示例：Diffie-Hellman 密钥交换

```

q;

Ilog2 (q);                 // 2 进制位数

repeat
    got_it, p := RandomPrime (1024, 1, q, 1000 : Proof := false);
until got_it;

> p;

Ilog2 (p);                 // 2 进制位数

tmp := Random(p);

exp := (p - 1) div q;

> g := Modexp (tmp, exp, p);

> g;

Modexp (g, q, p);

A_priv := Random (p-1);

> A_pub := Modexp (g, A_priv, p);

> B_priv := Random (p-1);

> B_pub := Modexp (g, B_priv, p);

> A_sec := Modexp (B_pub, A_priv, p);

> B_sec := Modexp (A_pub, B_priv, p);

A_sec eq B_sec;
```
