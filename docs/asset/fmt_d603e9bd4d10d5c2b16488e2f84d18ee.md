```c
index = PasBuf;

  al = *LetterTable; // 字符表"23456789ABCDEFGHJKLMNPQRSTUVWXYZ"
  pTable = LetterTable;

    pTable++;

    while (count > 0) {
      if((pos & var) == 0) {
        cl = bl;
        *index &= ~cl;
      }
      else
        *index |= bl;
      var <<= 1;
      bl <<= 1;
      // 已经处理完 8bit
      if(bl == 0) {
        index++; // 处理下一字节
        bl = 1;   // 又从第一位开始处理

      count--;
    }
  }

}
```
