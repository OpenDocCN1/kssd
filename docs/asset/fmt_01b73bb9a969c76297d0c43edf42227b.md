```c
typedef union _DR6
{
  struct
  {
    unsigned B0     : 1;  // b0
    unsigned B1     : 1;  // b1
    unsigned B2     : 1;  // b2
    unsigned B3     : 1;  // b3
```

```c
typedef union _DR7
{
  struct
  {
    unsigned L0     : 1;  // b0
    unsigned G0     : 1;  // b1
    unsigned L1     : 1;  // b2
    unsigned G1     : 1;  // b3
    unsigned L2     : 1;  // b4
    unsigned G2     : 1;  // b5
    unsigned L3     : 1;  // b6
    unsigned G3     : 1;  // b7
    unsigned LE     : 1;
    unsigned GE     : 1;
    unsigned unused1  : 3;
    unsigned GD     : 1;  // b13
    unsigned unused2  : 2;
    unsigned RW0    : 2;  // b16-17

    unsigned Len2    : 2;  // b26-27
    unsigned RW3    : 2;  // b28-29
```
