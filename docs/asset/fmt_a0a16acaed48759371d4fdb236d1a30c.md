## 算法总结

用于计算的字符只取用户名的前 4 位，多的不取，但取得的长度为整个用户名的长度。

每个计算字符取它的 `asc` 码的十六进制除以它的位置数再除以用户名的长度×（这个字符十六进制＋3）

如，`name:abcd`

```
key1=hex(a)/1/len(name)*(hex(a)+3)
.
.
key4=hex(d)/4/len(name)*(hex(d)+3)
SN=key1+..+key4(把 4 个部份连起来)
```

```vb
'Visual Basic 6.0 注册机源码
'编译环境: Win2003sp1
'程序作者: KuNgBiM

Private Sub Command1_Click()

Dim name, Lname, key1, key2, key3, key4, sn As String
Dim a, b, c, d As String

name = Text1.Text  '获取用户名
Lname = Len(name)  '取用户名长度

If Lname < 4 Then  '判断用户名是否小于 4 位
Else

'取 ASC 码
a = Asc(Mid(name, 1, 1)) '第 1 位
b = Asc(Mid(name, 2, 1)) '第 2 位
c = Asc(Mid(name, 3, 1)) '第 3 位
d = Asc(Mid(name, 4, 1)) '第 4 位

'每段注册码运算
key1 = Hex(a / 1 / Lname * a + 3)  '第 1 段
key2 = Hex(b / 2 / Lname * b + 3)  '第 2 段
key3 = Hex(c / 3 / Lname * c + 3)  '第 3 段
key4 = Hex(d / 4 / Lname * d + 3)  '第 4 段

sn = key1 & key2 & key3 & key4     '把 4 段注册码连起来

End If

Text2.Text = sn '输出注册码

End Sub
```
