procedure THookClass.Restore;

var
  nCount: Cardinal;

begin
  if (hProcess = 0) or (OldFunction = nil) or (NewFunction = nil) then
    Exit;

  try
    WriteProcessMemory(hProcess, OldFunction, @(Oldcode), 5, nCount);

    showmessage('');
  end;

  Trapped := false;
