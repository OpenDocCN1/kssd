Function `SNcount(sn)`  'SNcount 就是计算注册码

`sn` = `sn` And 255        '取低 8 位

`sn` = `sn` Mod 35        '取余数

`sn` = `sn` - 10

If `sn` > 0 Then

    If `sn` - 26 > 0 Then

        `sn` = 45

    Else

        `sn` = `sn` + 65    '避免麻烦，把减去的 10 加在这里了

    End If

Else

    `sn` = `sn` + 58      '这里也是

End If

`sn` = `Chr(sn)`

End Function
