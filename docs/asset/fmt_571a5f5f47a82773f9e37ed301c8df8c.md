; posType：1 表示计算 x 坐标，2 表示计算 y 坐标

CalculatePos proc left : DWORD, top :DWORD, posType : DWORD
    local @ret : DWORD
    pushad
    mov edx, left
    add edx, rectArea ;右下角 x 坐标
    mov ecx, top
    add ecx, rectArea ;右下角 y 坐标
    .if posType == 1
        mov eax, edx
        mov ebx, 30
        xor edx, edx
        div ebx
    .elseif
        mov eax, ecx
        sub eax, topSpace
        mov ecx, 30
        xor edx, edx
        div ecx
    .endif
    mov @ret, eax
    popad
    mov eax, @ret
    dec eax
    ret
CalculatePos endp
