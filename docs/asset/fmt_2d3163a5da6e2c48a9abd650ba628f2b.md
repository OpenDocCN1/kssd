void __declspec(naked) InitWindow()  

{  

  __asm  

  {  

    pushad  

    mov    eax,0x4DC3D4  

    push  [eax]  

    call  InitOnce  

    call  FixWindowPos  

    popad  

    mov     eax, dword ptr [esi]  

    mov     ecx, esi  

    call    dword ptr [eax+0x23C]  

    push  0x452729  

    ret  

  }  

}  

void __declspec(naked) Entry()  

{  

  __asm  

  {  

    push  eax  

    push  ecx  

    mov    ecx,[esp+0x18]  

    mov    eax,[ecx+0x190]  

    cmp    eax,[ecx+0x1DC]  

    pop    ecx  

    pop    eax  

    jnz    lab_Normal  

    pushad  

    mov    eax,0x4DC3D4  

    push  [eax]  

    call  InitOnce  

    call  AutoIt  

    popad  

lab_Normal:  

    pop     esi  

    pop     ebp  

    retn    4  

    mov    eax,InitWindow  

  }  

}
