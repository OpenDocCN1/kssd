showeax proc  ;坚决拒绝使用 wsprintf 等类似函数。  
  pushad      ;坚决不使用 msvcrt.dll 等运行时库。  
  mov ecx,8   ;坚决杜绝.net 与.frameset 等环境。  
  xor esi,esi ;坚决杜绝 java 虚拟机。  
  againp:     ;此函数在驱动及 ring0 中也可以运行哟！  
  dec ecx  
  rol eax,4  
  push eax  
  and eax,0Fh  
  cmp eax,9  
  jg big  
  add eax,30h  
  mov \[bufferl+esi\],al  
  pop eax  
  inc esi  
  cmp ecx,0  
  jne againp  
  je showp  
  big:  
  add eax,37h  
  mov \[bufferl+esi\],al  
  pop eax  
  inc esi  
  cmp ecx,0  
  jne againp  
  showp:  
  popad  
ret
