

# IE 

> 原文：[`www.pediy.com/kssd/pediy05/pediy50276.htm`](https://www.pediy.com/kssd/pediy05/pediy50276.htm)

将 IE 安全”选项的“自定义”和“默认”"按钮恢复成可以按 （该按钮原来是灰色的，不可按） 

> .386   
> .model flat,stdcall   
> option casemap:none   
>   
>       include \\masm32\\include\\windows.inc   
>       include \\masm32\\include\\user32.inc   
>       include \\masm32\\include\\kernel32.inc   
>       include \\masm32\\include\\gdi32.inc   
>       include \\masm32\\include\\comctl32.inc   
>       include \\masm32\\include\\comdlg32.inc   
>       include \\masm32\\include\\shell32.inc   
>       include \\masm32\\include\\advapi32.inc   
>   
>       includelib \\masm32\\lib\\user32.lib   
>       includelib \\masm32\\lib\\kernel32.lib   
>       includelib \\masm32\\lib\\gdi32.lib   
>       includelib \\masm32\\lib\\comctl32.lib   
>       includelib \\masm32\\lib\\comdlg32.lib   
>       includelib \\masm32\\lib\\shell32.lib   
>       includelib \\masm32\\lib\\advapi32.lib   
>   
> .data   
> SubKey        db 'Software\\Policies\\Microsoft\\Internet Explorer',0   
> RegKey        db "Control Panel",0   
> MsgCap        db "Fix Your IE Ver 0.1      by WsCn",0   
> MsgOK          db "Fixed OK!",0   
> MsgFailed      db "Sorry, Fix failed Perhaps your IE is not been Locked！",0   
> lhHRootUser    dd 0   
> hInstance      dd 0   
>   
> .code   
> start:   
>   
> invoke GetModuleHandle,NULL   
> mov hInstance,eax   
> call ModifyIt   
> invoke ExitProcess,NULL   
>   
> ModifyIt Proc   
>       pushad   
>       invoke RegOpenKeyEx,HKEY\_CURRENT\_USER,ADDR SubKey,NULL,KEY\_ALL\_ACCESS,ADDR lhHRootUser   
>     .if !eax   
>         invoke RegDeleteKey,lhHRootUser,ADDR RegKey   
>         .if !eax   
>             invoke MessageBox,NULL,addr MsgOK,addr MsgCap,MB\_OK   
>             jmp Modi\_exit   
>         .endif   
>     .endif   
>         invoke MessageBox,NULL,addr MsgFailed,addr MsgCap,MB\_OK   
> Modi\_exit:   
>     invoke RegCloseKey,lhHRootUser   
>     popad   
>     ret   
> ModifyIt Endp   
>   
> end start   
>
