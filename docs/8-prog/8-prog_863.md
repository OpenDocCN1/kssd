这个是个测试版   
//创建一个在其它进程地址空间中运行的线程(也称:创建远程线程).  
        \[DllImport("kernel32.dll")\]  
        public static extern IntPtr CreateRemoteThread(IntPtr hwnd, int attrib, int size, int address, int par, int flags, int threadid);
